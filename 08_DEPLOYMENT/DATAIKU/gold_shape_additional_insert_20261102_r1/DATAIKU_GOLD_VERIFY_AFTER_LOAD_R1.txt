# Gold 형태 개발계 적재 R1 — Dataiku LEGAL_RAG Python/Jupyter 새 셀
# 관리 폴더 ehA8dpVm에 아래 ZIP 파일을 이름 그대로 업로드합니다.
ACTION = 'verify_publish'  # 전수검증: LOADED_UNVERIFIED 뒤 실행
OLD_JOB_DIR = ''  # 보통 비워둠: 이전 R11 작업 폴더를 자동으로 사용
RETRY_MISSING_BATCH = False  # 결과 불명 상태에서는 임의로 True로 바꾸지 마세요.

import collections
import hashlib
import io
import json
import os
from pathlib import Path, PurePosixPath
import subprocess
import sys
import tempfile
import zipfile

ZIP_NAME = 'DATAIKU_GOLD_SHAPE_20261102_R1.zip'
ZIP_SHA256 = 'c1df3b82c7e589500cc81ffd8c7e2b3794eee9d7b39472beb589d3151f538a2e'


def _run_gold_shape():
    if ACTION not in ('start','load','verify_publish'):
        raise ValueError('ACTION은 start / load / verify_publish 중 하나입니다.')
    import dataiku
    folder=dataiku.Folder('ehA8dpVm')
    matches=[p for p in folder.list_paths_in_partition() if PurePosixPath(p).name==ZIP_NAME]
    if len(matches)!=1:
        raise ValueError('관리 폴더 ehA8dpVm에서 정확한 ZIP 한 개를 찾지 못했습니다.')
    with folder.get_download_stream(matches[0]) as stream:
        raw=stream.read(16*1024**2+1)
    if len(raw)>16*1024**2 or hashlib.sha256(raw).hexdigest()!=ZIP_SHA256:
        raise ValueError('ZIP 해시가 다릅니다. 이 TXT와 함께 받은 ZIP을 사용하세요.')
    root=Path(tempfile.gettempdir())/('legal-gold-shape-'+ZIP_SHA256[:16])
    root.mkdir(exist_ok=True)
    with zipfile.ZipFile(io.BytesIO(raw)) as archive:
        names=archive.namelist()
        if len(names)!=len(set(names)) or sum(i.file_size for i in archive.infolist())>32*1024**2:
            raise ValueError('ZIP 구성/크기 오류')
        for item in archive.infolist():
            name=PurePosixPath(item.filename)
            if name.is_absolute() or '..' in name.parts or item.is_dir() or (item.external_attr>>16)&0o170000==0o120000:
                raise ValueError('ZIP 경로 오류')
            path=root.joinpath(*name.parts)
            if root.resolve() not in path.resolve().parents:
                raise ValueError('ZIP 범위 오류')
            content=archive.read(item)
            if path.exists():
                if path.read_bytes()!=content:
                    raise ValueError('기존 추출 코드가 다릅니다. 실행을 중단합니다.')
            else:
                path.parent.mkdir(parents=True,exist_ok=True)
                with path.open('xb') as f:f.write(content)
        manifest=json.loads(archive.read('CODE_MANIFEST.json'))
        if set(names)!=set(manifest['files'])|{'CODE_MANIFEST.json'}:
            raise ValueError('코드 목록 오류')
        for name,sha in manifest['files'].items():
            if hashlib.sha256((root/name).read_bytes()).hexdigest()!=sha:
                raise ValueError('코드 파일 해시 오류')
    env=os.environ.copy()
    env['PYTHONDONTWRITEBYTECODE']='1'
    env['PYTHONPATH']=os.pathsep.join(p for p in sys.path if isinstance(p,str) and p)
    if OLD_JOB_DIR:env['G19_OLD_JOB_DIR']=OLD_JOB_DIR
    else:env.pop('G19_OLD_JOB_DIR',None)
    env['G19_RETRY_MISSING']='1' if RETRY_MISSING_BATCH else '0'
    lines=collections.OrderedDict()
    from IPython.display import clear_output
    process=subprocess.Popen([sys.executable,'-u',str(root/'run.py'),ACTION],cwd=str(root),env=env,
                             stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True,bufsize=1)
    try:
        for line in process.stdout:
            line=line.strip()
            if not line:continue
            try:
                value=json.loads(line)
                key=(value.get('phase'),value.get('table'),value.get('asset'))
                display_line=' | '.join(str(v) if k=='phase' else k+'='+str(v) for k,v in value.items())
            except ValueError:
                key=('notice',line);display_line=line
            lines[key]=display_line
            lines.move_to_end(key)
            while len(lines)>18:lines.popitem(last=False)
            clear_output(wait=True)
            print('[Gold 형태 R1] 실행 단계:',ACTION,'/ 최근 상태 최대18줄')
            print('\n'.join(lines.values()),flush=True)
        code=process.wait()
        if code:
            print('중단되었습니다. 이 화면을 캡처하고 후속 셀은 실행하지 마세요.')
        else:
            print('현재 단계 종료. INITIAL_SOURCE_LOADED / LOADED_UNVERIFIED / SOURCE_QUERY_VERIFIED를 구분하세요.')
    except KeyboardInterrupt:
        process.terminate()
        try:process.wait(timeout=10)
        except subprocess.TimeoutExpired:
            process.kill();process.wait(timeout=10)
        print('중단 요청을 전달했습니다. 진행 영수증을 보존했습니다. 마지막 DB 요청의 반영 여부는 재확인이 필요합니다.')


try:
    _run_gold_shape()
except Exception as _gold_error:
    print('[Gold 형태 R1 준비 실패]',type(_gold_error).__name__,str(_gold_error)[:220])
