운영계 Gold 데이터 확인 SQL R1 (2026-11-09) — 읽기 전용(SELECT만)

사용법
- 먼저 메모장에서 GOLDOWNER 를 실제 Gold owner 이름으로 모두 바꾼다(찾아 바꾸기).
- Dataiku SQL 노트북에서 [1]~[5]를 각각 새 Query에 하나씩 붙여 넣고 RUN 한다.
  (노트북이 ; 에서 문장을 자르므로 각 쿼리 끝에 ; 를 넣지 않았다.)
- [1]이 전부 '있음'이어야 [2]~[5]가 실행된다. '없음'이 있으면 그 테이블 이름/권한부터 확인.

==================== [1] 테이블 존재 확인 (17개) ====================
SELECT t.OWN AS OWNER, t.TBL AS TABLE_NAME,
       CASE WHEN a.TABLE_NAME IS NULL THEN '없음 또는 권한없음' ELSE '있음' END AS STATUS
FROM (
  SELECT 'GOLDOWNER' OWN, 'IDD_B003_LGL_DOC_VER' TBL FROM DUAL UNION ALL
  SELECT 'GOLDOWNER', 'IDD_B003_LGL_DOC_UNIT' FROM DUAL UNION ALL
  SELECT 'GOLDOWNER', 'IDD_B003_LGL_DOC_BLK' FROM DUAL UNION ALL
  SELECT 'GOLDOWNER', 'IDD_B003_LGL_SRC_POST' FROM DUAL UNION ALL
  SELECT 'GOLDOWNER', 'IDD_B003_LGL_SRCH_POST' FROM DUAL UNION ALL
  SELECT 'GOLDOWNER', 'IDD_B003_LGL_SRCH_ROW' FROM DUAL UNION ALL
  SELECT 'GOLDOWNER', 'IDD_B003_LGL_SRCH_CHNL' FROM DUAL UNION ALL
  SELECT 'GOLDOWNER', 'IDD_B003_LGL_ONTL_POST' FROM DUAL UNION ALL
  SELECT 'GOLDOWNER', 'IDD_B003_LGL_ONTL_HEAD' FROM DUAL UNION ALL
  SELECT 'GOLDOWNER', 'IDD_B003_LGL_ONTL_ANC' FROM DUAL UNION ALL
  SELECT 'GOLDOWNER', 'IDD_B003_LGL_VCTR_POST' FROM DUAL UNION ALL
  SELECT 'GOLDOWNER', 'IDD_B003_LGL_VCTR_ITEM' FROM DUAL UNION ALL
  SELECT 'GOLDOWNER', 'IDD_B003_LGL_VCTR_HEAD' FROM DUAL UNION ALL
  SELECT 'SIDDA', 'LGL_AGT_RUN_A' FROM DUAL UNION ALL
  SELECT 'SIDDA', 'LGL_AGT_REC_A' FROM DUAL UNION ALL
  SELECT 'SIDDA', 'LGL_AGT_BODY_A' FROM DUAL UNION ALL
  SELECT 'SIDDA', 'LGL_EXEC_ST_A' FROM DUAL
) t
LEFT JOIN ALL_TABLES a ON a.OWNER = t.OWN AND a.TABLE_NAME = t.TBL
ORDER BY t.OWN, t.TBL

==================== [3] 채널 → 검색 게시 → 원문 게시 → 문서 연결 ====================
-- 채널 한 줄마다 한 행. 서빙이 실제로 따라가는 순서다.
-- 정상: POST_STATUS=COMPLETE, SRC_POST_CNT=1, DOC/UNIT/BLK/SRCH_ROW/VECTOR 모두 0보다 큼.
-- 여기 나온 CHANNEL 값을 서빙 환경변수 DP_QUERY_CHANNEL 에 넣는다(행이 여러 개면 그중 하나를 골라야 함).
SELECT c.RFRN_CHNL_NAME AS CHANNEL,
       c.CHAN_NO,
       c.RFRN_POST_HASH_VALU AS QUERY_RELEASE_ID,
       CASE WHEN p.RFRN_POST_HASH_VALU IS NULL THEN '검색게시 없음' ELSE '있음' END AS SRCH_POST,
       JSON_VALUE(p.RFRN_POST_IFMN_CTNT, '$.status' RETURNING VARCHAR2(40)) AS POST_STATUS,
       JSON_VALUE(p.RFRN_POST_IFMN_CTNT, '$.source_release_id' RETURNING VARCHAR2(100)) AS SOURCE_RELEASE_ID,
       JSON_VALUE(p.RFRN_POST_IFMN_CTNT, '$.dataset_version' RETURNING VARCHAR2(100)) AS DATASET_VERSION,
       (SELECT COUNT(*) FROM GOLDOWNER.IDD_B003_LGL_SRCH_ROW r
         WHERE r.RFRN_POST_HASH_VALU = c.RFRN_POST_HASH_VALU) AS SRCH_ROW_CNT,
       (SELECT COUNT(*) FROM GOLDOWNER.IDD_B003_LGL_SRC_POST s
         WHERE s.BODY_POST_HASH_VALU = JSON_VALUE(p.RFRN_POST_IFMN_CTNT, '$.source_release_id' RETURNING VARCHAR2(100))) AS SRC_POST_CNT,
       (SELECT COUNT(*) FROM GOLDOWNER.IDD_B003_LGL_DOC_VER d
         WHERE d.DTLS_HASH_VALU = JSON_VALUE(p.RFRN_POST_IFMN_CTNT, '$.dataset_version' RETURNING VARCHAR2(100))) AS DOC_CNT,
       (SELECT COUNT(*) FROM GOLDOWNER.IDD_B003_LGL_DOC_UNIT u
         WHERE u.DTLS_HASH_VALU = JSON_VALUE(p.RFRN_POST_IFMN_CTNT, '$.dataset_version' RETURNING VARCHAR2(100))) AS UNIT_CNT,
       (SELECT COUNT(*) FROM GOLDOWNER.IDD_B003_LGL_DOC_BLK b
         WHERE b.DTLS_HASH_VALU = JSON_VALUE(p.RFRN_POST_IFMN_CTNT, '$.dataset_version' RETURNING VARCHAR2(100))) AS BLK_CNT,
       h.VCTR_POST_ID,
       (SELECT COUNT(*) FROM GOLDOWNER.IDD_B003_LGL_VCTR_ITEM i
         WHERE i.VCTR_POST_ID = h.VCTR_POST_ID) AS VECTOR_CNT
FROM GOLDOWNER.IDD_B003_LGL_SRCH_CHNL c
LEFT JOIN GOLDOWNER.IDD_B003_LGL_SRCH_POST p ON p.RFRN_POST_HASH_VALU = c.RFRN_POST_HASH_VALU
LEFT JOIN GOLDOWNER.IDD_B003_LGL_VCTR_HEAD h ON h.RFRN_POST_HASH_VALU = c.RFRN_POST_HASH_VALU
ORDER BY c.RFRN_CHNL_NAME

==================== [4] 문서별 단위·블록 수 (원문이 비어 있는 문서 찾기) ====================
-- UNIT_CNT 또는 BLK_CNT 가 0인 문서가 있으면 그 문서는 질문해도 원문이 안 나온다.
SELECT d.DTLS_HASH_VALU, d.RGLT_DOC_ID, d.RGLT_DOC_NAME,
       (SELECT COUNT(*) FROM GOLDOWNER.IDD_B003_LGL_DOC_UNIT u
         WHERE u.DTLS_HASH_VALU = d.DTLS_HASH_VALU AND u.RGLT_DOC_ID = d.RGLT_DOC_ID) AS UNIT_CNT,
       (SELECT COUNT(*) FROM GOLDOWNER.IDD_B003_LGL_DOC_BLK b
         WHERE b.DTLS_HASH_VALU = d.DTLS_HASH_VALU AND b.RGLT_DOC_ID = d.RGLT_DOC_ID) AS BLK_CNT,
       (SELECT COUNT(*) FROM GOLDOWNER.IDD_B003_LGL_VCTR_ITEM i
         WHERE i.RGLT_DOC_ID = d.RGLT_DOC_ID) AS VECTOR_CNT
FROM GOLDOWNER.IDD_B003_LGL_DOC_VER d
ORDER BY UNIT_CNT, BLK_CNT, d.RGLT_DOC_NAME

==================== [5] 검색 자료 종류별 행 수 ====================
-- 본문 검색·용어 등 종류별로 비어 있는 것이 없는지 본다.
SELECT r.RFRN_POST_HASH_VALU AS QUERY_RELEASE_ID, r.RFRN_DTA_TYPE_VALU AS DATA_TYPE, COUNT(*) AS ROW_CNT
FROM GOLDOWNER.IDD_B003_LGL_SRCH_ROW r
GROUP BY r.RFRN_POST_HASH_VALU, r.RFRN_DTA_TYPE_VALU
ORDER BY r.RFRN_POST_HASH_VALU, r.RFRN_DTA_TYPE_VALU
