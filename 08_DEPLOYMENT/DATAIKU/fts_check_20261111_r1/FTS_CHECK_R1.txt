-- Q1 검색 표·인덱스 통계(통계가 오래됐거나 비었는지, 인덱스 상태)
SELECT 'TABLE' KIND,TABLE_NAME NAME,NULL STATUS,NUM_ROWS,BLOCKS,NULL CLUSTERING_FACTOR,LAST_ANALYZED,STALE_STATS FROM ALL_TAB_STATISTICS WHERE OWNER='SIDDS' AND TABLE_NAME='IDD_B003_LGL_SRCH_ROW' AND OBJECT_TYPE='TABLE' UNION ALL SELECT 'INDEX',s.INDEX_NAME,i.STATUS,s.NUM_ROWS,s.LEAF_BLOCKS,s.CLUSTERING_FACTOR,s.LAST_ANALYZED,s.STALE_STATS FROM ALL_IND_STATISTICS s JOIN ALL_INDEXES i ON i.OWNER=s.OWNER AND i.INDEX_NAME=s.INDEX_NAME WHERE s.OWNER='SIDDS' AND s.TABLE_NAME='IDD_B003_LGL_SRCH_ROW' AND s.OBJECT_TYPE='INDEX' ORDER BY 1 DESC,2

-- Q2 서빙 채널·본문 검색 종류 이름·통계(조문 단위 수, 낱말 수)

-- Q5 같은 것을 문서 하나(주관업무준칙, 없으면 아무 문서) 행만 — 고친 방식
