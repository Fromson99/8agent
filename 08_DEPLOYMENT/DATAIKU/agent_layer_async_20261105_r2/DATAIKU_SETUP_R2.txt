# Dataiku 노트북 새 커널. 세 ZIP을 업로드하고 아래 실제 값들을 입력합니다.
PARAMS = {
    'CODE_FOLDER_ID': '',          # 세 ZIP을 올린 Managed Folder ID
    'WORK_DIR': '',                # 여러 레시피가 함께 쓰는 영속 POSIX 절대 경로
    'SETUP_CONNECTION': 'AIDP',    # SIDDA DDL/GRANT 가능 연결 또는 이미 권한을 가진 owner
    'EXPECTED_SETUP_USER': '',     # 위 연결의 실제 Oracle 로그인
    'GENOS_DB_USER': '',           # GENOS ORDS가 SQL을 실행하는 실제 DB 계정
    'AGENT_CONNECTION': 'AIDP',    # 중간 결과 SELECT용 Dataiku 연결
    'TARGET_CONNECTION': 'AIDP', 'EXPECTED_DB_USER': 'PIDD2APP',
    'EMBED_MODEL': 'openai:genos-embedding:embedding-kure-v1', 'EMBED_DIMENSIONS': 1024,
    'INPUT_MODE': 'SAMPLE_ZIP', 'ENVIRONMENT': 'DEV',
    'INPUT_FOLDER_ID': '', 'SUPPORT_FOLDER_ID': '', 'DOCUMENT_PREFIX': '', 'GOLD_OWNER': '',
}

BOOTSTRAP = "import hashlib\nimport io\nimport json\nfrom pathlib import Path, PurePosixPath\nimport stat\nimport sys\nimport tempfile\nimport zipfile\nimport dataiku\n\nZIP_NAME = 'DATAIKU_AGENT_FLOW_R2.zip'\nZIP_SHA256 = 'da993fde1d7b54b8af96c5382ba9abfeb9cb39d815ef7835e2cb39088a797e6a'\n\n\ndef open_package():\n    folder = dataiku.Folder(PARAMS['CODE_FOLDER_ID'])\n    names = [p for p in folder.list_paths_in_partition() if PurePosixPath(p).name == ZIP_NAME]\n    if len(names) != 1: raise ValueError('UPLOAD_ONE_CODE_ZIP:'+ZIP_NAME)\n    with folder.get_download_stream(names[0]) as stream: raw = stream.read(16*1024**2+1)\n    if len(raw)>16*1024**2 or hashlib.sha256(raw).hexdigest()!=ZIP_SHA256: raise ValueError('CODE_ZIP_HASH')\n    root = Path(_code_dir.name)\n    root.mkdir(exist_ok=True)\n    with zipfile.ZipFile(io.BytesIO(raw)) as z:\n        names = z.namelist(); manifest = json.loads(z.read('FLOW_MANIFEST.json'))\n        if len(names)!=len(set(names)) or set(names)!=set(manifest['files'])|{'FLOW_MANIFEST.json'}:\n            raise ValueError('CODE_ZIP_MEMBERS')\n        if sum(i.file_size for i in z.infolist())>32*1024**2: raise ValueError('CODE_ZIP_SIZE')\n        for item in z.infolist():\n            p = PurePosixPath(item.filename)\n            if p.is_absolute() or '..' in p.parts or '\\\\' in item.filename or item.is_dir() or stat.S_ISLNK(item.external_attr>>16):\n                raise ValueError('CODE_ZIP_PATH')\n            dest = root.joinpath(*p.parts)\n            if dest.is_symlink() or root.resolve() not in dest.resolve().parents: raise ValueError('CODE_PATH_ESCAPE')\n            content = z.read(item)\n            if item.filename!='FLOW_MANIFEST.json' and hashlib.sha256(content).hexdigest()!=manifest['files'][item.filename]:\n                raise ValueError('CODE_FILE_HASH')\n            if dest.exists():\n                if dest.read_bytes()!=content: raise ValueError('EXTRACTED_CODE_CHANGED')\n            else:\n                dest.parent.mkdir(parents=True,exist_ok=True); dest.write_bytes(content)\n    sys.path.insert(0,str(root))\n    return root\n\n\n_code_dir = tempfile.TemporaryDirectory(prefix='agent-flow-code-')\nFLOW_PACKAGE = open_package()\n"
exec(BOOTSTRAP)
from flow_install import install, validate
from agent_setup import setup
cfg=dict(PARAMS)
for k in ('SETUP_CONNECTION','EXPECTED_SETUP_USER','GENOS_DB_USER'):
    cfg.pop(k)
if cfg['INPUT_MODE']=='SAMPLE_ZIP':
    cfg['INPUT_FOLDER_ID']=cfg['SUPPORT_FOLDER_ID']=cfg['CODE_FOLDER_ID']
    from sample_input import INDEX
    source=dataiku.Folder(cfg['CODE_FOLDER_ID'],ignore_flow=True)
    for item in (INDEX['raw'],INDEX['support']):
        with source.get_download_stream(item['file']) as f:
            b=f.read(item['bytes']+1)
        if len(b)!=item['bytes'] or hashlib.sha256(b).hexdigest()!=item['sha256']:
            raise ValueError('UPLOAD_ORIGINAL_ZIP:'+item['file'])
validate(cfg)
setup(PARAMS)
install(cfg,BOOTSTRAP)
