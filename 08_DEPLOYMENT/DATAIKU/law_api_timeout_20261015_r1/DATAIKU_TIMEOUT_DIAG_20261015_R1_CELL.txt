# Python 코드입니다. 기존 R1 Notebook의 새 Code 셀에 전체를 붙여 넣으세요.
# 커널을 재시작하지 않습니다. 기존 인증값/check 객체를 사용합니다.
# 기존 TIMEOUT 요청의 실패 지점만 분류하는 수동 추가 진단 1회입니다.
# 기존 결과/차단 상태/카운터를 초기화하지 않습니다. 같은 셀의 재실행은 저장 결과만 표시합니다.
import hashlib
import json
import re
import time
from pathlib import Path
from urllib.parse import quote, quote_plus, urlsplit
import requests


def diagnose_existing_timeout(check):
    path = Path(check.out) / "timeout_diag_r1.json"
    if path.exists():
        saved = json.loads(path.read_text(encoding="utf-8"))
        print("이미 진단한 결과입니다. 추가 호출하지 않습니다.")
        _show_timeout_diagnosis(saved, path)
        return saved
    previous = check.records.get("law_published_body", {})
    if previous.get("status") != "TIMEOUT" or check.mode != "relay":
        raise ValueError("이 추가 셀은 R1의 law_published_body=TIMEOUT / relay 모드에만 사용합니다.")
    # Preserve and use the actual request's ID/type rather than a new arbitrary document.
    params = dict(previous.get("params", {}))
    if params.get("type") != "XML" or not str(params.get("ID", "")).isdigit():
        raise ValueError("기존 요청 ID/type을 확인할 수 없습니다. 원래 summary.json을 확인하세요.")
    if set(params) != {"ID", "type"}:
        raise ValueError("기존 요청에 예상 밖의 파라미터가 있습니다. 자동 변경하지 않습니다.")
    if check.verify is False or check.calls >= 8:
        raise ValueError("기존 TLS 설정 또는 남은 호출 한도를 확인하세요.")
    url = check.origin + "/law/DRF"
    target = check.law_origin + "/DRF/lawService.do"
    headers = {"Accept": "application/xml, text/xml, application/json"}
    if check.config.get("target_in_header", True):
        target += "?target=law"
    else:
        params["target"] = "law"
    headers["target_url"] = target
    if check.cookie:
        headers["Cookie"] = check.cookie
    public_params = dict(params)
    params["OC"] = check.oc
    settings = check.session.merge_environment_settings(url, {}, True, check.verify, None)
    selected_proxy = requests.utils.select_proxy(url, settings["proxies"])
    proxy_host = ""
    if selected_proxy:
        p = urlsplit(selected_proxy if "://" in selected_proxy else "http://" + selected_proxy)
        proxy_host = (p.hostname or "") + (":" + str(p.port) if p.port else "")
    report = {
        "schema": "dataiku-timeout-diagnostic-v1", "parent_run_id": check.run_id,
        "prior_r1_calls": check.calls, "diagnostic_calls": 0,
        "combined_known_calls": check.calls, "automatic_retries": 0,
        "relay_url": url, "target_url": target, "params_without_OC": public_params,
        "effective_proxy_enabled": bool(selected_proxy), "effective_proxy_host": proxy_host,
        "tls_verification_enabled": settings["verify"] is not False,
        "connect_timeout_seconds": 8, "read_timeout_seconds": 20,
        "status": "PREPARED", "stage": "before_request", "http_status": None,
        "legal_data_success": False, "first_bytes_received": 0,
        "parent_status": previous["status"], "parent_elapsed_seconds": previous.get("elapsed_seconds"),
        "started_utc": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
    }
    if not report["tls_verification_enabled"]:
        raise ValueError("환경에서 TLS 검증이 해제되어 있습니다. 진단을 중단합니다.")
    # Reject a modified adapter with hidden automatic retries.
    retries = getattr(check.session.get_adapter(url), "max_retries", None)
    if retries is None or retries.total not in (0, False):
        raise ValueError("자동 재시도가 없는 기존 R1 세션을 사용해야 합니다.")
    def save():
        tmp = path.with_suffix(".tmp")
        tmp.write_text(json.dumps(report, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
        tmp.chmod(0o600)
        tmp.replace(path)
    # Exclusive creation prevents concurrent double clicks from issuing a second request.
    with path.open("x", encoding="utf-8") as f:
        json.dump(report, f, ensure_ascii=False, indent=2)
    path.chmod(0o600)
    start = time.monotonic()
    report.update(status="IN_FLIGHT", stage="awaiting_response_headers",
                  diagnostic_calls=1, combined_known_calls=check.calls + 1)
    save()
    response = None
    try:
        response = check.session.request("GET", url, params=params, headers=headers, json=None,
                                         timeout=(8, 20), verify=check.verify,
                                         allow_redirects=False, stream=True)
        report.update(http_status=response.status_code, headers_received_seconds=round(time.monotonic()-start, 3))
        if response.status_code != 200:
            report.update(status="HTTP_" + str(response.status_code), stage="headers_received")
        else:
            report["stage"] = "reading_first_body_bytes"
            save()
            first_chunk = next(response.iter_content(chunk_size=1024), b"")
            report.update(status="HTTP_200_WITH_FIRST_BYTES" if first_chunk else "HTTP_200_EMPTY_BODY",
                          first_bytes_received=len(first_chunk), first_bytes_sha256=hashlib.sha256(first_chunk).hexdigest(),
                          stage="first_body_bytes_received")
    except requests.RequestException as exc:
        # Includes urllib3 read timeouts wrapped as requests.ConnectionError during streaming.
        errors, queue, seen = [], [exc], set()
        while queue and len(errors) < 12:
            err = queue.pop(0)
            if not isinstance(err, BaseException) or id(err) in seen:
                continue
            seen.add(id(err)); errors.append(type(err).__name__)
            queue.extend([err.__cause__, err.__context__, getattr(err, "reason", None)])
            queue.extend(x for x in err.args if isinstance(x, BaseException))
        report["exception_classes"] = errors
        if isinstance(exc, requests.exceptions.ConnectTimeout):
            report["status"] = "CONNECTION_SETUP_TIMEOUT"
        elif isinstance(exc, requests.exceptions.ProxyError):
            report["status"] = "PROXY_ERROR"
        elif isinstance(exc, requests.exceptions.SSLError):
            report["status"] = "TLS_ERROR"
        elif isinstance(exc, requests.exceptions.ReadTimeout) or "ReadTimeoutError" in errors:
            report["status"] = "RESPONSE_WAIT_TIMEOUT" if response is None else "BODY_READ_TIMEOUT"
        elif isinstance(exc, requests.exceptions.Timeout):
            report["status"] = "TIMEOUT_UNCLASSIFIED"
        else:
            report["status"] = "CONNECTION_OR_TRANSPORT_ERROR"
        # Keep a bounded root-cause message without query credentials or proxy credentials.
        message = str(exc)
        for secret in (check.oc, check.cookie, getattr(check, "making_oc", "")):
            if secret:
                for value in {secret, quote(secret, safe=""), quote_plus(secret)}:
                    message = message.replace(value, "REDACTED")
        message = re.sub(r"(https?://)[^/@\s]+@", r"\1REDACTED@", message)
        message = re.sub(r"(?i)(OC=)[^&\s\"']+", r"\1REDACTED", message)
        report["safe_error_message"] = message[:1200]
    finally:
        if response is not None:
            response.close()
        report["elapsed_seconds"] = round(time.monotonic() - start, 3)
        save()
    _show_timeout_diagnosis(report, path)
    return report


def _show_timeout_diagnosis(report, path):
    for key in ("status", "stage", "relay_url", "target_url", "effective_proxy_enabled",
                "effective_proxy_host", "elapsed_seconds", "http_status", "exception_classes"):
        print(key + ":", report.get(key))
    print("진단 파일:", path.resolve())
    print("기존 R1 결과/차단 상태는 유지했습니다. 이 결과는 전체 법령 수신 성공 판정이 아닙니다.")
    print("Managed Folder 저장: 기존 Notebook의 8번 셀만 다시 실행하면 이 JSON도 함께 저장됩니다.")


if "check" not in globals():
    print("기존 DATAIKU_LAW_API_20261015_R1 Notebook의 같은 커널에서 실행하세요.")
    print("이 셀은 별도 Python 터미널이나 새 커널에서 실행하는 파일이 아닙니다.")
else:
    _timeout_diagnosis = diagnose_existing_timeout(check)
