#!/usr/bin/env bash
set -euo pipefail
umask 077
export_script_dir="$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" && pwd -P)"
mkdir -p "$export_script_dir/logs"
export_log_file="$export_script_dir/logs/export-$(date -u +%Y%m%dT%H%M%S)-$$.log"
exec > >(tee -a "$export_log_file") 2>&1
trap 'export_exit_code=$?; echo "RAW_EXPORT_EXIT=$export_exit_code"; exit "$export_exit_code"' EXIT
date -u
python3 -c 'import sys; print("PYTHON",sys.executable,sys.version)'
df -h .
free -m
python3 - "$export_script_dir" <<'PY'
import hashlib, json, sys
from pathlib import Path
root = Path(sys.argv[1])
pins = json.loads((root / 'CODE_PINS.json').read_text())
for name, expected in pins['files'].items():
    path = root / name
    if path.is_symlink() or hashlib.sha256(path.read_bytes()).hexdigest() != expected:
        raise SystemExit('EXPORT_CODE_CHANGED: ' + name)
print('EXPORT_CODE_VERIFIED: GENOS_RAW_EXPORT_20261023_R2')
PY
python3 "$export_script_dir/export_inputs.py" "$@"
