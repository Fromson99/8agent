SELECT c.RFRN_CHNL_NAME, JSON_VALUE(q.RFRN_POST_IFMN_CTNT, '$.status') AS QUERY_STATUS, CASE WHEN JSON_EXISTS(q.RFRN_POST_IFMN_CTNT, '$.extension') THEN 'Y' ELSE 'N' END AS QUERY_EXTENSION, p.VCTR_POST_ID, CASE WHEN JSON_VALUE(p.VCTR_POST_IFMN_CTNT, '$.manifest.query_id') = c.RFRN_POST_HASH_VALU THEN 'Y' ELSE 'N' END AS QUERY_ID_MATCH, CASE WHEN JSON_VALUE(p.VCTR_POST_IFMN_CTNT, '$.manifest.source_release_id') = JSON_VALUE(q.RFRN_POST_IFMN_CTNT, '$.source_release_id') THEN 'Y' ELSE 'N' END AS SOURCE_RELEASE_MATCH, CASE WHEN JSON_VALUE(p.VCTR_POST_IFMN_CTNT, '$.manifest.dataset_version') = JSON_VALUE(q.RFRN_POST_IFMN_CTNT, '$.dataset_version') THEN 'Y' ELSE 'N' END AS DATASET_MATCH, (SELECT COUNT(*) FROM JSON_TABLE(p.VCTR_POST_IFMN_CTNT, '$.manifest.documents[*]' COLUMNS (d VARCHAR2(1000) PATH '$')) a) AS VECTOR_DOCS, (SELECT COUNT(*) FROM JSON_TABLE(q.RFRN_POST_IFMN_CTNT, '$.source_manifest.scope.document_version_ids[*]' COLUMNS (d VARCHAR2(1000) PATH '$')) b) AS QUERY_DOCS, (SELECT COUNT(*) FROM JSON_TABLE(p.VCTR_POST_IFMN_CTNT, '$.manifest.documents[*]' COLUMNS (d VARCHAR2(1000) PATH '$')) a WHERE a.d NOT IN (SELECT b.d FROM JSON_TABLE(q.RFRN_POST_IFMN_CTNT, '$.source_manifest.scope.document_version_ids[*]' COLUMNS (d VARCHAR2(1000) PATH '$')) b)) AS ONLY_IN_VECTOR, (SELECT COUNT(*) FROM JSON_TABLE(q.RFRN_POST_IFMN_CTNT, '$.source_manifest.scope.document_version_ids[*]' COLUMNS (d VARCHAR2(1000) PATH '$')) b WHERE b.d NOT IN (SELECT a.d FROM JSON_TABLE(p.VCTR_POST_IFMN_CTNT, '$.manifest.documents[*]' COLUMNS (d VARCHAR2(1000) PATH '$')) a)) AS ONLY_IN_QUERY, JSON_VALUE(p.VCTR_POST_IFMN_CTNT, '$.completion.status') AS VECTOR_STATUS, JSON_VALUE(p.VCTR_POST_IFMN_CTNT, '$.completion.count' RETURNING NUMBER) AS COMPLETION_COUNT, (SELECT COUNT(*) FROM SIDDS.LGL_G19_VCTR_ITEM_S_1 i WHERE i.VCTR_POST_ID = p.VCTR_POST_ID) AS ITEM_ROWS, JSON_VALUE(p.VCTR_POST_IFMN_CTNT, '$.manifest.profile.metric') AS METRIC, JSON_VALUE(p.VCTR_POST_IFMN_CTNT, '$.manifest.profile.normalization') AS NORMALIZATION, JSON_VALUE(p.VCTR_POST_IFMN_CTNT, '$.manifest.profile.dimensions' RETURNING NUMBER) AS DIMENSIONS, (SELECT COUNT(*) FROM SIDDS.LGL_G19_VCTR_HEAD_S_1 h WHERE h.RFRN_POST_HASH_VALU = c.RFRN_POST_HASH_VALU) AS HEAD_ROWS FROM SIDDS.LGL_G19_SRCH_CHNL_S_1 c JOIN SIDDS.LGL_G19_SRCH_POST_S_1 q ON q.RFRN_POST_HASH_VALU = c.RFRN_POST_HASH_VALU CROSS JOIN SIDDS.LGL_G19_VCTR_POST_S_1 p ORDER BY p.VCTR_POST_ID
