"""Read-only Dataiku inventory: serial exact counts, optional allocation metadata."""
import csv
from datetime import datetime, timezone
import json
from pathlib import Path
import re
import threading
import time
import uuid

CONNECTION = 'AIDP'
COUNT_EXACT = True
COUNT_DISCOVERED = False  # Unknown matching names are listed, not automatically scanned.
INCLUDE_ALLOCATED_BYTES = True

FILTER = "OWNER IN ('SIDDS','SIDDA') AND (SUBSTR(TABLE_NAME,1,4)='LGL_' OR SUBSTR(TABLE_NAME,1,6)='LEGAL_')"
METADATA_SQL = '''SELECT t.OWNER,t.TABLE_NAME,t.TABLESPACE_NAME,t.NUM_ROWS AS STATS_ROWS,
TO_CHAR(t.LAST_ANALYZED,'YYYY-MM-DD HH24:MI:SS') AS STATS_AT,t.PARTITIONED,
TO_CHAR(o.CREATED,'YYYY-MM-DD HH24:MI:SS') AS CREATED_AT
FROM ALL_TABLES t LEFT JOIN ALL_OBJECTS o
ON o.OWNER=t.OWNER AND o.OBJECT_NAME=t.TABLE_NAME AND o.OBJECT_TYPE='TABLE' AND o.SUBOBJECT_NAME IS NULL
WHERE ''' + FILTER.replace('OWNER IN','t.OWNER IN').replace('SUBSTR(TABLE_NAME','SUBSTR(t.TABLE_NAME') + ' ORDER BY t.OWNER,t.TABLE_NAME'
ALLOCATION_SQL = '''WITH targets AS (SELECT OWNER,TABLE_NAME FROM ALL_TABLES WHERE '''+FILTER+'''),
mapped AS (
SELECT OWNER AS TABLE_OWNER,TABLE_NAME,OWNER AS SEGMENT_OWNER,TABLE_NAME AS SEGMENT_NAME FROM targets
UNION SELECT i.TABLE_OWNER,i.TABLE_NAME,i.OWNER,i.INDEX_NAME FROM ALL_INDEXES i JOIN targets t ON t.OWNER=i.TABLE_OWNER AND t.TABLE_NAME=i.TABLE_NAME
UNION SELECT l.OWNER,l.TABLE_NAME,l.OWNER,l.SEGMENT_NAME FROM ALL_LOBS l JOIN targets t ON t.OWNER=l.OWNER AND t.TABLE_NAME=l.TABLE_NAME
UNION SELECT l.OWNER,l.TABLE_NAME,l.OWNER,l.INDEX_NAME FROM ALL_LOBS l JOIN targets t ON t.OWNER=l.OWNER AND t.TABLE_NAME=l.TABLE_NAME)
SELECT m.TABLE_OWNER AS OWNER,m.TABLE_NAME,
SUM(CASE WHEN s.SEGMENT_TYPE LIKE 'TABLE%' THEN s.BYTES ELSE 0 END) AS TABLE_BYTES,
SUM(CASE WHEN s.SEGMENT_TYPE LIKE 'INDEX%' THEN s.BYTES ELSE 0 END) AS INDEX_BYTES,
SUM(CASE WHEN s.SEGMENT_TYPE LIKE 'LOB%' THEN s.BYTES ELSE 0 END) AS LOB_BYTES,
SUM(s.BYTES) AS ALLOCATED_BYTES FROM mapped m JOIN DBA_SEGMENTS s
ON s.OWNER=m.SEGMENT_OWNER AND s.SEGMENT_NAME=m.SEGMENT_NAME
GROUP BY m.TABLE_OWNER,m.TABLE_NAME'''

FIELDS = ['OWNER','TABLE_NAME','GROUP','CURRENT_TARGET','VISIBILITY','EXACT_ROWS','COUNT_STATE',
          'COUNT_STARTED_UTC','COUNT_FINISHED_UTC','COUNT_SECONDS','STATS_ROWS','STATS_AT','CREATED_AT',
          'TABLESPACE_NAME','PARTITIONED','TABLE_BYTES','INDEX_BYTES','LOB_BYTES','ALLOCATED_BYTES',
          'SIZE_STATE','ERROR','REFERENCE']


def qualified(owner, table):
    if owner not in {'SIDDS','SIDDA'} or not re.fullmatch(r'(?:LGL_|LEGAL_)[A-Z0-9_$#]+',table):
        raise ValueError('UNEXPECTED_TABLE_IDENTIFIER')
    return owner+'.'+table


def exact_sql(owner,table):
    return 'SELECT /*+ NO_PARALLEL */ COUNT(*) AS EXACT_ROWS FROM '+qualified(owner,table)


def safe_error(exc):
    # These metadata/count queries carry no document text or credentials.
    messages=re.findall(r'(?:ORA|PLS)-\d{5}:[^\r\n]*',str(exc))
    return type(exc).__name__+': '+('\n'.join(messages) if messages else str(exc).splitlines()[0])[:2000]


def stamp():return datetime.now(timezone.utc).isoformat()


def write_csv(path,rows):
    temporary=path.with_suffix('.tmp')
    with temporary.open('w',encoding='utf-8-sig',newline='') as stream:
        writer=csv.DictWriter(stream,fieldnames=FIELDS,extrasaction='ignore')
        writer.writeheader();writer.writerows(rows)
    temporary.replace(path)


def inventory(query,catalog,output,*,count_exact=True,count_discovered=False,include_size=True,notify=print):
    output=Path(output);output.mkdir(parents=True,exist_ok=True)
    path=output/('table-inventory-'+datetime.now().strftime('%Y%m%d-%H%M%S')+'-'+uuid.uuid4().hex[:8]+'.csv')
    metadata=query(METADATA_SQL)
    visible={(r['OWNER'],r['TABLE_NAME']):r for r in metadata}
    known={(r['owner'],r['table']):r for r in catalog}
    rows=[]
    for key in sorted(set(visible)|set(known)):
        owner,table=key;qualified(owner,table)
        info=known.get(key,{})
        row={'OWNER':owner,'TABLE_NAME':table,'GROUP':info.get('group','추가 발견 — 우리 생성 여부 미확인'),
             'CURRENT_TARGET':'YES' if info.get('current_target') else 'UNCONFIRMED',
             'VISIBILITY':'VISIBLE' if key in visible else 'NOT_VISIBLE_OR_ABSENT',
             'COUNT_STATE':'PENDING' if key in visible else 'NOT_VISIBLE_OR_ABSENT',
             'REFERENCE':info.get('reference','ALL_TABLES 이름 발견'),'SIZE_STATE':'NOT_REQUESTED'}
        row.update(visible.get(key,{}));rows.append(row)
    write_csv(path,rows)
    notify('목록 저장: '+str(path))
    for row in rows:notify(row['OWNER']+'.'+row['TABLE_NAME']+' | '+row['GROUP']+' | '+row['VISIBILITY'])
    if include_size:
        try:
            allocated={(r['OWNER'],r['TABLE_NAME']):r for r in query(ALLOCATION_SQL)}
        except Exception as exc:
            error=safe_error(exc);notify('할당량 조회 불가 — 행 수 조회는 계속합니다. '+error)
            allocated=None
        for row in rows:
            if allocated is None:row['SIZE_STATE']='QUERY_FAILED_OR_NO_PRIVILEGE'
            elif row.get('PARTITIONED')=='YES':row['SIZE_STATE']='PARTITIONED_NOT_FULLY_MAPPED'
            else:
                size=allocated.get((row['OWNER'],row['TABLE_NAME']))
                if size:row.update(size);row['SIZE_STATE']='ALLOCATED_NOT_RECLAIMABLE_ESTIMATE'
                else:row['SIZE_STATE']='NO_SEGMENT_INFO'
        write_csv(path,rows)
    for i,row in enumerate(rows,1):
        key=(row['OWNER'],row['TABLE_NAME'])
        if row['VISIBILITY']!='VISIBLE':continue
        if not count_exact or (key not in known and not count_discovered):
            row['COUNT_STATE']='SKIPPED_NOT_EXACT';write_csv(path,rows);continue
        name=qualified(*key);start=time.monotonic();row['COUNT_STARTED_UTC']=stamp()
        row['COUNT_STATE']='RUNNING';write_csv(path,rows)
        notify(f'[{i}/{len(rows)}] COUNT 시작: {name} | 한 표씩 순차 조회')
        stop=threading.Event()
        def heartbeat():
            while not stop.wait(15):
                elapsed=int(time.monotonic()-start)
                notify(f'조회 중: {name} | {elapsed//60}분 {elapsed%60:02d}초 | 결과 미확정')
        thread=threading.Thread(target=heartbeat,daemon=True);thread.start()
        try:
            answer=query(exact_sql(*key))
            if len(answer)!=1 or answer[0].get('EXACT_ROWS') is None:raise ValueError('COUNT_RESPONSE_INVALID')
            n=answer[0]['EXACT_ROWS']
            if isinstance(n,bool) or int(n)!=n or n<0:raise ValueError('COUNT_VALUE_INVALID')
            row.update(EXACT_ROWS=int(n),COUNT_STATE='COUNTED')
            notify(f'완료: {name} = {int(n):,}행 | {(time.monotonic()-start):.1f}초')
        except KeyboardInterrupt:
            row['COUNT_STATE']='INTERRUPTED_UNKNOWN';notify('중단 기록 저장: '+str(path));raise
        except Exception as exc:
            row.update(COUNT_STATE='ERROR',ERROR=safe_error(exc));notify('조회 오류: '+name+' | '+row['ERROR'])
        finally:
            stop.set();thread.join(timeout=2)
            row.update(COUNT_FINISHED_UTC=stamp(),COUNT_SECONDS=round(time.monotonic()-start,3))
            write_csv(path,rows)
    notify('\n최종 목록 — 실제행수 / 할당MiB / 구분')
    for row in rows:
        count=f"{row['EXACT_ROWS']:,}" if row.get('COUNT_STATE')=='COUNTED' else row['COUNT_STATE']
        size=f"{row['ALLOCATED_BYTES']/1024**2:,.2f}" if row.get('ALLOCATED_BYTES') is not None else '미확인'
        notify(row['OWNER']+'.'+row['TABLE_NAME']+' | '+count+' | '+size+' | '+row['GROUP'])
    notify('결과 CSV: '+str(path))
    notify('삭제 판정 아님. 표별 COUNT 시점이 다르며 공용 계정의 생성자/현재 사용 여부는 별도 확인입니다.')
    return rows,path


def run(catalog):
    import dataiku
    executor=dataiku.SQLExecutor2(connection=CONNECTION)
    def query(sql):
        frame=executor.query_to_df(sql,infer_from_schema=True,parse_dates=False)
        frame=frame.astype(object).where(frame.notna(),None)
        return [{str(k).upper():v.item() if hasattr(v,'item') else v for k,v in row.items()}
                for row in frame.to_dict(orient='records')]
    return inventory(query,catalog,Path.cwd()/'gold-table-inventory',count_exact=COUNT_EXACT,
                     count_discovered=COUNT_DISCOVERED,include_size=INCLUDE_ALLOCATED_BYTES,
                     notify=lambda s:print(s,flush=True))
