SELECT ORD, CHECK_NAME, VALUE FROM (
SELECT 1 ORD, 'SESSION_USER' CHECK_NAME, SYS_CONTEXT('USERENV','SESSION_USER') VALUE FROM DUAL
UNION ALL SELECT 2, 'DB_NAME', SYS_CONTEXT('USERENV','DB_UNIQUE_NAME') FROM DUAL
UNION ALL SELECT 3, 'SRCH_ROW_OBJECT', (SELECT LISTAGG(OBJECT_TYPE||':'||STATUS,',') WITHIN GROUP (ORDER BY OBJECT_TYPE) FROM ALL_OBJECTS WHERE OWNER='SIDDS' AND OBJECT_NAME='LGL_G19_SRCH_ROW_S_1') FROM DUAL
UNION ALL SELECT 4, 'CHANNEL_RELEASE', (SELECT SUBSTR(MIN(RFRN_POST_HASH_VALU),1,16)||' rows='||COUNT(*) FROM SIDDS.LGL_G19_SRCH_CHNL_S_1 WHERE RFRN_CHNL_NAME='g19_7c1f8f4931c8be61b914f5bc') FROM DUAL
UNION ALL SELECT 5, 'POSTING_KIND', (SELECT MIN(RFRN_DTA_TYPE_VALU) FROM SIDDS.LGL_G19_SRCH_ROW_S_1 WHERE RFRN_POST_HASH_VALU=(SELECT RFRN_POST_HASH_VALU FROM SIDDS.LGL_G19_SRCH_CHNL_S_1 WHERE RFRN_CHNL_NAME='g19_7c1f8f4931c8be61b914f5bc') AND RFRN_DTA_TYPE_VALU LIKE 'asset%.unit_fts_posting') FROM DUAL
UNION ALL SELECT 6, 'LENGTH_KIND', (SELECT MIN(RFRN_DTA_TYPE_VALU) FROM SIDDS.LGL_G19_SRCH_ROW_S_1 WHERE RFRN_POST_HASH_VALU=(SELECT RFRN_POST_HASH_VALU FROM SIDDS.LGL_G19_SRCH_CHNL_S_1 WHERE RFRN_CHNL_NAME='g19_7c1f8f4931c8be61b914f5bc') AND RFRN_DTA_TYPE_VALU LIKE 'asset%.unit_fts_length') FROM DUAL
UNION ALL SELECT 7, 'JSON_CHECKS', (SELECT TO_CHAR(COUNT(*)) FROM ALL_CONSTRAINTS WHERE OWNER='SIDDS' AND TABLE_NAME='LGL_G19_SRCH_ROW_S_1' AND CONSTRAINT_TYPE='C' AND UPPER(SEARCH_CONDITION_VC) LIKE '%JSON%') FROM DUAL
UNION ALL SELECT 8, 'VPD_POLICIES', (SELECT TO_CHAR(COUNT(*)) FROM ALL_POLICIES WHERE OBJECT_OWNER='SIDDS' AND OBJECT_NAME='LGL_G19_SRCH_ROW_S_1') FROM DUAL
UNION ALL SELECT 9, 'SYNONYMS', (SELECT TO_CHAR(COUNT(*))||' '||SUBSTR(LISTAGG(OWNER||'.'||SYNONYM_NAME,',') WITHIN GROUP (ORDER BY OWNER),1,200) FROM ALL_SYNONYMS WHERE TABLE_OWNER='SIDDS' AND TABLE_NAME='LGL_G19_SRCH_ROW_S_1') FROM DUAL
UNION ALL SELECT 10, 'BAIP04APP_SELECT', (SELECT TO_CHAR(COUNT(*)) FROM ALL_TAB_PRIVS WHERE TABLE_SCHEMA='SIDDS' AND TABLE_NAME='LGL_G19_SRCH_ROW_S_1' AND GRANTEE='BAIP04APP' AND PRIVILEGE='SELECT') FROM DUAL
UNION ALL SELECT 11, 'NAME_COLLISIONS', (SELECT TO_CHAR(COUNT(*))||' '||SUBSTR(LISTAGG(OWNER||'.'||OBJECT_NAME||':'||OBJECT_TYPE,',') WITHIN GROUP (ORDER BY OWNER),1,300) FROM ALL_OBJECTS WHERE OBJECT_NAME IN ('P','R','L','J','F','K','N','DF','TF','DID','SID','COL','POS','TOKEN','PHRASE','SCORE','POSTINGS','POSITIONS','FREQUENCIES','LENGTHS','MATCHES') AND OWNER IN ('PUBLIC','SIDDS','BAIP04APP',SYS_CONTEXT('USERENV','SESSION_USER'))) FROM DUAL
)
ORDER BY ORD
