-- 시험 복사본에 기준 세대를 적재한 뒤 main 채널 만들기. 기준 세대 하나만 적재한 상태(g19_ 채널 1개)에서만 추가된다.
-- g19_ 채널이 2개 이상이면 0행 — 어느 세대인지 M01로 보고 직접 정한다.

-- [M01] 지금 채널 확인 — g19_ 한 줄이어야 함
SELECT RFRN_CHNL_NAME, RFRN_POST_HASH_VALU, CHAN_NO FROM SIDDS.TST_B003_LGL_SRCH_CHNL ORDER BY RFRN_CHNL_NAME

-- [M02] main 추가 (g19_ 채널이 딱 1개이고 main이 없을 때만)
INSERT INTO SIDDS.TST_B003_LGL_SRCH_CHNL (RFRN_CHNL_NAME, RFRN_POST_HASH_VALU, CHAN_NO) SELECT 'main', RFRN_POST_HASH_VALU, 1 FROM SIDDS.TST_B003_LGL_SRCH_CHNL WHERE RFRN_CHNL_NAME LIKE 'g19\_%' ESCAPE '\' AND NOT EXISTS (SELECT 1 FROM SIDDS.TST_B003_LGL_SRCH_CHNL WHERE RFRN_CHNL_NAME = 'main') AND (SELECT COUNT(*) FROM SIDDS.TST_B003_LGL_SRCH_CHNL WHERE RFRN_CHNL_NAME LIKE 'g19\_%' ESCAPE '\') = 1

-- [M03] 저장
COMMIT

-- [M04] 확인 — main 과 g19_ 줄이 같은 RFRN_POST_HASH_VALU
SELECT RFRN_CHNL_NAME, RFRN_POST_HASH_VALU, CHAN_NO FROM SIDDS.TST_B003_LGL_SRCH_CHNL ORDER BY RFRN_CHNL_NAME
