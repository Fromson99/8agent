-- Agent 중간표 SIDDA.LGL_PIPE_RESULT_A (GENOS가 구조화·색인 결과를 조각으로 넣는 표). 없을 때만, SIDDA 소유자 또는 DBA가 실행.
-- 권한: GENOS ORDS 계정 SELECT, INSERT / Dataiku 계정 SELECT, DELETE (DELETE는 증분이 적재 끝난 결과를 지울 때).

-- [A01] Agent 중간표 만들기
CREATE TABLE SIDDA.LGL_PIPE_RESULT_A (JOB_ID VARCHAR2(32 BYTE) NOT NULL, SEQ_NO NUMBER(12) NOT NULL, BIND_HASH VARCHAR2(64 BYTE) NOT NULL, DATA_HASH VARCHAR2(64 BYTE) NOT NULL, BYTE_CNT NUMBER(14) NOT NULL, PART_CNT NUMBER(12) NOT NULL, BODY_B64 CLOB, PRIMARY KEY (JOB_ID, SEQ_NO), CHECK (BYTE_CNT > 0 AND ((SEQ_NO = 0 AND PART_CNT > 0) OR (SEQ_NO > 0 AND PART_CNT = 0))))
