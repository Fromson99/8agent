-- Oracle 전체 9테이블 생성 SQL / 2026-10-19 R1 / UTF-8
-- 실행 위치: Dataiku SQL 노트북, AIDP 연결. 각 SQL을 별도 Query로 실행합니다.
-- 이 파일 전체를 단일 Query로 실행하지 않습니다. 번호 순서대로 실행합니다.
-- 생성 대상 owner: SIDDA. ORDS 접속 별칭 b04는 테이블 owner가 아닙니다.
-- SIDDA 소유자 또는 해당 스키마에 객체를 생성할 권한이 있는 계정으로 실행합니다.
--
-- 00에서 기존 객체를 먼저 확인합니다. 이미 있는 테이블/인덱스는 재생성하지 않고
-- 컬럼/키/인덱스 구조를 대조합니다. 다른 구조의 기존 객체를 자동 수정하지 않습니다.
-- ORA-00955 또는 응답 유실이면 객체/구조를 다시 확인합니다. 성공으로 간주하지 않습니다.
-- 조회 결과에 없더라도 메타데이터 조회 권한이 없으면 실제 부재를 확정할 수 없습니다.
-- CREATE 문은 없다고 확인한 객체에만 실행합니다. 오류가 나면 다음 단계로 넘기지 않습니다.
-- 기존 SIDDS 또는 *_S 테이블은 이번 생성 대상이 아닙니다.
--
-- 범위: 원문 3 + 자산/원문 릴리스 2 + 직접 조회 4 = 9테이블.
-- 추가 조회 인덱스 4개 포함. PK에 따른 인덱스는 Oracle이 생성합니다.
-- 데이터 이관, 조회 데이터 게시, 채널 활성화, GENOS 배포는 별도 실행입니다.
-- QUERY_STATE 테이블 생성만으로 기존 NFS 대화/대사 이력이 이전되지는 않습니다.
-- 원문 3테이블은 현재 oracle_dp_io.py의 SIDDA/*_A 컬럼·기본키 계약을 따릅니다.
-- 기존 *_S 제약명과 충돌하지 않도록 원문 PK 제약명에도 _A를 포함했습니다.
-- 자산 2테이블과 조회 4테이블/인덱스는 기존 유지 DDL을 그대로 사용합니다.
-- 현장 Oracle/ORDS에서 이 통합 파일을 실행한 결과는 아직 확인하지 않았습니다.

-- 00. 기존 객체 확인: 없는 객체만 아래에서 생성합니다.
SELECT OWNER, OBJECT_NAME, OBJECT_TYPE
FROM ALL_OBJECTS
WHERE OWNER = 'SIDDA'
  AND OBJECT_NAME IN (
    'LEGAL_SRC_DOCUMENT_A', 'LEGAL_SRC_SOURCE_UNIT_A', 'LEGAL_SRC_BLOCK_PAYLOAD_A',
    'LEGAL_RAG_ASSET_PART_A', 'LEGAL_RAG_RELEASE_A',
    'LEGAL_RAG_QUERY_RELEASE_A', 'LEGAL_RAG_QUERY_ROW_A',
    'LEGAL_RAG_QUERY_ACTIVE_A', 'LEGAL_RAG_QUERY_STATE_A',
    'LEGAL_Q_ROW_LOOKUP', 'LEGAL_Q_ROW_OWNER',
    'LEGAL_Q_ROW_TARGET', 'LEGAL_Q_ROW_DOCUMENT'
  )
ORDER BY OBJECT_TYPE, OBJECT_NAME;

-- 01. 원문: 문서 정보.
CREATE TABLE SIDDA.LEGAL_SRC_DOCUMENT_A (
    DATASET_VERSION VARCHAR2(64 BYTE) NOT NULL,
    DOCUMENT_ID VARCHAR2(512 BYTE) NOT NULL,
    NAME CLOB,
    SOURCE_ARTIFACT_SHA256 VARCHAR2(64 BYTE),
    SOURCE_JSON CLOB,
    CONSTRAINT LEGAL_SRC_DOCUMENT_A_PK PRIMARY KEY (DATASET_VERSION, DOCUMENT_ID)
);

-- 02. 원문: 문서 구조 단위 및 부모 연결.
CREATE TABLE SIDDA.LEGAL_SRC_SOURCE_UNIT_A (
    DATASET_VERSION VARCHAR2(64 BYTE) NOT NULL,
    DOCUMENT_ID VARCHAR2(512 BYTE) NOT NULL,
    UNIT_ID VARCHAR2(512 BYTE) NOT NULL,
    PARENT_ID VARCHAR2(512 BYTE),
    SOURCE_JSON CLOB,
    CONSTRAINT LEGAL_SRC_SOURCE_UNIT_A_PK PRIMARY KEY (DATASET_VERSION, DOCUMENT_ID, UNIT_ID)
);

-- 03. 원문: 본문 블록.
CREATE TABLE SIDDA.LEGAL_SRC_BLOCK_PAYLOAD_A (
    DATASET_VERSION VARCHAR2(64 BYTE) NOT NULL,
    DOCUMENT_ID VARCHAR2(512 BYTE) NOT NULL,
    BLOCK_ID VARCHAR2(512 BYTE) NOT NULL,
    UNIT_ID VARCHAR2(512 BYTE),
    TEXT CLOB,
    SOURCE_ARTIFACT_SHA256 VARCHAR2(64 BYTE),
    SOURCE_JSON CLOB,
    JSON_TEXT_ELIDED NUMBER(1),
    CONSTRAINT LEGAL_SRC_BLOCK_PAYLOAD_A_PK PRIMARY KEY (DATASET_VERSION, DOCUMENT_ID, BLOCK_ID)
);

-- 04. 자산: 기존 게시 계약의 분할 자산 보관.
-- 직접 조회 서빙은 이 자산을 내려받아 SQLite로 복원하지 않습니다.
CREATE TABLE SIDDA.LEGAL_RAG_ASSET_PART_A (
    ASSET_SHA256 VARCHAR2(64 BYTE) NOT NULL,
    PART_NO NUMBER(12) NOT NULL,
    RAW_BYTES NUMBER(12) NOT NULL,
    PART_SHA256 VARCHAR2(64 BYTE) NOT NULL,
    DATA_B64 CLOB NOT NULL,
    CONSTRAINT LEGAL_RAG_ASSET_PART_PK PRIMARY KEY (ASSET_SHA256, PART_NO),
    CONSTRAINT LEGAL_RAG_ASSET_PART_CK CHECK (PART_NO >= 0 AND RAW_BYTES > 0 AND RAW_BYTES <= 1048576)
);

-- 05. 원문·자산 게시 완료 릴리스.
CREATE TABLE SIDDA.LEGAL_RAG_RELEASE_A (
    RELEASE_ID VARCHAR2(64 BYTE) NOT NULL,
    DATASET_VERSION VARCHAR2(64 BYTE) NOT NULL,
    RELEASE_JSON CLOB NOT NULL,
    CREATED_AT TIMESTAMP WITH TIME ZONE DEFAULT SYSTIMESTAMP NOT NULL,
    CONSTRAINT LEGAL_RAG_RELEASE_PK PRIMARY KEY (RELEASE_ID)
);

-- 06. 직접 조회: 완료 릴리스 명세.
CREATE TABLE SIDDA.LEGAL_RAG_QUERY_RELEASE_A (
        RELEASE_ID VARCHAR2(64 BYTE) PRIMARY KEY,
        MANIFEST_JSON CLOB NOT NULL,
        CONSTRAINT LEGAL_Q_RELEASE_JSON CHECK (MANIFEST_JSON IS JSON));

-- 07. 직접 조회: 검색·용어·관계·문서 구조 행.
CREATE TABLE SIDDA.LEGAL_RAG_QUERY_ROW_A (
        RELEASE_ID VARCHAR2(64 BYTE) NOT NULL, KIND VARCHAR2(64 BYTE) NOT NULL,
        ROW_ID VARCHAR2(64 BYTE) NOT NULL, DOCUMENT_ID VARCHAR2(512 BYTE),
        LOOKUP_ID VARCHAR2(512 BYTE), OWNER_ID VARCHAR2(512 BYTE),
        TARGET_ID VARCHAR2(512 BYTE), ROW_SHA256 VARCHAR2(64 BYTE) NOT NULL,
        VALUE_JSON CLOB NOT NULL,
        CONSTRAINT LEGAL_Q_ROW_PK PRIMARY KEY (RELEASE_ID,KIND,ROW_ID),
        CONSTRAINT LEGAL_Q_ROW_JSON CHECK (VALUE_JSON IS JSON));

-- 08. 직접 조회: 조회 키 인덱스.
CREATE INDEX SIDDA.LEGAL_Q_ROW_LOOKUP ON SIDDA.LEGAL_RAG_QUERY_ROW_A(RELEASE_ID,KIND,LOOKUP_ID);

-- 09. 직접 조회: 소유 단위 인덱스.
CREATE INDEX SIDDA.LEGAL_Q_ROW_OWNER ON SIDDA.LEGAL_RAG_QUERY_ROW_A(RELEASE_ID,KIND,OWNER_ID);

-- 10. 직접 조회: 관계 대상 인덱스.
CREATE INDEX SIDDA.LEGAL_Q_ROW_TARGET ON SIDDA.LEGAL_RAG_QUERY_ROW_A(RELEASE_ID,KIND,TARGET_ID);

-- 11. 직접 조회: 문서 범위 인덱스.
CREATE INDEX SIDDA.LEGAL_Q_ROW_DOCUMENT ON SIDDA.LEGAL_RAG_QUERY_ROW_A(RELEASE_ID,KIND,DOCUMENT_ID);

-- 12. 직접 조회: 채널별 활성 릴리스. 06 테이블이 먼저 있어야 합니다.
CREATE TABLE SIDDA.LEGAL_RAG_QUERY_ACTIVE_A (
        CHANNEL VARCHAR2(64 BYTE) PRIMARY KEY, RELEASE_ID VARCHAR2(64 BYTE) NOT NULL,
        REVISION NUMBER(18) NOT NULL CHECK (REVISION>=1),
        CONSTRAINT LEGAL_Q_ACTIVE_RELEASE FOREIGN KEY (RELEASE_ID) REFERENCES SIDDA.LEGAL_RAG_QUERY_RELEASE_A(RELEASE_ID));

-- 13. 직접 조회: revision으로 동시 수정 충돌을 검사하는 JSON 상태 저장소.
CREATE TABLE SIDDA.LEGAL_RAG_QUERY_STATE_A (
        SCOPE_HASH VARCHAR2(64 BYTE) NOT NULL, STATE_KEY VARCHAR2(128 BYTE) NOT NULL,
        REVISION NUMBER(18) NOT NULL CHECK (REVISION>=1),
        VALUE_SHA256 VARCHAR2(64 BYTE) NOT NULL, VALUE_JSON CLOB NOT NULL,
        CONSTRAINT LEGAL_Q_STATE_PK PRIMARY KEY (SCOPE_HASH,STATE_KEY),
        CONSTRAINT LEGAL_Q_STATE_JSON CHECK (VALUE_JSON IS JSON));

-- 14. 생성 후 테이블 확인: 아래 9테이블이 모두 조회되어야 합니다.
SELECT OWNER, TABLE_NAME
FROM ALL_TABLES
WHERE OWNER = 'SIDDA'
  AND TABLE_NAME IN (
    'LEGAL_SRC_DOCUMENT_A', 'LEGAL_SRC_SOURCE_UNIT_A', 'LEGAL_SRC_BLOCK_PAYLOAD_A',
    'LEGAL_RAG_ASSET_PART_A', 'LEGAL_RAG_RELEASE_A',
    'LEGAL_RAG_QUERY_RELEASE_A', 'LEGAL_RAG_QUERY_ROW_A',
    'LEGAL_RAG_QUERY_ACTIVE_A', 'LEGAL_RAG_QUERY_STATE_A'
  )
ORDER BY TABLE_NAME;

-- 15. 컬럼 확인: 전체 46행. 본문/JSON/이름의 CLOB과 VARCHAR2 BYTE 길이를 유지합니다.
SELECT TABLE_NAME, COLUMN_ID, COLUMN_NAME, DATA_TYPE, DATA_LENGTH,
       CHAR_USED, DATA_PRECISION, DATA_SCALE, NULLABLE
FROM ALL_TAB_COLUMNS
WHERE OWNER = 'SIDDA'
  AND TABLE_NAME IN (
    'LEGAL_SRC_DOCUMENT_A', 'LEGAL_SRC_SOURCE_UNIT_A', 'LEGAL_SRC_BLOCK_PAYLOAD_A',
    'LEGAL_RAG_ASSET_PART_A', 'LEGAL_RAG_RELEASE_A',
    'LEGAL_RAG_QUERY_RELEASE_A', 'LEGAL_RAG_QUERY_ROW_A',
    'LEGAL_RAG_QUERY_ACTIVE_A', 'LEGAL_RAG_QUERY_STATE_A'
  )
ORDER BY TABLE_NAME, COLUMN_ID;

-- 16. PK/FK/CHECK 확인: STATUS=ENABLED, VALIDATED=VALIDATED.
-- 각 테이블 PK, ACTIVE의 RELEASE FK, JSON/revision 및 자산 크기 CHECK를 확인합니다.
SELECT TABLE_NAME, CONSTRAINT_NAME, CONSTRAINT_TYPE, STATUS, VALIDATED,
       SEARCH_CONDITION_VC, R_OWNER, R_CONSTRAINT_NAME
FROM ALL_CONSTRAINTS
WHERE OWNER = 'SIDDA'
  AND TABLE_NAME IN (
    'LEGAL_SRC_DOCUMENT_A', 'LEGAL_SRC_SOURCE_UNIT_A', 'LEGAL_SRC_BLOCK_PAYLOAD_A',
    'LEGAL_RAG_ASSET_PART_A', 'LEGAL_RAG_RELEASE_A',
    'LEGAL_RAG_QUERY_RELEASE_A', 'LEGAL_RAG_QUERY_ROW_A',
    'LEGAL_RAG_QUERY_ACTIVE_A', 'LEGAL_RAG_QUERY_STATE_A'
  )
ORDER BY TABLE_NAME, CONSTRAINT_TYPE, CONSTRAINT_NAME;

-- 17. 추가 조회 인덱스: 4종, 각각 3컬럼 = 12행. INDEX_STATUS=VALID.
SELECT i.INDEX_NAME, i.STATUS AS INDEX_STATUS, c.TABLE_NAME,
       c.COLUMN_POSITION, c.COLUMN_NAME
FROM ALL_INDEXES i
JOIN ALL_IND_COLUMNS c
  ON c.INDEX_OWNER = i.OWNER AND c.INDEX_NAME = i.INDEX_NAME
WHERE i.OWNER = 'SIDDA'
  AND i.INDEX_NAME IN (
    'LEGAL_Q_ROW_LOOKUP', 'LEGAL_Q_ROW_OWNER',
    'LEGAL_Q_ROW_TARGET', 'LEGAL_Q_ROW_DOCUMENT'
  )
ORDER BY i.INDEX_NAME, c.COLUMN_POSITION;
