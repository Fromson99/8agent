-- R6_STAGE_GRANT_R1: Dataiku Oracle SQL 노트북 (LGL_PIPE_RESULT_A 데이터셋과 같은 연결).
-- 이 파일 전체를 Query 하나에 붙여넣고, 부분 선택을 해제한 뒤 RUN. 끝까지 복사하세요.
-- 문장 구분 주석(-- DKU_END_STATEMENT) 두 줄도 포함합니다. SQL*Plus / 는 붙이지 않습니다.
-- 하는 일: 이 DB가 ORDS DB(pidddacd_vnylh)이고 SIDDA.LGL_PIPE_RESULT_A(7컬럼)와 BAIP04APP
--          계정이 있으면 GRANT SELECT, INSERT ON SIDDA.LGL_PIPE_RESULT_A TO BAIP04APP.
-- 조건이 안 맞으면 권한을 바꾸지 않고 ORA-2000x 오류로 이유를 표시합니다.
--   ORA-20001 DB_MISMATCH       이 연결의 DB가 ORDS DB와 다름 → ORDS DB 연결에서 실행
--   ORA-20002 TABLE_NOT_FOUND   SIDDA.LGL_PIPE_RESULT_A 7컬럼이 안 보임 (보이는 비슷한 테이블 표시)
--   ORA-20003 GRANTEE_NOT_FOUND 이 DB에 BAIP04APP 계정 없음
--   ORA-20004 NO_GRANT_PRIVILEGE 이 연결 계정에 부여 권한 없음 → SIDDA 소유자/DBA 연결로 실행
-- 완료: 오류 없이 결과 5행, SELECT_GRANT / INSERT_GRANT = PASS.
--   연결 계정이 SIDDA가 아니면 두 행이 NOT_VISIBLE일 수 있습니다(권한 부여는 완료).
-- 테이블 생성·삭제·행 변경 없음. 다시 실행해도 같은 결과입니다.
-- 이후 GENOS pipeline 코드스페이스 TRANSFER에서: bash ./GENOS_R6_ORDS_CHECK_R1.txt
DECLARE
  n NUMBER;
  v_db VARCHAR2(128) := SYS_CONTEXT('USERENV','DB_UNIQUE_NAME');
  v_found VARCHAR2(4000);
BEGIN
  IF LOWER(v_db) <> 'pidddacd_vnylh' THEN
    RAISE_APPLICATION_ERROR(-20001, 'DB_MISMATCH: connection DB='||v_db||', ORDS DB=pidddacd_vnylh');
  END IF;
  SELECT COUNT(*) INTO n FROM ALL_TAB_COLUMNS
   WHERE OWNER='SIDDA' AND TABLE_NAME='LGL_PIPE_RESULT_A'
     AND COLUMN_NAME IN ('JOB_ID','SEQ_NO','BIND_HASH','DATA_HASH','BYTE_CNT','PART_CNT','BODY_B64');
  IF n <> 7 THEN
    SELECT SUBSTR(LISTAGG(OWNER||'.'||TABLE_NAME, ', ') WITHIN GROUP (ORDER BY OWNER, TABLE_NAME),1,1500)
      INTO v_found FROM ALL_TABLES WHERE UPPER(TABLE_NAME) LIKE '%PIPE_RESULT%';
    RAISE_APPLICATION_ERROR(-20002, 'TABLE_NOT_FOUND: SIDDA.LGL_PIPE_RESULT_A columns='||n||'/7, visible: '||NVL(v_found,'none'));
  END IF;
  SELECT COUNT(*) INTO n FROM ALL_USERS WHERE USERNAME='BAIP04APP';
  IF n <> 1 THEN
    RAISE_APPLICATION_ERROR(-20003, 'GRANTEE_NOT_FOUND: BAIP04APP not in DB '||v_db);
  END IF;
  BEGIN
    EXECUTE IMMEDIATE 'GRANT SELECT, INSERT ON SIDDA.LGL_PIPE_RESULT_A TO BAIP04APP';
  EXCEPTION WHEN OTHERS THEN
    IF SQLCODE = -1031 THEN
      RAISE_APPLICATION_ERROR(-20004, 'NO_GRANT_PRIVILEGE: '||SYS_CONTEXT('USERENV','SESSION_USER')||' cannot grant; run as SIDDA owner or DBA');
    END IF;
    RAISE;
  END;
END;

-- DKU_END_STATEMENT
SELECT CHECK_NAME, ACTUAL, RESULT FROM (
  SELECT 1 ORD, 'DB_NAME' CHECK_NAME, SYS_CONTEXT('USERENV','DB_UNIQUE_NAME') ACTUAL,
         CASE WHEN LOWER(SYS_CONTEXT('USERENV','DB_UNIQUE_NAME'))='pidddacd_vnylh' THEN 'PASS' ELSE 'FAIL' END RESULT
    FROM DUAL
  UNION ALL
  SELECT 2, 'CONNECTION_USER', SYS_CONTEXT('USERENV','SESSION_USER'), 'INFO' FROM DUAL
  UNION ALL
  SELECT 3, 'TABLE_COLUMNS', TO_CHAR(COUNT(*)), CASE WHEN COUNT(*)=7 THEN 'PASS' ELSE 'FAIL' END
    FROM ALL_TAB_COLUMNS WHERE OWNER='SIDDA' AND TABLE_NAME='LGL_PIPE_RESULT_A'
  UNION ALL
  SELECT CASE P.PRIV WHEN 'SELECT' THEN 4 ELSE 5 END, P.PRIV||'_GRANT', NVL(T.GRANTOR,'-'),
         CASE WHEN T.PRIVILEGE IS NOT NULL THEN 'PASS'
              WHEN U.USR='SIDDA' THEN 'FAIL' ELSE 'NOT_VISIBLE' END
    FROM (SELECT 'SELECT' PRIV FROM DUAL UNION ALL SELECT 'INSERT' FROM DUAL) P
    CROSS JOIN (SELECT SYS_CONTEXT('USERENV','SESSION_USER') USR FROM DUAL) U
    LEFT JOIN (SELECT PRIVILEGE, MAX(GRANTOR) GRANTOR FROM ALL_TAB_PRIVS
                WHERE TABLE_SCHEMA='SIDDA' AND TABLE_NAME='LGL_PIPE_RESULT_A' AND GRANTEE='BAIP04APP'
                GROUP BY PRIVILEGE) T ON T.PRIVILEGE=P.PRIV
)
ORDER BY ORD

-- DKU_END_STATEMENT
