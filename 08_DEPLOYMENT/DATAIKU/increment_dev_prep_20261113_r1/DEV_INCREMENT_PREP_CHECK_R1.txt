-- 개발계 증분 준비 점검 R1 (2026-11-13) — 읽기 전용, 행·표를 바꾸지 않음. 소유자 SIDDS.
-- Dataiku SQL 노트북에서 [번호] 한 개씩 별도 Query로 실행 (문장 끝 세미콜론 없음, 버튼은 RUN).

-- [P01] 채널 — 지금 서빙이 보는 채널(main 또는 g19_…)과 세대 번호. 이 채널들은 손대지 않는다
SELECT RFRN_CHNL_NAME, RFRN_POST_HASH_VALU, CHAN_NO FROM SIDDS.IDD_B003_LGL_SRCH_CHNL ORDER BY RFRN_CHNL_NAME

-- [P02] 세대(조회 게시본)별 형식 — 합치기 조건: GEN_SCHEMA term-source-generation-v2, NAV_SCHEMA term-navigation-index-v7, SCOPE_CONTRACT N
SELECT RFRN_POST_HASH_VALU, JSON_VALUE(RFRN_POST_IFMN_CTNT,'$.projection') AS PROJECTION, JSON_VALUE(RFRN_POST_IFMN_CTNT,'$.source_manifest.generation_id') AS GENERATION_ID, JSON_VALUE(RFRN_POST_IFMN_CTNT,'$.source_manifest.schema_version') AS GEN_SCHEMA, JSON_VALUE(RFRN_POST_IFMN_CTNT,'$.source_manifest.counts.document') AS DOCS, CASE WHEN JSON_EXISTS(RFRN_POST_IFMN_CTNT,'$.source_manifest.append_lineage') THEN 'Y' ELSE 'N' END AS APPENDED, CASE WHEN JSON_EXISTS(RFRN_POST_IFMN_CTNT,'$.search_selection.schema_version') THEN 'Y' ELSE 'N' END AS SEARCH_SELECTION, JSON_VALUE(RFRN_POST_IFMN_CTNT,'$.metadata."navigation/manifest.json".value.schema_version') AS NAV_SCHEMA, CASE WHEN JSON_EXISTS(RFRN_POST_IFMN_CTNT,'$.metadata."navigation/manifest.json".value.scope_contract') THEN 'Y' ELSE 'N' END AS SCOPE_CONTRACT, JSON_VALUE(RFRN_POST_IFMN_CTNT,'$.metadata."navigation/manifest.json".value.legal_sha256') AS GLOSSARY_SHA, JSON_VALUE(RFRN_POST_IFMN_CTNT,'$.metadata."fulltext/manifest.json".value.granularity') AS FT_GRANULARITY, JSON_VALUE(RFRN_POST_IFMN_CTNT,'$.metadata."fulltext/manifest.json".value.tokenizer.name') AS TOKENIZER, JSON_VALUE(RFRN_POST_IFMN_CTNT,'$.metadata."fulltext/manifest.json".value.tokenizer.version') AS TOKENIZER_VERSION, DBMS_LOB.GETLENGTH(RFRN_POST_IFMN_CTNT) AS MANIFEST_CHARS FROM SIDDS.IDD_B003_LGL_SRCH_POST ORDER BY RFRN_POST_HASH_VALU

-- [P03] 세대별 검색 행 수 — 행이 많으면 수 분 걸림
SELECT RFRN_POST_HASH_VALU, COUNT(*) AS ROWS_N FROM SIDDS.IDD_B003_LGL_SRCH_ROW GROUP BY RFRN_POST_HASH_VALU ORDER BY RFRN_POST_HASH_VALU

-- [P04] 원문 세대별 문서 수
SELECT DTLS_HASH_VALU, COUNT(*) AS DOCS FROM SIDDS.IDD_B003_LGL_DOC_VER GROUP BY DTLS_HASH_VALU ORDER BY DTLS_HASH_VALU

-- [P05] 원문 게시본 — 언제 들어온 세대인지
SELECT BODY_POST_HASH_VALU, DTLS_HASH_VALU, RGSN_ASTM FROM SIDDS.IDD_B003_LGL_SRC_POST ORDER BY RGSN_ASTM

-- [P06] 벡터 — 세대마다 연결된 벡터 게시본과 벡터 수
SELECT h.RFRN_POST_HASH_VALU, h.VCTR_POST_ID, h.CHAN_NO, (SELECT COUNT(*) FROM SIDDS.IDD_B003_LGL_VCTR_ITEM i WHERE i.VCTR_POST_ID=h.VCTR_POST_ID) AS VECTORS FROM SIDDS.IDD_B003_LGL_VCTR_HEAD h ORDER BY h.RFRN_POST_HASH_VALU

-- [P07] 쓰고 있는 공간(MB) — 권한이 없어 ORA-00942가 나면 P07~P09는 DBA에게 요청
SELECT TABLESPACE_NAME, ROUND(SUM(BYTES)/1048576) AS USED_MB FROM DBA_SEGMENTS WHERE OWNER='SIDDS' GROUP BY TABLESPACE_NAME

-- [P08] 남은 공간(MB)
SELECT TABLESPACE_NAME, ROUND(SUM(BYTES)/1048576) AS FREE_MB FROM DBA_FREE_SPACE GROUP BY TABLESPACE_NAME ORDER BY TABLESPACE_NAME

-- [P09] SIDDS 공간 한도(MAX_BYTES -1이면 무제한)
SELECT TABLESPACE_NAME, ROUND(BYTES/1048576) AS USED_MB, MAX_BYTES FROM DBA_TS_QUOTAS WHERE USERNAME='SIDDS'

-- [P10] 이 계정(PIDD2APP)이 Gold 13표·Agent 중간표에 직접 받은 권한 — 새 세대 적재에 SELECT·INSERT, 채널 교체에 UPDATE, 옛 세대 정리에 DELETE 필요
SELECT TABLE_SCHEMA, TABLE_NAME, LISTAGG(PRIVILEGE, ',') WITHIN GROUP (ORDER BY PRIVILEGE) PRIVILEGES FROM ALL_TAB_PRIVS WHERE GRANTEE=USER AND ((TABLE_SCHEMA='SIDDS' AND TABLE_NAME IN ('IDD_B003_LGL_DOC_BLK','IDD_B003_LGL_DOC_UNIT','IDD_B003_LGL_DOC_VER','IDD_B003_LGL_ONTL_ANC','IDD_B003_LGL_ONTL_HEAD','IDD_B003_LGL_ONTL_POST','IDD_B003_LGL_SRCH_CHNL','IDD_B003_LGL_SRCH_POST','IDD_B003_LGL_SRCH_ROW','IDD_B003_LGL_SRC_POST','IDD_B003_LGL_VCTR_HEAD','IDD_B003_LGL_VCTR_ITEM','IDD_B003_LGL_VCTR_POST')) OR (TABLE_SCHEMA='SIDDA' AND TABLE_NAME IN ('LGL_AGT_BODY_A','LGL_AGT_REC_A','LGL_AGT_RUN_A','LGL_EXEC_ST_A','LGL_PIPE_RESULT_A'))) GROUP BY TABLE_SCHEMA, TABLE_NAME ORDER BY 1,2

-- [P11] 이 계정의 ANY 권한 (있으면 P10에 없어도 됨, ANALYZE ANY는 통계 갱신용)
SELECT PRIVILEGE FROM SESSION_PRIVS WHERE PRIVILEGE IN ('SELECT ANY TABLE','INSERT ANY TABLE','UPDATE ANY TABLE','DELETE ANY TABLE','CREATE ANY TABLE','CREATE ANY INDEX','ANALYZE ANY') ORDER BY 1
