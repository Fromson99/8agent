import hashlib,json,sys,zipfile
from pathlib import Path,PurePosixPath
with zipfile.ZipFile(sys.argv[1]) as z:
    names=z.namelist()
    if len(names)!=len(set(names)): raise ValueError('duplicate ZIP paths')
    manifest=json.loads(z.read('CODE_MANIFEST.json'))
    if manifest['format']!='8agent-code/1': raise ValueError('unsupported package')
    rows=manifest['files'];expected={row['path']:row for row in rows}
    if len(rows)!=len(expected) or set(names)!=set(expected)|{'CODE_MANIFEST.json'}: raise ValueError('ZIP inventory mismatch')
    for name,row in expected.items():
        pure=PurePosixPath(name)
        if pure.is_absolute() or '..' in pure.parts or '\\' in name or str(pure)!=name: raise ValueError('unsafe path')
        if pure.suffix.lower() in {'.js','.jsx','.mjs','.cjs'}: raise ValueError('excluded JS attachment')
        raw=z.read(name)
        if len(raw)!=row['bytes'] or hashlib.sha256(raw).hexdigest()!=row['sha256']: raise ValueError('package hash mismatch: '+name)
        target=Path(sys.argv[2])/name
        target.parent.mkdir(parents=True,exist_ok=True);target.write_bytes(raw)
print('PACKAGE_HASHES PASS')
PY_VERIFY_PACKAGE
