    for name,row in expected.items():
        pure=PurePosixPath(name)
        assert not pure.is_absolute() and '..' not in pure.parts and '\\' not in name
        assert pure.suffix.lower() not in {'.js','.jsx','.mjs','.cjs'}
        raw=z.read(name)
