VERSION = 'genos-time-report-20261107-r1'
DEFAULT_ROOT = '/nfs-root/diamo/8-agent/rag-request-reports'
KST = timezone(timedelta(hours=9))
MAX_REPORT_BYTES = 1024 * 1024 * 1024
MAX_ROW_BYTES = 4 * 1024 * 1024
LIST_LIMIT = 12
LOOK_BACK_DAYS = 3
# Journal records that only bracket a model call; they say nothing about work done between calls.
QUIET_KINDS = frozenset({'MODEL_VIEW', 'WIRE_PROJECTION', 'MODEL_OUTPUT_CONTRACT', 'MODEL_MINIMAL_OUTPUT_CONTRACT',
                         'MODEL_CORRECTION_OWNER', 'ATTEMPT_RESERVED', 'ATTEMPT_SENT', 'ATTEMPT_FINISHED',

MODEL_END = ('model.call.finished', 'model.attempt.received', 'model.attempt.failed')
OTHER_SKIP = ('step.', 'journal.transaction.timings', 'model.', 'oracle.sql')


class Stop(Exception):
    def __init__(self, code, message):
        super().__init__(message)
        self.code = code


def width(text):
    return sum(2 if unicodedata.east_asian_width(ch) in 'WF' else 1 for ch in text)


def pad(text, size, right=False):
    text = str(text)
    gap = max(0, size - width(text))
    return (' ' * gap + text) if right else (text + ' ' * gap)


def cut(text, size):
    text = str(text)
    if width(text) <= size:
        return text
    out = ''
    for ch in text:
        if width(out + ch) > size - 1:
            break
        out += ch
    return out + '…'


def sec(ms):
    return f'{ms / 1000:,.1f}초'


def pct(part, whole):
    return f'{100 * part / whole:5.1f}%' if whole else '   - '


def bar(part, whole, size=20):
    if not whole:
        return ''
    filled = max(0, min(size, int(round(size * part / whole))))
    return '#' * filled + '.' * (size - filled)


def num(value):
    return value if type(value) in (int, float) and math.isfinite(value) else None


def stamp(value):
    if not isinstance(value, str):
        return None
    try:
        parsed = datetime.fromisoformat(value.replace('Z', '+00:00'))
    except ValueError:
        return None
    return parsed if parsed.tzinfo else parsed.replace(tzinfo=timezone.utc)

def report_root(argument):
    for value in (argument, os.environ.get('GENOS_SHARED_LOG_DIR', '').strip(), DEFAULT_ROOT):
        if value:
            root = Path(value).expanduser()
            if root.is_dir():
                return root
            if value == argument:
                raise Stop(2, f'요청 보고서 폴더가 없습니다: {root}')
    raise Stop(2, '요청 보고서 폴더를 찾지 못했습니다. 코드서빙의 GENOS_SHARED_LOG_DIR 값을 --root 로 주세요.\n'
                  f'  기본 위치 {DEFAULT_ROOT} 없음. 확인: ls -d /nfs-root/*/8-agent/rag-request-reports')


def recent_requests(root):
    today = datetime.now(KST).date()
    rows, unreadable = [], 0
    for back in range(LOOK_BACK_DAYS + 1):
        day = root / (today - timedelta(days=back)).isoformat()
        if not day.is_dir():
            continue
        seen = set()
        for row_path in day.glob('*/*/_qa/*.row'):
            try:
                with row_path.open('rb') as source:
                    raw = source.read(MAX_ROW_BYTES + 1)
                if len(raw) > MAX_ROW_BYTES:
                    raise ValueError
                row = json.loads(raw)
                when = stamp(row['requested_at'])
                if when is None:
                    raise ValueError
                report = row_path.parent.parent / str(row.get('report') or '')
                seen.add(report.name)
                rows.append({'when': when, 'seconds': num(row.get('seconds')), 'team': str(row.get('team') or ''),
                             'component': row_path.parent.parent.name, 'status': str(row.get('status') or ''),
                             'http': row.get('http_status'), 'report': report})
            except (OSError, ValueError, TypeError, KeyError):
                unreadable += 1
        # A request whose summary row was not written still has its JSON report.
        for report in day.glob('*/*/*.json'):
            if report.name in seen:
                continue
            try:
                when = datetime.fromtimestamp(report.stat().st_mtime, timezone.utc)
            except OSError:
                continue
            rows.append({'when': when, 'seconds': None, 'team': report.parent.parent.name,
                         'component': report.parent.name, 'status': '요약 줄 없음(파일 시각)', 'http': None,
                         'report': report})
        if len(rows) >= LIST_LIMIT:
            break
    rows.sort(key=lambda r: r['when'], reverse=True)
    return rows, unreadable


def print_list(root, rows, unreadable, out):
    out(f'요청 보고서 위치: {root}')
    out(f'최근 요청 {min(len(rows), LIST_LIMIT)}건 (최근 {LOOK_BACK_DAYS + 1}일, 새것부터)'
        + (f' · 읽지 못한 요약 줄 {unreadable}건' if unreadable else ''))
    out(pad('번호', 5) + pad('요청 시각(한국)', 21) + pad('서버 처리', 11, True) + '  ' + pad('HTTP', 5)
        + pad('상태', 12) + '팀/구분')
    for index, row in enumerate(rows[:LIST_LIMIT], 1):
        seconds = f'{row["seconds"]:,.1f}초' if row['seconds'] is not None else '-'
        exists = '' if row['report'].is_file() else '  (JSON 아직 없음)'
        out(pad(f'[{index}]', 5) + pad(row['when'].astimezone(KST).strftime('%m-%d %H:%M:%S'), 21)
            + pad(seconds, 11, True) + '  ' + pad(str(row['http'] or '-'), 5) + pad(cut(row['status'], 11), 12)
            + f'{row["team"]}/{row["component"]}{exists}')
    if not rows:
        out('요청이 없습니다. --root 위치와 날짜 폴더를 확인하세요.')
    out('특정 요청 보고: bash ./GENOS_TIME_REPORT_R1.txt <번호>')


def pick(root, choice):
    rows, unreadable = recent_requests(root)
    if not rows:
        raise Stop(3, f'최근 {LOOK_BACK_DAYS + 1}일 안의 요청 보고서가 없습니다: {root}')
    index = int(choice) if choice else 1
    if not 1 <= index <= min(len(rows), LIST_LIMIT):
        raise Stop(2, f'번호는 1~{min(len(rows), LIST_LIMIT)} 입니다. 목록: bash ./GENOS_TIME_REPORT_R1.txt list')
    row = rows[index - 1]
    if not row['report'].is_file():
        raise Stop(3, f'요청 보고서 JSON이 아직 없습니다(응답 뒤 기록 중이거나 기록 실패): {row["report"]}\n'
                      '  10~20초 뒤 다시 실행하세요. 계속 없으면 같은 이름의 .log 파일을 확인합니다.')
    return row['report']


def load(path):
    try:
        size = path.stat().st_size
    except OSError as exc:
        raise Stop(3, f'보고서를 읽지 못했습니다: {path} ({type(exc).__name__})')
    if size > MAX_REPORT_BYTES:
        raise Stop(3, f'보고서가 너무 큽니다({size:,} bytes): {path}')
    opener = open
    if path.suffix == '.gz':
        import gzip
        opener = gzip.open
    try:
        with opener(path, 'rt', encoding='utf-8') as source:
            report = json.load(source)
    except (OSError, ValueError) as exc:
        raise Stop(3, f'보고서 JSON을 읽지 못했습니다: {path} ({type(exc).__name__})')
    if not isinstance(report, dict) or not isinstance(report.get('service_events'), list):
        raise Stop(3, f'요청 보고서 형식이 아닙니다(service_events 없음): {path}')
    return report, size

def journal_events(journal):
    """Every timeline record of the run journal and any child journals, with time and kind."""
    found = []

    def walk(node, depth):
        if depth > 6:
            return
        if isinstance(node, dict):
            timeline = node.get('timeline')
            if isinstance(timeline, list):
                for item in timeline:
                    event = item.get('event') if isinstance(item, dict) else None
                    if isinstance(event, dict):
                        when = stamp(event.get('created_at'))
                        if when is not None:
                            found.append((when, str(event.get('kind') or '?')))
            for key, value in node.items():
                if key != 'timeline' and isinstance(value, (dict, list)) and key in ('children', 'child', 'journal'):
                    walk(value, depth + 1)
        elif isinstance(node, list):
            for value in node:
                walk(value, depth + 1)

    walk(journal, 0)
    found.sort(key=lambda item: item[0])
    return found


def analyse(report):
    events = []
    for row in report['service_events']:
        if isinstance(row, dict):
            when = stamp(row.get('ts'))
            if when is not None:
                events.append((when, row))
    events.sort(key=lambda item: item[0])
    if not events:
        raise Stop(3, '보고서에 시각이 있는 서비스 기록이 없습니다.')
    first = lambda name: next((row for when, row in events if row.get('event') == name), None)
    started_row = first('request.started')
    start = stamp(started_row['ts']) if started_row else events[0][0]
    returning = [row for when, row in events if row.get('event') in ('request.returning', 'request.finished')]
    total_ms = num(returning[-1].get('duration_ms')) if returning else None
    if total_ms is None:
        total_ms = (events[-1][0] - start).total_seconds() * 1000
    end = start + timedelta(milliseconds=total_ms)

    def offset(when):
        return (when - start).total_seconds() * 1000

    # Model calls: started → first end event of the same attempt.
    calls, open_calls = [], {}
    for when, row in events:
        name, attempt = row.get('event'), row.get('attempt_id')
        if not attempt:
            continue
        if name == 'model.attempt.started':
            open_calls[attempt] = {'start': when, 'end': None, 'stage': row.get('source_stage') or row.get('phase') or '?',
                                   'schema': row.get('schema') or '', 'attempt_no': row.get('attempt_no'),
                                   'status': None, 'prompt_tokens': None, 'completion_tokens': None}
            calls.append(open_calls[attempt])
        elif name in MODEL_END and attempt in open_calls:
            call = open_calls[attempt]
            if call['end'] is None or name != 'model.call.finished':
                if call['end'] is None:
                    call['end'] = when
                if name == 'model.attempt.received':
                    call['status'] = row.get('http_status')
                    call['prompt_tokens'] = num(row.get('prompt_tokens'))
                    call['completion_tokens'] = num(row.get('completion_tokens'))
                elif name == 'model.attempt.failed':
                    call['status'] = row.get('error_type') or row.get('failure_kind') or 'failed'
    for call in calls:
        if call['end'] is None:
            call['end'], call['status'] = end, call['status'] or '끝 기록 없음'
    calls = [c for c in calls if c['end'] >= c['start']]
    model_ms = sum((c['end'] - c['start']).total_seconds() * 1000 for c in calls)
    metrics = next((row.get('execution_stats') for when, row in reversed(events)
                    if row.get('event') == 'execution.metrics' and isinstance(row.get('execution_stats'), dict)), None)
    ui_model_ms = num(metrics.get('llm_duration_ms')) if metrics else None

    # Segments: G0, M1, G1, M2, ..., Gn.
    segments, cursor = [], start
    for index, call in enumerate(calls, 1):
        if call['start'] > cursor:
            segments.append({'kind': 'G', 'start': cursor, 'end': call['start'], 'before': calls[index - 2] if index > 1 else None,
                             'after': call, 'number': index - 1})
        segments.append({'kind': 'M', 'start': call['start'], 'end': call['end'], 'call': call, 'number': index})
        cursor = max(cursor, call['end'])
    if end > cursor:
        segments.append({'kind': 'G', 'start': cursor, 'end': end, 'before': calls[-1] if calls else None,
                         'after': None, 'number': len(calls)})
    for segment in segments:
        segment['ms'] = (segment['end'] - segment['start']).total_seconds() * 1000
        segment.update(sql_slow=[0, 0.0], sql_debug=[0, 0.0], commits=[0, 0.0], steps=defaultdict(float),
                       kinds=Counter(), others=Counter())

    def segment_of(when):
        for segment in segments:
            if segment['start'] <= when <= segment['end']:
                return segment
        return None

    sql_summary = next((row for when, row in reversed(events) if row.get('event') == 'oracle.sql.summary'), None)
    slow, debug_sql, failed_sql = [], [], []
    commits = [0, 0.0, 0.0]
    steps = {}
    last_label = '요청 시작'
    for when, row in events:
        name = row.get('event') or '?'
        segment = segment_of(when)
        if name == 'oracle.sql.slow':
            ms = num(row.get('duration_ms')) or 0.0
            slow.append({'when': when, 'ms': ms, 'family': str(row.get('sql_family') or '?'),
                         'write': bool(row.get('write')), 'segment': segment, 'context': last_label})
            if segment:
                segment['sql_slow'][0] += 1; segment['sql_slow'][1] += ms
        elif name == 'oracle.sql':
            status = row.get('status')
            ms = num(row.get('duration_ms'))
            if status == 'ACKNOWLEDGED' and ms is not None:
                debug_sql.append({'when': when, 'ms': ms, 'family': str(row.get('sql_family') or '?')})
                if segment:
                    segment['sql_debug'][0] += 1; segment['sql_debug'][1] += ms
            elif status not in ('STARTED', 'ACKNOWLEDGED'):
                failed_sql.append({'when': when, 'status': str(status), 'family': str(row.get('sql_family') or '?'),
                                   'ms': ms or 0.0})
        elif name == 'journal.transaction.timings':
            ms = sum(num(row.get(k)) or 0.0 for k in ('begin_ms', 'body_ms', 'commit_ms'))
            commits[0] += 1; commits[1] += ms; commits[2] += num(row.get('commit_ms')) or 0.0
            if segment:
                segment['commits'][0] += 1; segment['commits'][1] += ms
        elif name in ('step.finished', 'step.failed'):
            label = str(row.get('step') or '?')
            ms = num(row.get('duration_ms')) or 0.0
            item = steps.setdefault(label, [0, 0.0, 0.0, 0.0, 0])
            item[0] += 1; item[1] += ms; item[2] = max(item[2], ms); item[3] += num(row.get('thread_cpu_ms')) or 0.0
            item[4] += name == 'step.failed'
            if segment:
                segment['steps'][label] += ms
        elif not name.startswith(OTHER_SKIP) and name not in ('request.started', 'request.returning', 'request.finished'):
            if segment:
                segment['others'][name] += 1
        if name == 'step.started':
            last_label = 'step ' + str(row.get('step') or '?') + ' 시작'
        elif name in ('step.finished', 'step.failed'):
            last_label = 'step ' + str(row.get('step') or '?') + ' 끝'
        elif name not in ('oracle.sql', 'oracle.sql.slow', 'journal.transaction.timings'):
            last_label = name

    journal = journal_events(report.get('journal'))
    for when, kind in journal:
        segment = segment_of(when)
        if segment and kind not in QUIET_KINDS:
            segment['kinds'][kind] += 1

    # Longest stretches with no record at all, outside model calls.
    step_word = {'step.started': '시작', 'step.finished': '끝', 'step.failed': '실패'}
    marks = [(when, f"{row.get('step')} {step_word[row.get('event')]}" if row.get('event') in step_word
              else str(row.get('event'))) for when, row in events]
    marks += [(when, 'journal:' + kind) for when, kind in journal]
    for item in slow:
        marks.append((item['when'] - timedelta(milliseconds=item['ms']), 'SQL ' + item['family'][:8] + ' 시작'))
    marks.sort(key=lambda item: item[0])
    quiet = []
    for (a, label_a), (b, label_b) in zip(marks, marks[1:]):
        if b <= a or a < start or b > end + timedelta(seconds=1):
            continue
        inside = sum(max(0.0, (min(b, c['end']) - max(a, c['start'])).total_seconds()) for c in calls)
        span = (b - a).total_seconds() - inside
        if span * 1000 >= 500:
            quiet.append({'start': a, 'ms': span * 1000, 'before': label_a, 'after': label_b})
    quiet.sort(key=lambda item: -item['ms'])

    others = Counter()
    for when, row in events:
        name = row.get('event') or ''
        if row.get('level') in ('WARNING', 'ERROR', 'CRITICAL') and name != 'oracle.sql':
            others[name] += 1
    release = next((row for when, row in events if row.get('event') == 'oracle.release.selected'), None)
    return {'start': start, 'end': end, 'total_ms': total_ms, 'calls': calls, 'model_ms': model_ms,
            'ui_model_ms': ui_model_ms, 'metrics': metrics, 'segments': segments, 'sql_summary': sql_summary,
            'slow': slow, 'debug_sql': debug_sql, 'failed_sql': failed_sql, 'commits': commits, 'steps': steps,
            'quiet': quiet, 'warnings': others, 'release': release, 'journal_count': len(journal),
            'event_count': len(events), 'offset': offset}

def families(text):
    out = []
    for part in str(text or '').split('-'):
        bits = part.split(':')
        if len(bits) == 3 and bits[1].isdigit():
            try:
                out.append((bits[0], int(bits[1]), float(bits[2])))
            except ValueError:
                pass
    return out


def render(path, size, report, a, out):
    total = a['total_ms'] or 0.0
    offset = a['offset']
    result = report.get('result') if isinstance(report.get('result'), dict) else {}
    body = result.get('body') if isinstance(result.get('body'), dict) else {}
    request = report.get('request') if isinstance(report.get('request'), dict) else {}
    inner = request.get('json') if isinstance(request.get('json'), dict) else request
    question = inner.get('question') or inner.get('q') or ''
    answer = body.get('answer') if isinstance(body.get('answer'), dict) else {}
    out('=' * 78)
    out(f'GENOS 처리 시간 보고 R1  ({VERSION})')
    out(f'보고서  {path}  ({size / 1048576:,.1f}MB)')
    out(f'요청    {a["start"].astimezone(KST):%Y-%m-%d %H:%M:%S} KST · 팀 {report.get("team_id")} · '
        f'HTTP {result.get("status", "-")} · 질문 {len(str(question))}자 · 답변 {len(str(answer.get("text") or ""))}자')
    release = a['release']
    if release:
        out(f'데이터  채널 {release.get("query_channel")} · 게시 {str(release.get("query_release_id"))[:24]} · '
            f'공유캐시 {"켜짐" if release.get("serving_data_cache") else "꺼짐"}')
    out(f'기록    서비스 이벤트 {a["event_count"]:,}개 · 원장 기록 {a["journal_count"]:,}개')
    out('=' * 78)

    # [1] 전체 분해
    summary = a['sql_summary']
    sql_ms = num(summary.get('sql_ms')) if summary else None
    sql_n = num(summary.get('sql_count')) if summary else None
    model_ms = a['model_ms']
    other_ms = total - model_ms - (sql_ms or 0.0)
    out(f'[1] 전체 {sec(total)} 어디에 썼나 (요청 시작→응답 반환)')
    out('    ' + pad('모델 호출', 14) + pad(sec(model_ms), 10, True) + ' ' + pct(model_ms, total) + '  '
        + bar(model_ms, total) + f'  {len(a["calls"])}회'
        + (f' (화면 표시 {sec(a["ui_model_ms"])})' if a['ui_model_ms'] is not None else ''))
    if summary:
        avg = (sql_ms / sql_n) if sql_n else 0.0
        out('    ' + pad('Oracle SQL', 14) + pad(sec(sql_ms), 10, True) + ' ' + pct(sql_ms, total) + '  '
            + bar(sql_ms, total) + f'  {int(sql_n):,}회 · 평균 {avg:,.0f}ms/회')
    else:
        out('    ' + pad('Oracle SQL', 14) + '  요약 줄(oracle.sql.summary) 없음 — MAIN167 이전 코드이거나 DP=oracle 아님')
    out('    ' + pad('그 외', 14) + pad(sec(max(other_ms, 0.0)), 10, True) + ' ' + pct(max(other_ms, 0.0), total) + '  '
        + bar(max(other_ms, 0.0), total) + '  파이썬 계산·기록 저장·로그·대기')
    if other_ms < -1000:
        out('    ※ SQL 합이 전체보다 큼: 여러 SQL이 동시에 실행된 구간이 있음(합계는 겹친 시간 포함)')
    if summary:
        out(f'    SQL 세부  쓰기 {int(num(summary.get("sql_write_count")) or 0):,}회 · 실패 '
            f'{int(num(summary.get("sql_failed_count")) or 0):,}회 · 가장 느린 1회 {sec(num(summary.get("sql_slowest_ms")) or 0)} · '
            f'공유캐시 적중 {int(num(summary.get("sql_cache_hits")) or 0):,} / 미적중 {int(num(summary.get("sql_cache_misses")) or 0):,}')
    commits = a['commits']
    if commits[0]:
        out(f'    원장 저장 {commits[0]:,}회 {sec(commits[1])} (그중 커밋 {sec(commits[2])}; Oracle 쓰기 SQL 시간이 여기에 포함될 수 있음)')
    out('')

    # [2] 모델 호출과 그 사이
    out('[2] 순서대로: M=모델 호출, G=모델 호출 사이(모델 밖)   +시작후 경과 / 걸린 시간')
    segments = a['segments']
    detail_cut = max(total * 0.03, 1000.0)
    shown = segments
    if len(segments) > 25:
        keep = {id(s) for s in sorted(segments, key=lambda s: -s['ms'])[:20]}
        shown = [s for s in segments if id(s) in keep]
        out(f'    (구간 {len(segments)}개 중 긴 20개만)')
    for segment in shown:
        head = f'{segment["kind"]}{segment["number"]}'
        line = '    ' + pad(head, 5) + pad(f'+{offset(segment["start"]) / 1000:,.1f}s', 10, True) + '  ' \
            + pad(sec(segment['ms']), 9, True) + ' ' + pct(segment['ms'], total) + '  '
        if segment['kind'] == 'M':
            call = segment['call']
            tokens = ''
            if call['prompt_tokens'] is not None:
                tokens = f' · 토큰 {int(call["prompt_tokens"]):,}→{int(call["completion_tokens"] or 0):,}'
            status = '' if call['status'] in (200, None) else f' · 상태 {call["status"]}'
            out(line + cut(f'{call["stage"]} ({call["schema"]})' + tokens + status, 62))
            continue
        before = segment['before']['stage'] if segment['before'] else '요청 시작'
        after = segment['after']['stage'] if segment['after'] else '응답 반환'
        out(line + cut(f'{before} → {after}', 58) + ('  ◀ 김' if segment['ms'] >= total * 0.15 else ''))
        if segment['ms'] < detail_cut:
            continue
        parts = []
        if segment['sql_slow'][0]:
            parts.append(f'느린 SQL {segment["sql_slow"][0]}개 {sec(segment["sql_slow"][1])}')
        if segment['sql_debug'][0]:
            parts.append(f'SQL {segment["sql_debug"][0]:,}회 {sec(segment["sql_debug"][1])}')
        if segment['commits'][0]:
            parts.append(f'원장 저장 {segment["commits"][0]}회 {sec(segment["commits"][1])}')
        top_steps = sorted(segment['steps'].items(), key=lambda item: -item[1])[:3]
        if top_steps:
            parts.append('단계 ' + ', '.join(f'{name} {sec(ms)}' for name, ms in top_steps))
        if parts:
            out('          └ ' + cut(' · '.join(parts), 92))
        notes = [f'{kind}×{n}' if n > 1 else kind for kind, n in segment['kinds'].most_common(4)]
        notes += [f'{name}×{n}' if n > 1 else name for name, n in segment['others'].most_common(2)]
        if notes:
            out('          └ 기록: ' + cut(', '.join(notes), 86))
    out('')

    # [3] SQL
    out('[3] Oracle SQL')
    tops = families(summary.get('sql_top')) if summary else []
    if tops:
        out('    시간 많이 쓴 SQL 종류(문장 해시 앞 12자, 요청 전체 합계 상위 3)')
        for family, count, ms in tops:
            out('      ' + pad(family, 14) + pad(f'{count:,}회', 9, True) + pad(sec(ms), 10, True) + ' '
                + pct(ms, total) + f'  평균 {ms / count if count else 0:,.0f}ms')
    debug = a['debug_sql']
    if debug:
        grouped = defaultdict(lambda: [0, 0.0, 0.0])
        for item in debug:
            row = grouped[item['family']]; row[0] += 1; row[1] += item['ms']; row[2] = max(row[2], item['ms'])
        out(f'    SQL 하나하나 기록 있음(DEBUG): {len(debug):,}회 {sec(sum(i["ms"] for i in debug))} — 종류별 상위 8')
        for family, (count, ms, worst) in sorted(grouped.items(), key=lambda item: -item[1][1])[:8]:
            out('      ' + pad(family, 14) + pad(f'{count:,}회', 9, True) + pad(sec(ms), 10, True)
                + f'  최대 {worst:,.0f}ms')
    slow = a['slow']
    if slow:
        out(f'    느린 SQL(기본 1초 이상, GENOS_ORACLE_SQL_SLOW_MS) {len(slow)}개, 합계 {sec(sum(i["ms"] for i in slow))} — 느린 순 상위 10')
        for item in sorted(slow, key=lambda i: -i['ms'])[:10]:
            seg = item['segment']
            where = f'{seg["kind"]}{seg["number"]}' if seg else '-'
            out('      ' + pad(f'+{offset(item["when"]) / 1000:,.1f}s', 9, True) + pad(sec(item['ms']), 9, True)
                + '  ' + pad(item['family'][:12], 13) + pad(where, 4) + ('쓰기 ' if item['write'] else '')
                + cut('직전: ' + item['context'], 40))
    elif summary:
        out('    느린 SQL(기본 1초 이상) 없음 → 짧은 SQL이 많이 반복된 것(횟수 문제)')
    if a['failed_sql']:
        out(f'    실패 SQL {len(a["failed_sql"])}개: ' + cut(', '.join(f'{i["status"]}:{i["family"][:8]}'
                                                               for i in a['failed_sql'][:4]), 56))
    out('')

    # [4] 단계
    steps = sorted(a['steps'].items(), key=lambda item: -item[1][1])[:12]
    out('[4] 시간 많이 쓴 단계(step) 상위 12 — 단계끼리 포함 관계가 있어 합이 전체보다 클 수 있음')
    if steps:
        out('    ' + pad('단계', 34) + pad('횟수', 6, True) + pad('합계', 10, True) + pad('최대 1회', 10, True)
            + pad('CPU', 9, True) + pad('대기', 9, True))
        for name, (count, ms, worst, cpu, failed) in steps:
            out('    ' + pad(cut(name, 33), 34) + pad(f'{count:,}', 6, True) + pad(sec(ms), 10, True)
                + pad(sec(worst), 10, True) + pad(sec(cpu), 9, True) + pad(sec(max(ms - cpu, 0)), 9, True)
                + (f' 실패{failed}' if failed else ''))
        out('    (대기 = 합계-CPU: SQL·네트워크·파일 기다린 시간)')
    else:
        out('    단계 기록 없음')
    out('')

    # [5] 기록 없는 긴 공백
    out('[5] 모델 밖에서 아무 기록 없이 흐른 긴 시간 상위 6 (이 사이에 시간을 쓴 코드가 있음)')
    for item in a['quiet'][:6]:
        out('    ' + pad(f'+{offset(item["start"]) / 1000:,.1f}s', 9, True) + pad(sec(item['ms']), 9, True) + '  '
            + cut(f'{item["before"]} → {item["after"]}', 58))
    if not a['quiet']:
        out('    0.5초 넘는 공백 없음')
    out('')

    # [6] 읽는 법
    out('[6] 한눈에')
    if summary and sql_ms is not None and total:
        share = sql_ms / total
        avg = sql_ms / sql_n if sql_n else 0.0
        if share >= 0.5:
            reason = ('짧은 SQL을 아주 많이 보냄(횟수를 줄여야 함)' if sql_n and sql_n >= 300 and avg < 1000 else
                      '무거운 SQL 몇 개가 오래 걸림' if a['slow'] and sum(i['ms'] for i in a['slow']) >= sql_ms * 0.5 else
                      'SQL 횟수·건당 시간 모두 확인 필요')
            out(f'    Oracle SQL이 전체의 {share * 100:.0f}% ({int(sql_n or 0):,}회 × 평균 {avg:,.0f}ms) → {reason}')
        else:
            out(f'    Oracle SQL은 전체의 {share * 100:.0f}% ({int(sql_n or 0):,}회 × 평균 {avg:,.0f}ms)')
    if total:
        out(f'    모델 호출 {model_ms / total * 100:.0f}% · 모델 밖 {100 - model_ms / total * 100:.0f}%')
    biggest = max((s for s in segments if s['kind'] == 'G'), key=lambda s: s['ms'], default=None)
    if biggest:
        out(f'    가장 긴 모델 밖 구간: G{biggest["number"]} {sec(biggest["ms"])} ('
            + (biggest['before']['stage'] if biggest['before'] else '요청 시작') + ' → '
            + (biggest['after']['stage'] if biggest['after'] else '응답 반환') + ')')
    if a['warnings']:
        out('    경고·오류 기록: ' + cut(', '.join(f'{n}×{c}' if c > 1 else n for n, c in a['warnings'].most_common(5)), 60))
    out('    질문·답변·원문·SQL 본문은 출력하지 않았습니다.')
    out('=' * 78)


def main(argv):
    root_arg, rest = None, []
    items = list(argv)
    while items:
        item = items.pop(0)
        if item == '--root':
            if not items:
                raise Stop(2, '--root 다음에 폴더를 주세요.')
            root_arg = items.pop(0)
        elif item in ('-h', '--help', 'help'):
            print(__doc__)
            return 0
        else:
            rest.append(item)
    lines = []

    def out(text=''):
        lines.append(text)
        print(text, flush=True)

    if rest and rest[0] == 'list':
        root = report_root(root_arg)
        rows, unreadable = recent_requests(root)
        print_list(root, rows, unreadable, out)
        return 0
    if rest and (rest[0].endswith('.json') or rest[0].endswith('.json.gz') or '/' in rest[0]):
        path = Path(rest[0]).expanduser()
        if not path.is_file():
            raise Stop(2, f'파일이 없습니다: {path}')
    else:
        if rest and not rest[0].isdigit():
            raise Stop(2, f'알 수 없는 인자: {rest[0]}  (list / 번호 / 보고서.json 경로)')
        path = pick(report_root(root_arg), rest[0] if rest else None)
    report, size = load(path)
    render(path, size, report, analyse(report), out)
    return 0


if __name__ == '__main__':
    try:
        sys.exit(main(sys.argv[1:]))
    except Stop as stop:
        print('중단: ' + str(stop), file=sys.stderr)
        sys.exit(stop.code)
