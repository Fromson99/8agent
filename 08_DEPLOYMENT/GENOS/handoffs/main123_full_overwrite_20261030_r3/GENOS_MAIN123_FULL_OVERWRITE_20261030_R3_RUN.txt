#!/usr/bin/env bash
set -Eeuo pipefail
umask 077
TASK_LOG_DIR="$PWD/main123-full-overwrite-r3-logs"
if [[ -L "$TASK_LOG_DIR" ]]; then echo "Refuse symlink log folder" >&2; exit 2; fi
mkdir -p "$TASK_LOG_DIR"
TASK_LOG="$TASK_LOG_DIR/$(date -u +%Y%m%dT%H%M%SZ)-$$.log"
exec > >(tee -a "$TASK_LOG") 2>&1
trap 'TASK_RC=$?; printf "%s\n" "$TASK_RC" > "$TASK_LOG.exit"; echo "MAIN123_FULL_EXIT=$TASK_RC LOG=$TASK_LOG"' EXIT
python3 -u - "$@" <<'MAIN123_FULL_PY'
"""Build the ordinary release-gate ZIP locally, never as a nested mail attachment."""
import json
from pathlib import Path
import zipfile


def materialize_code_zip(package):
    spec = json.loads((package / 'PACKAGE.json').read_bytes())
    with zipfile.ZipFile(package / 'code.zip', 'w', compression=zipfile.ZIP_STORED) as archive:
        content = [('CODE_MANIFEST.json', (package / 'CODE_MANIFEST.json').read_bytes())]
        content += [(name, (package / spec['payload_paths'][name]).read_bytes()) for name in sorted(spec['code'])]
        for name, raw in content:
            info = zipfile.ZipInfo(name, date_time=(1980, 1, 1, 0, 0, 0))
            info.create_system = 3
            info.external_attr = 0o100644 << 16
            info.compress_type = zipfile.ZIP_STORED
            archive.writestr(info, raw)

"""Verify the full R3 delivery, prepare it in TRANSFER, and invoke the installer."""
import argparse
import fcntl
import hashlib
import json
from pathlib import Path, PurePosixPath
import stat
import subprocess
import sys
import tempfile
import zipfile

NAME = 'GENOS_MAIN123_FULL_OVERWRITE_20261030_R3'
ZIP_SHA256 = 'e4b2e731e1d8f3e5835f27dfbad2c2277e6476367c3b2180a32d602f96fc3533'
SPEC_SHA256 = '9146c4379d7a59b20d6e923507a2b33f6726f1a26ede3a8ee5f1b4a4429b8cd6'


def digest(raw):
    return hashlib.sha256(raw).hexdigest()


def plain(path):
    if any(p.is_symlink() for p in (path, *path.parents)):
        raise ValueError('Symlink path refused: ' + str(path))
    return path


def verify(package, *, include_generated=True):
    plain(package)
    raw = plain(package / 'PACKAGE.json').read_bytes()
    if digest(raw) != SPEC_SHA256:
        raise ValueError('R3 package manifest changed; existing state preserved')
    for name, expected in json.loads(raw)['payload'].items():
        if not include_generated and name == 'code.zip':
            continue
        if digest(plain(package / name).read_bytes()) != expected:
            raise ValueError('R3 package file changed: ' + name)


def prepare(transfer):
    package = plain(transfer / NAME)
    if package.exists():
        verify(package)
        print('REUSE_PACKAGE: existing backup and install state retained', flush=True)
        return package
    archive = plain(transfer / (NAME + '.zip'))
    if not archive.is_file():
        raise ValueError('Run in the uploaded TRANSFER folder containing ' + archive.name)
    if digest(archive.read_bytes()) != ZIP_SHA256:
        raise ValueError('R3 ZIP hash mismatch; no code changed')
    staging = Path(tempfile.mkdtemp(prefix='.' + NAME + '.prepare-', dir=transfer))
    print('PREPARING_PACKAGE=' + str(staging), flush=True)
    with zipfile.ZipFile(archive) as source:
        members = source.infolist()
        names = [i.filename for i in members]
        if len(names) != len(set(names)):
            raise ValueError('Duplicate archive member')
        for member in members:
            path = PurePosixPath(member.filename)
            if (path.is_absolute() or '..' in path.parts or '\\' in member.filename
                    or stat.S_ISLNK(member.external_attr >> 16)):
                raise ValueError('Unsafe archive member: ' + member.filename)
        source.extractall(staging)
    verify(staging, include_generated=False)
    materialize_code_zip(staging)
    verify(staging)
    staging.rename(package)
    return package


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('command', nargs='?', default='apply',
                        choices=['check', 'apply', 'gate', 'gate-code', 'gate-sqlite', 'rollback', 'status'])
    parser.add_argument('--root', type=Path, default=Path.home() / 'DEV')
    args = parser.parse_args()
    transfer = Path.cwd().resolve(strict=True)
    plain(transfer)
    with plain(transfer / '.main123-full-overwrite-r3.lock').open('a') as lock:
        fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        package = prepare(transfer)
        print('TRANSFER=' + str(transfer), flush=True)
        print('PACKAGE=' + str(package), flush=True)
        print('DEV_REQUESTED=' + str(args.root), flush=True)
        command = [sys.executable, '-u', str(package / 'runner.py')]
        if args.command == 'apply':
            checked = subprocess.call(command + ['check', '--root', str(args.root)])
            if checked:
                return checked
        return subprocess.call(command + [args.command, '--root', str(args.root)])


if __name__ == '__main__':
    try:
        raise SystemExit(main())
    except (OSError, ValueError, KeyError, zipfile.BadZipFile) as exc:
        print('MAIN123_FULL_FAILED: ' + str(exc), file=sys.stderr, flush=True)
        raise SystemExit(2)
