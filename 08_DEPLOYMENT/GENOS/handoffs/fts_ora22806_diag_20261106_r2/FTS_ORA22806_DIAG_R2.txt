def codes(text):
    return sorted(set(re.findall(r'ORA-\d{5}', str(text)[:65536])))[:8]


def details(item):
    """Allowlisted facts from one ORDS error item; never the raw text."""
    out = {'oracle_codes': set(), 'messages': set(), 'positions': set(), 'error_keys': set()}
    if any(k in item for k in ('errorCode', 'errorMessage', 'errorDetails')):
        out['error_keys'].update(k for k in item if isinstance(k, str) and KEY.match(k))
    line, column = (str(item.get(k, '')) for k in ('errorLine', 'errorColumn'))
    if re.fullmatch(r'\d{1,6}', line) and re.fullmatch(r'\d{1,7}', column):
        out['positions'].add((int(line), int(column)))
    number = str(item.get('errorCode', ''))
    if re.fullmatch(r'-?\d{1,5}', number) and int(number):
        out['oracle_codes'].add('ORA-' + str(abs(int(number))).zfill(5))
    for key in ('errorMessage', 'errorDetails', 'message'):
        text = item.get(key)
        if not isinstance(text, str):
            continue
        out['oracle_codes'].update(codes(text))
        for line in text.splitlines():
            if MESSAGE.match(line.strip()):
                out['messages'].add(line.strip())
        for line, column in POSITION.findall(text):
            out['positions'].add((int(line), int(column)))
    return out


class Observer:
    """Delegating ORDS client that keeps allowlisted facts of the last response."""
    def __init__(self, client):
        self.client = client
        self.last = None

    def __getattr__(self, name):
        return getattr(self.client, name)

    def request(self, method, path, json=None, **kw):
        self.last = {'oracle_codes': set(), 'messages': set(), 'positions': set(), 'http_status': set(), 'error_keys': set()}
        try:
            response = self.client.request(method, path, json=json, **kw)
        except Exception as exc:
            status = getattr(getattr(exc, 'response', None), 'status_code', None)
            if type(status) is int:
                self.last['http_status'].add(status)
            self.last['http_status'].update(int(s) for s in re.findall(r'ORDS_(?:AUTH|SQL)_HTTP:\s*(\d{3})', str(exc)))
            self.last['oracle_codes'].update(codes(str(exc)))
            for k, v in details({'errorMessage': str(exc)[:65536]}).items():
                if k != 'error_keys':
                    self.last[k].update(v)
            raise
        body = response if isinstance(response, dict) else getattr(response, 'raw', None)
        if not isinstance(body, dict):
            candidate = getattr(response, 'json', None)
            try:
                body = candidate() if callable(candidate) else candidate
            except Exception:
                body = None
        if isinstance(body, dict):
            items = body.get('items')
            for item in [body] + (items if isinstance(items, list) else []):
                if isinstance(item, dict):
                    for k, v in details(item).items():
                        self.last[k].update(v)
        return response


def safe_error(exc):
    tags, found, current, seen = set(), set(), exc, set()
    for _ in range(5):
        if current is None or id(current) in seen:
            break
        seen.add(id(current))
        text = str(current)[:65536]
        tags.update(t for t in TAGS if re.search(r'\b' + t + r'\b', text))
        found.update(codes(text))
        current = current.__cause__ or current.__context__
    return {'exception_type': type(exc).__name__, 'tags': sorted(tags), 'oracle_codes': sorted(found)}


def cte_body(cte, name):
    start = cte.index(name + ' AS (') + len(name + ' AS (')
    depth, i, quoted = 1, start, False
    while depth:
        ch = cte[i]
        if ch == "'":
            quoted = not quoted
        elif not quoted and ch == '(':
            depth += 1
        elif not quoted and ch == ')':
            depth -= 1
        i += 1
    return cte[start:i - 1]


def used(statement, binds):
    return [b for b in binds if re.search(':' + b[0] + r'\b', statement)]

        step(label + '_V02_PROJECTION', lambda: count(full, binds), full)
        parts = [('V03_POSTINGS', 'postings'), ('V04_LENGTHS', 'lengths')]
        if 'positions AS' in cte:
            parts.append(('V05_POSITIONS', 'positions'))
        for name, part in parts:
            statement = 'SELECT COUNT(*) AS N FROM (' + cte_body(cte, part) + ')'
            step(label + '_' + name, lambda s=statement: count(s, used(s, binds)), statement)
        matches = 'SELECT COUNT(*) AS N FROM (' + cte + 'SELECT k,did,sid,score FROM matches)'
        step(label + '_V06_MATCHES_COUNT', lambda: count(matches, binds), matches)
        documents = [b for b in binds if b[0] == 'q_documents'][0]
        jt = "SELECT COUNT(*) AS N FROM JSON_TABLE(:q_documents,'$[*]' COLUMNS(did VARCHAR2(512) PATH '$')) j"
        step(label + '_V07_JSONTABLE_CLOB_BIND', lambda: count(jt, [documents]), jt)
        small = len(documents[1].encode()) <= 4000
        if small:
            as_text = [(n, v, 'VARCHAR2' if n == 'q_documents' else k) for n, v, k in binds]
            step(label + '_V08_JSONTABLE_VARCHAR_BIND', lambda: count(jt, [('q_documents', documents[1], 'VARCHAR2')]), jt)
            step(label + '_V09_PROJECTION_VARCHAR_DOCS', lambda: count(full, as_text), full)
        else:
            result['steps'].append({'name': label + '_V08_V09', 'status': 'SKIP', 'reason': 'document list over 4000 bytes'})
            print(label + '_V08_V09 SKIP (document list over 4000 bytes)')
        if physical:
            inline = ('(SELECT JSON_ARRAYAGG(d RETURNING CLOB) FROM (SELECT DISTINCT RGLT_DOC_ID AS d FROM ' + physical +
                      ' WHERE RFRN_POST_HASH_VALU=:q_release AND RFRN_DTA_TYPE_VALU=:q_length))')
            no_clob = re.sub(r':q_documents\b', lambda _m: inline, full)
            step(label + '_V10_PROJECTION_NO_CLOB_BIND',
                 lambda: count(no_clob, [b for b in binds if b[0] != 'q_documents']), no_clob)
    step('V11_ARRAYAGG_DUAL', lambda: count(ARRAYAGG_DUAL), ARRAYAGG_DUAL)
    if physical and args.channel == DEFAULT_CHANNEL:
        step('V12_NOTEBOOK_Q1_TEXT', lambda: count(NOTEBOOK_Q1), NOTEBOOK_Q1)
    result['sql_calls'] = sql.calls
    failed = [s['name'] for s in result['steps'] if s.get('status') == 'FAIL']
    result['failed'] = failed
    print('FAILED', len(failed), ' '.join(failed))
    return finish(0)
