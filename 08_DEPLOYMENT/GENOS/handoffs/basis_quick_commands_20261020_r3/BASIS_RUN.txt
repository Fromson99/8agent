#!/usr/bin/env bash
# R3: reuse the installed, sealed R4 or ENTRY_FIX runner from actual TRANSFER.
set -Eeuo pipefail
umask 077
if [[ $# == 0 || "${1:-}" == help || "${1:-}" == --help ]]; then
  cat <<'HELP'
전체 내규 조사 시작 / 중단 후 같은 전체 작업 이어하기:
  bash ./BASIS_RUN.txt all
전체 상태:
  bash ./BASIS_RUN.txt status all
전체 HTML:
  bash ./BASIS_RUN.txt report all
다음에 새 전체 조사:
  bash ./BASIS_RUN.txt all all2
코드/엔진 해시 진단:
  bash ./BASIS_RUN.txt check

기존 일부 조사 명령도 지원:
  bash ./BASIS_RUN.txt new job1 3
  bash ./BASIS_RUN.txt plan job2 5
  bash ./BASIS_RUN.txt run job2
  bash ./BASIS_RUN.txt resume job1
  bash ./BASIS_RUN.txt report job1
  bash ./BASIS_RUN.txt export all

전체 = 선택 세대에서 현재 risk_team 권한으로 접근 가능한 최신 내규 중 본문이 있는 문서.
각 내규를 기존 RAG로 조사하여 대응 외규 연결 후보를 JSON에 저장합니다.
20개씩 순차 실행합니다. 같은 all은 재개이며 새 예산으로 처음부터 다시 돌리지 않습니다.
계획 단계의 BASIS_STRUCTURE 문서는 실패로 기록하고 나머지를 진행합니다.
기존 R2 all 계획·완료 결과를 유지해 재개합니다. 구조 실패가 있으면 최종 partial/종료2입니다.
후보 승인·운영 SQLite 반영은 기존 HTML 검토/이관 절차입니다.
실제 TRANSFER에서 실행하세요. 기존 R4 또는 ENTRY_FIX 원본 패키지가 필요합니다.
HELP
  exit 0
fi
TASK_HERE="$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" && pwd -P)"
TASK_KIT="$TASK_HERE/BASIS_QUICK_R3"
TASK_LOG_DIR="$PWD/basis-quick-r3-logs"
[[ ! -L "$TASK_LOG_DIR" ]] || { echo "심볼릭 링크 로그 경로 거부" >&2; exit 1; }
mkdir -p "$TASK_LOG_DIR"
TASK_LOG="$TASK_LOG_DIR/$(date -u +%Y%m%dT%H%M%SZ)-$$.log"
exec > >(tee -a "$TASK_LOG") 2>&1
trap 'TASK_RC=$?; printf "%s\n" "$TASK_RC" > "$TASK_LOG.exit"; printf "BASIS_QUICK_R3_EXIT=%s LOG=%s\n" "$TASK_RC" "$TASK_LOG"' EXIT
[[ -f "$TASK_KIT/MANIFEST.json" ]] || {
  echo "실제 TRANSFER에서 먼저 ZIP을 푸세요:"
  echo "python3 -m zipfile -e ./GENOS_BASIS_QUICK_20261020_R3.zip ."
  exit 1
}
python3 - "$TASK_KIT" <<'PY'
import hashlib, json, sys
from pathlib import Path
kit = Path(sys.argv[1])
for name, expected in json.loads((kit / "MANIFEST.json").read_text("utf-8"))["files"].items():
    path = kit / name
    if path.is_symlink() or not path.is_file() or hashlib.sha256(path.read_bytes()).hexdigest() != expected:
        raise SystemExit("간편 실행 파일 해시 불일치: " + name)
PY
python3 -u "$TASK_KIT/basis_quick.py" "$@" --kit "$TASK_KIT"
