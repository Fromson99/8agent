#!/usr/bin/env bash
set -Eeuo pipefail
umask 077

# R10: no automatic PDF output cutoff; keep waiting with progress, never resubmit.
# GENOS_PDF_OUTPUT_WAIT_SECONDS is retired and ignored, including old values.
# R8: PDF item counts, current file, elapsed time and 10-second heartbeat.
# R7: apply exact error-display edits with verified backups, then resume.
# R6: check PyMuPDF in the actual python3; recover validated saved PDFs without /run.
# R5: reuse the exact verified r2 backup; preserve interrupted partial archives.
# R4: unbuffered line reader replaces awk input buffering.
# R3: existing PREPROCESSOR settings also serve the same /run PDF conversion.
# All terminal/log lines have KST timestamps; failures name the stage and line.
# Run in the GENOS Codespace with python3, after INSTALL_R3 MODE=apply.
# Builds a DRAFT only. It never changes CURRENT or the code-serving revision.
RUN_LOG="$(mktemp "${TMPDIR:-/tmp}/genos-full-e2e-XXXXXXXX.log")"
exec > >(python3 -u -c '
import sys
from datetime import datetime, timezone, timedelta
kst = timezone(timedelta(hours=9))
for line in sys.stdin:
    print(datetime.now(kst).strftime("[%Y-%m-%d %H:%M:%S KST]"), line.rstrip("\n"), flush=True)
' | tee -a "$RUN_LOG") 2>&1
export PYTHONUNBUFFERED=1
START_SECONDS=$SECONDS
STAGE=preflight
ERROR_LINE=unknown
stage() {
  STAGE="$1"
  printf 'STEP_START stage=%s elapsed_seconds=%s\n' "$STAGE" "$((SECONDS - START_SECONDS))"
}
finish() {
  local code=$?
  trap - ERR EXIT
  if (( code != 0 )); then
    printf 'FAILED stage=%s exit_code=%s line=%s elapsed_seconds=%s\n' "$STAGE" "$code" "$ERROR_LINE" "$((SECONDS - START_SECONDS))"
    printf 'Read the error above. Full terminal log: %s\n' "$RUN_LOG"
    if [[ -n "${GENOS_DATA_ROOT:-}" && -n "${RUN_ID:-}" ]]; then
      printf 'Document details: %s/LOGS/%s/detail.log\n' "$GENOS_DATA_ROOT" "$RUN_ID"
      printf 'Preparation receipts: %s/LOGS/%s/preparation\n' "$GENOS_DATA_ROOT" "$RUN_ID"
    fi
  fi
  printf 'FINISHED status=%s stage=%s elapsed_seconds=%s log=%s\n' "$code" "$STAGE" "$((SECONDS - START_SECONDS))" "$RUN_LOG"
  exit "$code"
}
trap 'ERROR_LINE=$LINENO' ERR

ROOT = Path.cwd().resolve()
def sha(raw):
    return hashlib.sha256(raw).hexdigest()
def target(rel):
    p = Path(rel)
    if p.is_absolute() or '..' in p.parts:
        raise ValueError('Unsafe code path: ' + rel)
    out = ROOT / p
    if any(q.is_symlink() for q in (out, *out.parents)) or not out.is_file():
        raise ValueError('Code file missing or symlink: ' + str(out))
    return out
def atomic(path, raw):
    temp = path.with_name('.error-patch-' + uuid.uuid4().hex)
    try:
        with temp.open('xb') as out:
            out.write(raw)
        if path.exists():
            os.chmod(temp, path.stat().st_mode & 0o777)
        os.replace(temp, path)
    finally:
        temp.unlink(missing_ok=True)

import json, sqlite3, sys
from pathlib import Path
data, receipt_path, generation, release = map(Path, sys.argv[1:5])
scope = json.loads((data/'CATALOG/SCOPES/photo-required-80-hwpx-20260917.json').read_text())['documents'][:61]
shared = set(json.loads(receipt_path.read_text())['shared_document_names'])
if len(scope) != 61 or len(shared) != 9:
    raise SystemExit('Photo 61 or shared scope changed')
with sqlite3.connect('file:' + str(generation/'store.sqlite') + '?mode=ro', uri=True) as db:
    failures = []
    risk_ids, finance_ids = set(), set()
    for did, raw in db.execute('SELECT document_id,json FROM document'):
        meta = json.loads(raw).get('document_metadata') or {}
        teams = {loc.get('team') for loc in [meta, *(meta.get('aliases') or [])] if isinstance(loc, dict)}
        if 'risk_team' in teams: risk_ids.add(did)
        if 'financial_team' in teams: finance_ids.add(did)
    for doc in scope:
        name = doc['name']
        ids = {f['sha256'][:16] for f in doc['files'] if f['role'] == 'body'}
        found = False
        for did in ids:
            row = db.execute('SELECT name,json FROM document WHERE document_id=?', (did,)).fetchone()
            if row is None: continue
            meta = json.loads(row[1]).get('document_metadata') or {}
            title = meta.get('document_title') or row[0].rsplit(' > ', 1)[-1]
            blocks = db.execute('SELECT 1 FROM source_block WHERE document_id=? LIMIT 1', (did,)).fetchone()
            if title == name and did in finance_ids and (name not in shared or did in risk_ids) and blocks:
                found = True
                break
        if not found: failures.append(name)
if failures:
    raise SystemExit('PHOTO61_FAILED: ' + ', '.join(failures))
manifest = json.loads((generation/'manifest.json').read_text())
nav = generation/manifest['navigation']['path']
if not (nav/'reference-resolution-apply.json').is_file():
    raise SystemExit('LLM reference overlay missing')
if not (generation/'glossary/bundle/manifest.json').is_file():
    raise SystemExit('Generation glossary bundle missing')
with sqlite3.connect('file:' + str(nav/'navigation.sqlite') + '?mode=ro', uri=True) as db:
    counts = {table: db.execute('SELECT count(*) FROM '+table).fetchone()[0]
              for table in ('term','edge','occurrence','definition','reference','reference_end')}
if not counts['term'] or not counts['occurrence']:
    raise SystemExit('Term/document link index is empty')
print(json.dumps({'photo61': 'PASS', 'shared_document_types': len(shared),
                  'risk_document_versions': len(risk_ids), 'financial_document_versions': len(finance_ids),
                  'shared_document_versions': len(risk_ids & finance_ids),
                  'index_rows': counts, 'reference_overlay': 'PRESENT'}, ensure_ascii=False), flush=True)
PY
