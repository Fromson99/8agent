sys.path.insert(0, str(root / '02_BACKEND'))
def sha(raw):
    return hashlib.sha256(raw).hexdigest()
def target(rel):
    dest = root / rel
    if any(p.is_symlink() for p in (dest, *dest.parents)):
        raise ValueError('Code symlink refused: ' + rel)
    return dest
def write(dest, raw):
    temp = dest.with_name('.docx-native-' + uuid.uuid4().hex)
    temp.write_bytes(raw)
    if dest.exists():
        os.chmod(temp, dest.stat().st_mode & 0o777)
    os.replace(temp, dest)

    from lawrag.pipeline.genos_stages import offline_data_root
    data_setting = offline_data_root(root)

                saved_new = backup / 'after' / rel

                saved_new.write_bytes(new)
