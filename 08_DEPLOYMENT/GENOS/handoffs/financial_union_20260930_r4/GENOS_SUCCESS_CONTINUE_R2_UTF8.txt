def sha(raw): return hashlib.sha256(raw).hexdigest()
def target(rel):
    p = root / rel
    if any(x.is_symlink() for x in (p, *p.parents)):
        raise ValueError('Code symlink refused: ' + rel)
    return p
def atomic(p, raw):
    temp = p.with_name('.adoption-' + uuid.uuid4().hex)
    temp.write_bytes(raw)
    if p.exists(): os.chmod(temp, p.stat().st_mode & 0o777)
    os.replace(temp, p)

            if row['path'] not in allowed or sha(target(row['path']).read_bytes()) != row['after_sha256']:
