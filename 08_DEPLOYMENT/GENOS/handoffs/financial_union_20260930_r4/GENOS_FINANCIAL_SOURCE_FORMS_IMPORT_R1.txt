import copy
from datetime import datetime, timezone
import fcntl
import hashlib
import json
import os
from pathlib import Path, PurePosixPath
import shutil
import stat
import sys
import uuid
import zipfile

data, archive_path, checkout = map(Path, sys.argv[1:4])
scope_path = data / 'CATALOG/SCOPES/photo-required-80-hwpx-20260917.json'
layout_path = data / 'CATALOG/TEAM_LAYOUT.json'
scope_sha = 'aba1df17dbd2648203cf3ceb9719b784f398822bdaf7a163f566765b7cd3025f'
package_rel = Path('PACKAGES/EXTERNAL/source-forms-e95be5b5f8bc3d36/gen-e95be5b5f8bc3d3698cd2146')
prefix = '01_DATA/' + package_rel.as_posix() + '/'
manifest_sha = '8b3643d60032a2eeb45b18cf8534ac33047972d30c35628e647437843ff8134e'

def digest(raw):
    return hashlib.sha256(raw).hexdigest()

def file_sha(path):
    h = hashlib.sha256()
    with path.open('rb') as source:
        for chunk in iter(lambda: source.read(1024 * 1024), b''):
            h.update(chunk)
    return h.hexdigest()

def safe(path):
    if any(p.is_symlink() for p in (path, *path.parents)):
        raise ValueError('symlink in destination: ' + str(path))
    return path

def install(source, target, expected):
    safe(target)
    if target.exists():
        if file_sha(target) != expected:
            raise ValueError('destination collision: ' + str(target))
        return False
    target.parent.mkdir(parents=True, exist_ok=True)
    temp = target.with_name('.' + target.name + '.tmp-' + uuid.uuid4().hex)
    try:
        with temp.open('xb') as out:
            with source.open('rb') as stream:
                shutil.copyfileobj(stream, out)
        if file_sha(temp) != expected:
            raise ValueError('source changed while copying: ' + str(source))
        try:
            os.link(temp, target)
        except FileExistsError:
            if file_sha(target) != expected:
                raise ValueError('destination changed while copying: ' + str(target))
        return True
    finally:
        temp.unlink(missing_ok=True)

def atomic_json(path, value):
    target = safe(path)
    target.parent.mkdir(parents=True, exist_ok=True)
    temp = target.with_name('.' + target.name + '.tmp-' + uuid.uuid4().hex)
    try:
        temp.write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')
        os.replace(temp, target)
    finally:
        temp.unlink(missing_ok=True)

if file_sha(scope_path) != scope_sha:
    raise ValueError('photo61 scope hash differs')

if len(scope) != 10 or len(expected) != 11:

    manifest_name = prefix + 'manifest.json'

    if digest(manifest_raw) != manifest_sha:
        raise ValueError('source package manifest changed')

    if (manifest.get('schema_version') != 'dataset-package-v2'
            or manifest.get('security_class') != 'restricted'
            or manifest.get('export_policy', {}).get('allowed') is not False):
        raise ValueError('source package security contract differs')
    entries = {prefix + row['path']: row['sha256'] for row in manifest['artifacts']}
    if set(names) != set(entries) | {manifest_name}:
        raise ValueError('source package file list differs')
    for name, expected_sha in entries.items():
        if digest(archive.read(name)) != expected_sha:
            raise ValueError('source package artifact hash differs: ' + name)
    for doc, item in expected:
        if (item['source'], item['security_class'], item['export_allowed']) != ('internal', 'restricted', False):
            raise ValueError('unexpected source authority: ' + doc['name'])
        member = '01_DATA/' + item['path']
        if entries.get(member) != item['sha256']:
            raise ValueError('scope original is not in the source package: ' + doc['name'])
    print('PACKAGE_VERIFIED originals=11 scope_sha256=' + scope_sha, flush=True)

    lock_path = safe(data / '.raw-migration.lock')
    with lock_path.open('a') as lock:
        fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        layout_before_raw = layout_path.read_bytes()

        layout = json.loads(layout_before_raw)
