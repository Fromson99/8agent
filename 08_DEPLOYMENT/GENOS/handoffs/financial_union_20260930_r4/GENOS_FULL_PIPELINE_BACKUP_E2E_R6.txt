#!/usr/bin/env bash
set -Eeuo pipefail
umask 077

# R6: check PyMuPDF in the actual python3; recover validated saved PDFs without /run.
# R5: reuse the exact verified r2 backup; preserve interrupted partial archives.
# R4: unbuffered line reader replaces awk input buffering.
# R3: existing PREPROCESSOR settings also serve the same /run PDF conversion.
# All terminal/log lines have KST timestamps; failures name the stage and line.
# Run in the GENOS Codespace with python3, after INSTALL_R3 MODE=apply.
# Builds a DRAFT only. It never changes CURRENT or the code-serving revision.
RUN_LOG="$(mktemp "${TMPDIR:-/tmp}/genos-full-e2e-XXXXXXXX.log")"
exec > >(python3 -u -c '
import sys
from datetime import datetime, timezone, timedelta
kst = timezone(timedelta(hours=9))
for line in sys.stdin:
    print(datetime.now(kst).strftime("[%Y-%m-%d %H:%M:%S KST]"), line.rstrip("\n"), flush=True)
' | tee -a "$RUN_LOG") 2>&1
export PYTHONUNBUFFERED=1
START_SECONDS=$SECONDS
STAGE=preflight
ERROR_LINE=unknown
stage() {
  STAGE="$1"
  printf 'STEP_START stage=%s elapsed_seconds=%s\n' "$STAGE" "$((SECONDS - START_SECONDS))"
}
finish() {
  local code=$?
  trap - ERR EXIT
  if (( code != 0 )); then
    printf 'FAILED stage=%s exit_code=%s line=%s elapsed_seconds=%s\n' "$STAGE" "$code" "$ERROR_LINE" "$((SECONDS - START_SECONDS))"
    printf 'Read the error above. Full terminal log: %s\n' "$RUN_LOG"
    if [[ -n "${GENOS_DATA_ROOT:-}" && -n "${RUN_ID:-}" ]]; then
      printf 'Document details: %s/LOGS/%s/detail.log\n' "$GENOS_DATA_ROOT" "$RUN_ID"
      printf 'Preparation receipts: %s/LOGS/%s/preparation\n' "$GENOS_DATA_ROOT" "$RUN_ID"
    fi
  fi
  printf 'FINISHED status=%s stage=%s elapsed_seconds=%s log=%s\n' "$code" "$STAGE" "$((SECONDS - START_SECONDS))" "$RUN_LOG"
  exit "$code"
}
trap 'ERROR_LINE=$LINENO' ERR
trap finish EXIT
stage preflight
DEPLOY_ROOT="$(realpath "${DEPLOY_ROOT:-$HOME/DEV}")"
cd "$DEPLOY_ROOT"
export GENOS_DATA_ROOT="$(realpath "${GENOS_DATA_ROOT:-$DEPLOY_ROOT/01_DATA}")"
RUN_ID="${GENOS_RUN_ID:-financial-risk-full-e2e-20260930-r1}"
BASE="${GENOS_BASE_GENERATION:-}"
[[ "$RUN_ID" =~ ^[A-Za-z0-9][A-Za-z0-9_.-]{0,79}$ ]] || { echo 'Invalid GENOS_RUN_ID'; exit 2; }
[[ -n "$BASE" && -f "$BASE/manifest.json" ]] || {
  echo 'Set GENOS_BASE_GENERATION to the exact previous serving generation directory.'; exit 2;
}
BASE="$(realpath "$BASE")"
# The serving generation can be on a separate NFS mount from Codespace 01_DATA.
[[ -d "$BASE" && -f "$BASE/manifest.json" ]] || { echo 'Base generation disappeared'; exit 2; }
command -v python3 >/dev/null || { echo 'python3 missing'; exit 2; }
# pdf_check imports fitz; a missing import was previously reported as no PDF.
stage pdf_validation_dependency
if ! python3 -c 'import fitz' >/dev/null 2>&1; then
  echo 'PyMuPDF import failed in the pipeline python3. Installing the checkout-pinned dependency.'
  python3 -c 'import fitz' || true
  PDF_REQUIREMENT="$(python3 - <<'PY'
import re
from pathlib import Path
matches = re.findall(r'^pymupdf==([^\\s#]+)', Path('requirements.txt').read_text(), re.M | re.I)
if len(matches) != 1:
    raise SystemExit('ERROR: exactly one PyMuPDF pin is required in requirements.txt')
print('pymupdf==' + matches[0])
PY
)"
  if [[ -n "${GENOS_PDF_WHEEL_ROOT:-}" ]]; then
    [[ -d "$GENOS_PDF_WHEEL_ROOT" ]] || { echo 'GENOS_PDF_WHEEL_ROOT is missing'; exit 2; }
    python3 -m pip install --disable-pip-version-check --no-index --find-links "$GENOS_PDF_WHEEL_ROOT" "$PDF_REQUIREMENT"
  else
    # Uses Codespace's configured pip index; no hard-coded external index.
    python3 -m pip install --disable-pip-version-check "$PDF_REQUIREMENT"
  fi
fi
python3 - <<'PY'
import fitz, sys
print('PDF_VALIDATOR_READY python=' + sys.executable + ' PyMuPDF=' + str(fitz.VersionBind), flush=True)
PY
stage preflight
command -v tar >/dev/null || { echo 'tar missing'; exit 2; }
command -v flock >/dev/null || { echo 'flock missing'; exit 2; }
mkdir -p "$GENOS_DATA_ROOT/LOGS/$RUN_ID"
exec 9>"$GENOS_DATA_ROOT/LOGS/$RUN_ID/full-e2e-driver.lock"
flock -n 9 || { echo 'This full E2E run is already active'; exit 2; }
for path in 06_TOOLS/COMMANDS/genos_data_stages.py 06_TOOLS/pipeline.py \
  06_TOOLS/COMMANDS/genos_generation_e2e.py 02_BACKEND/lawrag/pipeline/union_file.py \
  02_BACKEND/lawrag/pipeline/reference_resolution.py; do
  [[ -f "$path" ]] || { echo "Required checkout code missing: $path"; exit 2; }
  sha256sum "$path"
done
printf 'CHECKOUT=%s BASE_GENERATION=%s RUN_ID=%s\n' "$DEPLOY_ROOT" "$BASE" "$RUN_ID"
sha256sum "$BASE/manifest.json"
RECEIPT="$GENOS_DATA_ROOT/CATALOG/TEAM_LAYOUT_HISTORY/financial-external-19-correction-r3/receipt.json"
[[ -f "$RECEIPT" ]] || { echo 'Run INSTALL_R3 MODE=apply before full E2E'; exit 2; }

# Use the existing parser service for conversion unless a separate converter
# endpoint is explicitly configured. Both clients use /run with file_path.
# Explicit converter settings always win; API keys are never printed.
if [[ -z "${GENOS_PDF_CONVERTER_URL:-}" && -z "${GENOS_PDF_CONVERTER_ID:-}" ]]; then
  CONVERTER_FROM_PREPROCESSOR=1
  if [[ -n "${GENOS_PREPROCESSOR_URL:-}" ]]; then
    export GENOS_PDF_CONVERTER_URL="$GENOS_PREPROCESSOR_URL"
  elif [[ -n "${GENOS_PREPROCESSOR_ID:-}" ]]; then
    export GENOS_PDF_CONVERTER_ID="$GENOS_PREPROCESSOR_ID"
  else
    echo 'ERROR: set GENOS_PREPROCESSOR_ID or GENOS_PREPROCESSOR_URL in this terminal environment.'
    exit 2
  fi
  for suffix in REMOTE_ROOT REVISION TIMEOUT API_KEY; do
    converter_key="GENOS_PDF_CONVERTER_$suffix"
    parser_key="GENOS_PREPROCESSOR_$suffix"
    if [[ -z "${!converter_key:-}" && -n "${!parser_key:-}" ]]; then
      export "$converter_key=${!parser_key}"
    fi
  done
  echo 'CONFIG: PDF conversion uses the configured preprocessor service.'
fi
if [[ -z "${GENOS_PDF_CONVERTER_SHARED_ROOT:-}" ]]; then
  if [[ "${CONVERTER_FROM_PREPROCESSOR:-0}" == 1 ]]; then
    export GENOS_PDF_CONVERTER_SHARED_ROOT="${GENOS_PREPROCESSOR_SHARED_ROOT:-}"
  else
    echo 'ERROR: separate converter selected; set its GENOS_PDF_CONVERTER_SHARED_ROOT.'
    exit 2
  fi
fi
for key in GENOS_PDF_CONVERTER_SHARED_ROOT GENOS_PREPROCESSOR_SHARED_ROOT; do
  [[ "${!key:-}" == /* && -d "${!key:-/nonexistent}" && -r "${!key}" && -w "${!key}" && -x "${!key}" ]] || {
    echo "ERROR: $key must be an existing absolute shared directory readable/writable from this terminal."
    exit 2
  }
done
[[ -n "${GENOS_PREPROCESSOR_ID:-${GENOS_PREPROCESSOR_URL:-}}" ]] || {
  echo 'ERROR: set GENOS_PREPROCESSOR_ID or GENOS_PREPROCESSOR_URL.'; exit 2;
}
python3 - <<'PY'

    p = urlsplit(url)
    if (p.scheme not in {'http', 'https'} or not p.hostname or p.username or p.password
            or p.query or p.fragment or p.path not in {'', '/', '/run'}):
        raise SystemExit('ERROR: converter requires a service base URL or /run; /parser is parsing-only. Set GENOS_PDF_CONVERTER_ID for conversion.')
else:
    if not os.environ.get('GENOS_PDF_CONVERTER_ID', '').isdecimal():

stage saved_pdf_validation_and_recovery
# Hold the same RAW lock used by preparation. Only saved artifacts are examined.
python3 - "$GENOS_DATA_ROOT" "$RUN_ID" <<'PY'
import fcntl, hashlib, json, os, sys
from datetime import datetime, timezone
from pathlib import Path
sys.path.insert(0, str(Path.cwd() / '02_BACKEND'))
from lawrag.pipeline import raw_preprocess
from lawrag.pipeline import genos_build
from semantic_runtime.contracts import sha
root, run_id = Path(sys.argv[1]), sys.argv[2]
preparation = root / 'LOGS' / run_id / 'preparation'
lock_path = raw_preprocess.plain_path(root / '.raw-migration.lock')
expected_shared = raw_preprocess.plain_path(os.environ['GENOS_PDF_CONVERTER_SHARED_ROOT'])
expected_url = os.environ.get('GENOS_PDF_CONVERTER_URL', '').strip()
if not expected_url:
    expected_url = 'http://preprocessor-' + os.environ['GENOS_PDF_CONVERTER_ID'] + ':8080/run'
else:
    from urllib.parse import urlsplit
    if urlsplit(expected_url).path in {'', '/'}:
        expected_url = expected_url.rstrip('/') + '/run'
recovered = 0
with lock_path.open('a') as lock:
    fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
    for path in sorted(preparation.glob('*/conversion.json')):
        path = raw_preprocess.plain_path(path)
        prior = path.read_bytes()
        record = json.loads(prior)
        if record.get('status') not in {'FAILED', 'PENDING'}:
            continue
        print('SAVED_JOB_CHECK receipt=' + str(path) + ' http_status=' + str(record.get('http_status'))
              + ' api_code=' + str(record.get('api_code')), flush=True)
        if record.get('transport_error_type') or not isinstance(record.get('http_status'), int):
            raise SystemExit('SAVED_JOB_UNCERTAIN: no completed HTTP response; inspect server job before retry')
        staging = raw_preprocess.plain_path(record['staging'])
        if (staging.parent.resolve() != expected_shared.resolve() or record.get('url') != expected_url
                or record.get('revision', '') != os.environ.get('GENOS_PDF_CONVERTER_REVISION', '')):
            raise SystemExit('SAVED_JOB_BINDING_CHANGED: preserve job and restore its original converter settings')
        request_name = Path(record['request']['file_path']).name
        source = raw_preprocess.plain_path(staging / request_name)
        pdf = raw_preprocess.plain_path(source.with_suffix('.pdf'))
        if not source.is_file() or sha(source) != record.get('input_sha256'):
            raise SystemExit('SAVED_JOB_INPUT_CHANGED: original staged input differs')
        if not pdf.is_file():
            raise SystemExit('SAVED_PDF_STILL_MISSING: ' + str(pdf) + '; no automatic /run retry')
        pdf_sha = sha(pdf)
        try:
            pages = raw_preprocess.pdf_check(pdf)
        except Exception as exc:
            print('SAVED_PDF_VALIDATION_FAILED type=' + type(exc).__name__, flush=True)
            raise
        if sha(pdf) != pdf_sha or sha(source) != record['input_sha256']:
            raise SystemExit('SAVED_JOB_CHANGED_DURING_VALIDATION: preserve files and stop')
        prior_sha = hashlib.sha256(prior).hexdigest()
        preserved = path.with_name('conversion-before-recovery-' + prior_sha + '.json')
        if preserved.exists():
            if preserved.read_bytes() != prior:
                raise SystemExit('Recovery history differs')
        else:
            with preserved.open('xb') as out:
                out.write(prior)
        if path.read_bytes() != prior:
            raise SystemExit('Receipt changed during recovery')
        record.update(status='CONVERTED' if record['http_status'] == 200 and record.get('api_code') == 0
                      else 'PDF_WITH_API_WARNING', pages=pages, pdf_sha256=pdf_sha, pdf_path=str(pdf))
        record['saved_pdf_recovery'] = {'schema_version': 'saved-pdf-recovery-v1',
            'prior_receipt': str(preserved), 'prior_receipt_sha256': prior_sha,
            'validated_at': datetime.now(timezone.utc).isoformat(), 'new_service_requests': 0}
        genos_build.atomic(path, record)
        recovered += 1
        print('SAVED_PDF_RECOVERED pages=' + str(pages) + ' status=' + record['status']
              + ' sha256=' + pdf_sha + ' prior_receipt=' + str(preserved), flush=True)
print('SAVED_PDF_RECOVERY_COMPLETE recovered=' + str(recovered) + ' new_service_requests=0', flush=True)
PY

stage previous_backup_sha256_verification
python3 - "$GENOS_DATA_ROOT" "$BASE" "$ARCHIVE" "$BACKUP_RECEIPT" <<'PY'
import hashlib, json, sys
from pathlib import Path
root, base, archive, receipt = map(Path, sys.argv[1:5])
manifest = hashlib.sha256((base / 'manifest.json').read_bytes()).hexdigest()
if archive.exists() != receipt.exists():
    raise SystemExit('Partial backup exists; inspect it before rerun')
if archive.exists():
    old = json.loads(receipt.read_text())
    import time
    h = hashlib.sha256()
    verified, next_report = 0, 512*1024*1024
    started = time.monotonic()
    print('BACKUP_SHA256_START bytes=' + str(archive.stat().st_size), flush=True)
    with archive.open('rb') as stream:
        for chunk in iter(lambda: stream.read(8*1024*1024), b''):
            h.update(chunk)
            verified += len(chunk)
            if verified >= next_report:
                print('BACKUP_VERIFY_PROGRESS bytes=' + str(verified) + '/' + str(archive.stat().st_size)
                      + ' elapsed_seconds=' + str(round(time.monotonic()-started, 1)), flush=True)
                next_report = verified + 512*1024*1024
    if old.get('archive_sha256') != h.hexdigest() or old.get('base_manifest_sha256') != manifest:
        raise SystemExit('Backup binding differs; use a new GENOS_RUN_ID')
    print('BACKUP_REUSED bytes=' + str(archive.stat().st_size), flush=True)
    raise SystemExit(0)
print('BACKUP_REQUIRED base=' + str(base) + ' manifest_sha256=' + manifest, flush=True)
PY

import hashlib, json, sys
from datetime import datetime, timezone
from pathlib import Path
base, archive, receipt, root = map(Path, sys.argv[1:5])
def sha(path):
    h=hashlib.sha256()
    with path.open('rb') as stream:
        for chunk in iter(lambda: stream.read(8*1024*1024), b''):
            h.update(chunk)
    return h.hexdigest()
control = [root/'CATALOG/TEAM_LAYOUT.json', root/'RAW/union_file/manifest.json', root/'generations/CURRENT.json']
