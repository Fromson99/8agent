#!/usr/bin/env bash
set -Eeuo pipefail
umask 077

# Read-only. Print unresolved source and DOC/DOCX processing failures only.
# Normal DOC alternatives and successful runs are omitted.
RUN_LOG="$(mktemp "${TMPDIR:-/tmp}/genos-financial-failures-only-XXXXXXXX.log")"
exec > >(tee -a "$RUN_LOG") 2>&1
trap 'code=$?; printf "FINISHED status=%s log=%s\n" "$code" "$RUN_LOG"' EXIT
DEPLOY_ROOT="${DEPLOY_ROOT:-/home/genos/DEV}"
GENOS_DATA_ROOT="${GENOS_DATA_ROOT:-$DEPLOY_ROOT/01_DATA}"
PYTHON_BIN="${PYTHON_BIN:-$DEPLOY_ROOT/.venv/bin/python}"
if [[ ! -x "$PYTHON_BIN" ]]; then PYTHON_BIN="$(command -v python3)"; fi
[[ -x "$PYTHON_BIN" ]] || { echo 'Python 3 required'; exit 2; }
"$PYTHON_BIN" - "$(realpath "$GENOS_DATA_ROOT")" "$DEPLOY_ROOT" <<'PY'
import hashlib
import json
import os
from pathlib import Path
import sys

data, checkout = map(Path, sys.argv[1:3])
scope_path = data / 'CATALOG/SCOPES/photo-required-80-hwpx-20260917.json'
expected_scope_sha = 'aba1df17dbd2648203cf3ceb9719b784f398822bdaf7a163f566765b7cd3025f'
sizes = [96768, 85504, 84480, 211968, 238592, 455938, 2802688,
         536576, 450048, 56832, 956416]

def sha(path):
    h = hashlib.sha256()
    with path.open('rb') as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b''):
            h.update(chunk)
    return h.hexdigest()

if sha(scope_path) != expected_scope_sha:
    raise ValueError('photo scope hash differs; stop before diagnosing wrong versions')
docs = json.loads(scope_path.read_text(encoding='utf-8'))['documents'][51:61]
expected = [(d['name'], f, size) for (d, f), size in zip(
    [(d, f) for d in docs for f in d['files']], sizes)]
if len(docs) != 10 or len(expected) != 11:
    raise ValueError('expected ten documents and eleven files')
layout = json.loads((data / 'CATALOG/TEAM_LAYOUT.json').read_text(encoding='utf-8'))['files']
wanted = {f['sha256']: (name, Path(f['path']).name, size) for name, f, size in expected}
found = {key: [] for key in wanted}
for root in (data / 'RAW', checkout / 'TRANSFER', data / 'PACKAGES/EXTERNAL'):
    if not root.is_dir():
        continue
    for directory, dirs, names in os.walk(root):
        dirs[:] = [d for d in dirs if d not in {'.git', 'union_file'}]
        for name in names:
            path = Path(directory) / name
            try:
                if path.is_symlink() or path.stat().st_size not in sizes:
                    continue
                digest = sha(path)
            except OSError:
                continue
            if digest in found and len(found[digest]) < 3:
                found[digest].append(str(path))

source_issues = 0
for name, item, _ in expected:
    digest = item['sha256']
    registered = [path for path, row in layout.items()
                  if path.startswith('RAW/internal/financial_team/')
                  and row.get('sha256') == digest and row.get('scope_document') == name]
    if registered and found[digest]:
        continue
    source_issues += 1
    print('SOURCE_ISSUE ' + json.dumps({
        'document': name, 'file': Path(item['path']).name,
        'reason': 'financial_registration_missing' if found[digest] else 'exact_bytes_not_found_in_scanned_roots',
        'exact_path_sample': found[digest][:1],
    }, ensure_ascii=False), flush=True)

# Keep only the newest recorded outcome for each physical document ID.
latest = {}
result_paths = sorted((data / 'LOGS').glob('*/result-*.json'),
                      key=lambda p: (p.stat().st_mtime_ns, str(p)))
for path in result_paths:
    try:
        result = json.loads(path.read_text(encoding='utf-8'))
    except (OSError, ValueError):
        continue
    for did, row in result.get('documents', {}).items():
        name = str(row.get('name', ''))
        if name.lower().endswith(('.doc', '.docx')):
            latest[did] = (path.parent.name, name, row)
failures = [(did, run, name, row) for did, (run, name, row) in latest.items()
            if row.get('structure') == 'FAILED' or row.get('index') == 'FAILED']
for did, run, name, row in failures[:25]:
    print('DOC_FAILURE ' + json.dumps({'run': run, 'id': did,
        'file': Path(name).name, 'structure': row.get('structure'),
        'index': row.get('index'), 'error': str(row.get('error', ''))[:250]},
        ensure_ascii=False), flush=True)

# Input errors are not result rows; report each distinct DOC path/error once.
input_failures = {}
for path in sorted((data / 'LOGS').glob('*/inventory-*.json'),
                   key=lambda p: (p.stat().st_mtime_ns, str(p))):
    try:
        inventory = json.loads(path.read_text(encoding='utf-8'))
    except (OSError, ValueError):
        continue
    for row in inventory.get('errors', []):
        if not isinstance(row, dict):
            continue
        item = str(row.get('item', ''))
        if Path(item).suffix.lower() in {'.doc', '.docx'}:
            input_failures[item] = (path.parent.name, str(row.get('error', ''))[:250])
for item, (run, error) in list(input_failures.items())[:25]:
    print('DOC_INPUT_FAILURE ' + json.dumps({'run': run,
        'file': Path(item).name, 'error': error}, ensure_ascii=False), flush=True)
print('FAILURE_COUNTS ' + json.dumps({'source_issues': source_issues,
    'doc_structure_or_index_failures': len(failures),
    'doc_input_failures': len(input_failures)}, ensure_ascii=False), flush=True)
print('READ_ONLY_COMPLETE', flush=True)
PY
