def site_name(site):
    inner, _, outer = str(site).partition('<')
    module, _, function = inner.rpartition('.')
    label = SITE_NAMES.get(function) if module == 'oracle_agent_records' else None

class Stop(Exception):
    def __init__(self, code, message):
        super().__init__(message)
        self.code = code


def width(text):
    return sum(2 if unicodedata.east_asian_width(ch) in 'WF' else 1 for ch in text)


def pad(text, size, right=False):
    text = str(text)
    gap = max(0, size - width(text))
    return (' ' * gap + text) if right else (text + ' ' * gap)


def cut(text, size):
    text = str(text)
    if width(text) <= size:
        return text
    out = ''
    for ch in text:
        if width(out + ch) > size - 1:
            break
        out += ch
    return out + '…'


def sec(ms):
    return f'{ms / 1000:,.1f}초'


def pct(part, whole):
    return f'{100 * part / whole:5.1f}%' if whole else '   - '


def bar(part, whole, size=20):
    if not whole:
        return ''
    filled = max(0, min(size, int(round(size * part / whole))))
    return '#' * filled + '.' * (size - filled)


def num(value):
    return value if type(value) in (int, float) and math.isfinite(value) else None


def stamp(value):
    if not isinstance(value, str):
        return None
    try:
        parsed = datetime.fromisoformat(value.replace('Z', '+00:00'))
    except ValueError:
        return None
    return parsed if parsed.tzinfo else parsed.replace(tzinfo=timezone.utc)

def report_root(argument):
    for value in (argument, os.environ.get('GENOS_SHARED_LOG_DIR', '').strip(), DEFAULT_ROOT):
        if value:
            root = Path(value).expanduser()
            if root.is_dir():
                return root
            if value == argument:
                raise Stop(2, f'요청 보고서 폴더가 없습니다: {root}')
    raise Stop(2, '요청 보고서 폴더를 찾지 못했습니다. 코드서빙의 GENOS_SHARED_LOG_DIR 값을 --root 로 주세요.\n'
                  f'  기본 위치 {DEFAULT_ROOT} 없음. 확인: ls -d /nfs-root/*/8-agent/rag-request-reports')


def recent_requests(root):
    today = datetime.now(KST).date()
    rows, unreadable = [], 0
    for back in range(LOOK_BACK_DAYS + 1):
        day = root / (today - timedelta(days=back)).isoformat()
        if not day.is_dir():
            continue
        seen = set()
        for row_path in day.glob('*/*/_qa/*.row'):
            try:
                with row_path.open('rb') as source:
                    raw = source.read(MAX_ROW_BYTES + 1)
                if len(raw) > MAX_ROW_BYTES:
                    raise ValueError
                row = json.loads(raw)
                when = stamp(row['requested_at'])
                if when is None:
                    raise ValueError
                report = row_path.parent.parent / str(row.get('report') or '')
                seen.add(report.name)
                rows.append({'when': when, 'seconds': num(row.get('seconds')), 'team': str(row.get('team') or ''),
                             'component': row_path.parent.parent.name, 'status': str(row.get('status') or ''),
                             'http': row.get('http_status'), 'report': report})
            except (OSError, ValueError, TypeError, KeyError):
                unreadable += 1
        # A request whose summary row was not written still has its JSON report.
        for report in day.glob('*/*/*.json'):
            if report.name in seen:
                continue
            try:
                when = datetime.fromtimestamp(report.stat().st_mtime, timezone.utc)
            except OSError:
                continue
            rows.append({'when': when, 'seconds': None, 'team': report.parent.parent.name,
                         'component': report.parent.name, 'status': '요약 줄 없음(파일 시각)', 'http': None,
                         'report': report})
        if len(rows) >= LIST_LIMIT:
            break
    rows.sort(key=lambda r: r['when'], reverse=True)
    return rows, unreadable


def print_list(root, rows, unreadable, out):
    out(f'요청 보고서 위치: {root}')
    out(f'최근 요청 {min(len(rows), LIST_LIMIT)}건 (최근 {LOOK_BACK_DAYS + 1}일, 새것부터)'
        + (f' · 읽지 못한 요약 줄 {unreadable}건' if unreadable else ''))
    out(pad('번호', 5) + pad('요청 시각(한국)', 21) + pad('서버 처리', 11, True) + '  ' + pad('HTTP', 5)
        + pad('상태', 12) + '팀/구분')
    for index, row in enumerate(rows[:LIST_LIMIT], 1):
        seconds = f'{row["seconds"]:,.1f}초' if row['seconds'] is not None else '-'
        exists = '' if row['report'].is_file() else '  (JSON 아직 없음)'
        out(pad(f'[{index}]', 5) + pad(row['when'].astimezone(KST).strftime('%m-%d %H:%M:%S'), 21)
            + pad(seconds, 11, True) + '  ' + pad(str(row['http'] or '-'), 5) + pad(cut(row['status'], 11), 12)
            + f'{row["team"]}/{row["component"]}{exists}')
    if not rows:
        out('요청이 없습니다. --root 위치와 날짜 폴더를 확인하세요.')
    out(f'특정 요청 보고: {COMMAND} <번호>   ·   여러 요청 합산: {COMMAND} sum')


def pick(root, choice):
    rows, unreadable = recent_requests(root)
    if not rows:
        raise Stop(3, f'최근 {LOOK_BACK_DAYS + 1}일 안의 요청 보고서가 없습니다: {root}')
    index = int(choice) if choice else 1
    if not 1 <= index <= min(len(rows), LIST_LIMIT):
        raise Stop(2, f'번호는 1~{min(len(rows), LIST_LIMIT)} 입니다. 목록: {COMMAND} list')
    row = rows[index - 1]
    if not row['report'].is_file():
        raise Stop(3, f'요청 보고서 JSON이 아직 없습니다(응답 뒤 기록 중이거나 기록 실패): {row["report"]}\n'
                      '  10~20초 뒤 다시 실행하세요. 계속 없으면 같은 이름의 .log 파일을 확인합니다.')
    return row['report']


def load(path):
    try:
        size = path.stat().st_size
    except OSError as exc:
        raise Stop(3, f'보고서를 읽지 못했습니다: {path} ({type(exc).__name__})')
    if size > MAX_REPORT_BYTES:
        raise Stop(3, f'보고서가 너무 큽니다({size:,} bytes): {path}')
    opener = open
    if path.suffix == '.gz':
        import gzip
        opener = gzip.open
    try:
        with opener(path, 'rt', encoding='utf-8') as source:
            report = json.load(source)
    except (OSError, ValueError) as exc:
        raise Stop(3, f'보고서 JSON을 읽지 못했습니다: {path} ({type(exc).__name__})')
    if not isinstance(report, dict) or not isinstance(report.get('service_events'), list):
        raise Stop(3, f'요청 보고서 형식이 아닙니다(service_events 없음): {path}')
    return report, size

def journal_events(journal):
    """Every timeline record of the run journal and any child journals, with time and kind."""
    found = []

    def walk(node, depth):
        if depth > 6:
            return
        if isinstance(node, dict):
            timeline = node.get('timeline')
            if isinstance(timeline, list):
                for item in timeline:
                    event = item.get('event') if isinstance(item, dict) else None
                    if isinstance(event, dict):
                        when = stamp(event.get('created_at'))
                        if when is not None:
                            found.append((when, str(event.get('kind') or '?')))
            for key, value in node.items():
                if key != 'timeline' and isinstance(value, (dict, list)) and key in ('children', 'child', 'journal'):
                    walk(value, depth + 1)
        elif isinstance(node, list):
            for value in node:
                walk(value, depth + 1)

    walk(journal, 0)
    found.sort(key=lambda item: item[0])
    return found


SNAPSHOT_KEYS = ('sql_count_total', 'sql_ms_total', 'agent_ms_total', 'cpu_ms_total')


def snapshot(row):
    """Cumulative counters a MAIN176+ model event carries; None for older reports."""
    if not isinstance(row, dict) or num(row.get('sql_count_total')) is None:
        return None
    return {k: num(row.get(k)) or 0.0 for k in SNAPSHOT_KEYS}


def delta(before, after):
    if before is None or after is None:
        return None
    return {k: max(0.0, after[k] - before[k]) for k in SNAPSHOT_KEYS}


def analyse(report):
    events = []
    for row in report['service_events']:
        if isinstance(row, dict):
            when = stamp(row.get('ts'))
            if when is not None:
                events.append((when, row))
    events.sort(key=lambda item: item[0])
    if not events:
        raise Stop(3, '보고서에 시각이 있는 서비스 기록이 없습니다.')
    first = lambda name: next((row for when, row in events if row.get('event') == name), None)
    started_row = first('request.started')
    start = stamp(started_row['ts']) if started_row else events[0][0]
    returning = [row for when, row in events if row.get('event') in ('request.returning', 'request.finished')]
    total_ms = num(returning[-1].get('duration_ms')) if returning else None
    if total_ms is None:
        total_ms = (events[-1][0] - start).total_seconds() * 1000
    end = start + timedelta(milliseconds=total_ms)

    def offset(when):
        return (when - start).total_seconds() * 1000

    # Model calls: started → first end event of the same attempt.
    calls, open_calls = [], {}
    for when, row in events:
        name, attempt = row.get('event'), row.get('attempt_id')
        if not attempt:
            continue
        if name == 'model.attempt.started':
            open_calls[attempt] = {'start': when, 'end': None, 'stage': row.get('source_stage') or row.get('phase') or '?',
                                   'schema': row.get('schema') or '', 'attempt_no': row.get('attempt_no'),
                                   'status': None, 'prompt_tokens': None, 'completion_tokens': None,
                                   'snap_start': snapshot(row), 'snap_end': None}
            calls.append(open_calls[attempt])
        elif name in MODEL_END and attempt in open_calls:
            call = open_calls[attempt]
            if call['end'] is None or name != 'model.call.finished':
                if call['end'] is None:
                    call['end'] = when
                if name in ('model.attempt.received', 'model.attempt.failed') and call['snap_end'] is None:
                    call['snap_end'] = snapshot(row)
                if name == 'model.attempt.received':
                    call['status'] = row.get('http_status')
                    call['prompt_tokens'] = num(row.get('prompt_tokens'))
                    call['completion_tokens'] = num(row.get('completion_tokens'))
                elif name == 'model.attempt.failed':
                    call['status'] = row.get('error_type') or row.get('failure_kind') or 'failed'
    for call in calls:
        if call['end'] is None:
            call['end'], call['status'] = end, call['status'] or '끝 기록 없음'
    calls = [c for c in calls if c['end'] >= c['start']]
    model_ms = sum((c['end'] - c['start']).total_seconds() * 1000 for c in calls)
    metrics = next((row.get('execution_stats') for when, row in reversed(events)
                    if row.get('event') == 'execution.metrics' and isinstance(row.get('execution_stats'), dict)), None)
    ui_model_ms = num(metrics.get('llm_duration_ms')) if metrics else None

    # Segments: G0, M1, G1, M2, ..., Gn.
    segments, cursor = [], start
    for index, call in enumerate(calls, 1):
        if call['start'] > cursor:
            segments.append({'kind': 'G', 'start': cursor, 'end': call['start'], 'before': calls[index - 2] if index > 1 else None,
                             'after': call, 'number': index - 1})
        segments.append({'kind': 'M', 'start': call['start'], 'end': call['end'], 'call': call, 'number': index})
        cursor = max(cursor, call['end'])
    if end > cursor:
        segments.append({'kind': 'G', 'start': cursor, 'end': end, 'before': calls[-1] if calls else None,
                         'after': None, 'number': len(calls)})
    for segment in segments:
        segment['ms'] = (segment['end'] - segment['start']).total_seconds() * 1000
        segment.update(sql_slow=[0, 0.0], sql_debug=[0, 0.0], commits=[0, 0.0], steps=defaultdict(float),
                       kinds=Counter(), others=Counter())

    def segment_of(when):
        for segment in segments:
            if segment['start'] <= when <= segment['end']:
                return segment
        return None

    sql_summary = next((row for when, row in reversed(events) if row.get('event') == 'oracle.sql.summary'), None)
    end_snap = None
    if sql_summary and num(sql_summary.get('agent_ms')) is not None:
        end_snap = {'sql_count_total': num(sql_summary.get('sql_count')) or 0.0, 'sql_ms_total': num(sql_summary.get('sql_ms')) or 0.0,
                    'agent_ms_total': num(sql_summary.get('agent_ms')) or 0.0, 'cpu_ms_total': num(sql_summary.get('request_cpu_ms')) or 0.0}
    has_snaps = end_snap is not None or any(c['snap_start'] for c in calls)
    zero = {k: 0.0 for k in SNAPSHOT_KEYS} if has_snaps else None
    for segment in segments:
        if segment['kind'] == 'M':
            segment['delta'] = delta(segment['call']['snap_start'], segment['call']['snap_end'])
        else:
            before = segment['before']['snap_end'] if segment['before'] else zero
            after = segment['after']['snap_start'] if segment['after'] else end_snap
            segment['delta'] = delta(before, after)
    log_ms = sum(num(row.get('previous_log_ms')) or 0.0 for when, row in events)
    slow, debug_sql, failed_sql = [], [], []
    commits = [0, 0.0, 0.0]
    steps = {}
    last_label = '요청 시작'
    for when, row in events:
        name = row.get('event') or '?'
        segment = segment_of(when)
        if name == 'oracle.sql.slow':
            ms = num(row.get('duration_ms')) or 0.0
            slow.append({'when': when, 'ms': ms, 'family': str(row.get('sql_family') or '?'),
                         'write': bool(row.get('write')), 'segment': segment, 'context': last_label,
                         'site': str(row.get('sql_site') or '')})
            if segment:
                segment['sql_slow'][0] += 1; segment['sql_slow'][1] += ms
        elif name == 'oracle.sql':
            status = row.get('status')
            ms = num(row.get('duration_ms'))
            if status == 'ACKNOWLEDGED' and ms is not None:
                debug_sql.append({'when': when, 'ms': ms, 'family': str(row.get('sql_family') or '?'), 'segment': segment})
                if segment:
                    segment['sql_debug'][0] += 1; segment['sql_debug'][1] += ms
            elif status not in ('STARTED', 'ACKNOWLEDGED'):
                failed_sql.append({'when': when, 'status': str(status), 'family': str(row.get('sql_family') or '?'),
                                   'ms': ms or 0.0, 'site': str(row.get('sql_site') or ''), 'segment': segment,
                                   'error': str(row.get('error') or ''), 'context': last_label})
        elif name == 'journal.transaction.timings':
            ms = sum(num(row.get(k)) or 0.0 for k in ('begin_ms', 'body_ms', 'commit_ms'))
            commits[0] += 1; commits[1] += ms; commits[2] += num(row.get('commit_ms')) or 0.0
            if segment:
                segment['commits'][0] += 1; segment['commits'][1] += ms
        elif name in ('step.finished', 'step.failed'):
            label = str(row.get('step') or '?')
            ms = num(row.get('duration_ms')) or 0.0
            item = steps.setdefault(label, [0, 0.0, 0.0, 0.0, 0])
            item[0] += 1; item[1] += ms; item[2] = max(item[2], ms); item[3] += num(row.get('thread_cpu_ms')) or 0.0
            item[4] += name == 'step.failed'
            if segment:
                segment['steps'][label] += ms
        elif not name.startswith(OTHER_SKIP) and name not in ('request.started', 'request.returning', 'request.finished'):
            if segment:
                segment['others'][name] += 1
        if name == 'step.started':
            last_label = 'step ' + str(row.get('step') or '?') + ' 시작'
        elif name in ('step.finished', 'step.failed'):
            last_label = 'step ' + str(row.get('step') or '?') + ' 끝'
        elif name not in ('oracle.sql', 'oracle.sql.slow', 'journal.transaction.timings'):
            last_label = name

    journal = journal_events(report.get('journal'))
    for when, kind in journal:
        segment = segment_of(when)
        if segment and kind not in QUIET_KINDS:
            segment['kinds'][kind] += 1

    # Longest stretches with no record at all, outside model calls.
    step_word = {'step.started': '시작', 'step.finished': '끝', 'step.failed': '실패'}
    marks = [(when, f"{row.get('step')} {step_word[row.get('event')]}" if row.get('event') in step_word
              else str(row.get('event'))) for when, row in events]
    marks += [(when, 'journal:' + kind) for when, kind in journal]
    for item in slow:
        marks.append((item['when'] - timedelta(milliseconds=item['ms']), 'SQL ' + item['family'][:8] + ' 시작'))
    marks.sort(key=lambda item: item[0])
    quiet = []
    for (a, label_a), (b, label_b) in zip(marks, marks[1:]):
        if b <= a or a < start or b > end + timedelta(seconds=1):
            continue
        inside = sum(max(0.0, (min(b, c['end']) - max(a, c['start'])).total_seconds()) for c in calls)
        span = (b - a).total_seconds() - inside
        if span * 1000 >= 500:
            quiet.append({'start': a, 'ms': span * 1000, 'before': label_a, 'after': label_b})
    quiet.sort(key=lambda item: -item['ms'])

    others = Counter()
    for when, row in events:
        name = row.get('event') or ''
        if row.get('level') in ('WARNING', 'ERROR', 'CRITICAL') and name != 'oracle.sql':
            others[name] += 1
    release = next((row for when, row in events if row.get('event') == 'oracle.release.selected'), None)
    return {'start': start, 'end': end, 'total_ms': total_ms, 'calls': calls, 'model_ms': model_ms,
            'ui_model_ms': ui_model_ms, 'metrics': metrics, 'segments': segments, 'sql_summary': sql_summary,
            'slow': slow, 'debug_sql': debug_sql, 'failed_sql': failed_sql, 'commits': commits, 'steps': steps,
            'quiet': quiet, 'warnings': others, 'release': release, 'journal_count': len(journal), 'log_ms': log_ms,
            'has_snaps': has_snaps,
            'event_count': len(events), 'offset': offset}

def site_text(site, size=60):
    """Raw '[보낸 함수<요청 쪽 함수]' first and never cut (we map it back to SQL at home), then the Korean meaning."""
    site = str(site or '')
    if not site or site in ('unknown', '?'):
        return '위치 기록 없음'
    raw = '[' + site + ']'
    label = site_label(site)
    room = size - width(raw) - 2
    return raw + ('  ' + cut(label, room) if label != '-' and room >= 8 else '')


def site_label(site):
    site = str(site or '')
    inner, _, outer = site.partition('<')
    return ' · '.join(x for x in (OUTER.get(outer.split('.')[0]) if outer else OUTER.get(inner.split('.')[0]), INNER.get(inner)) if x) or '-'


# ---------------------------------------------------------------- R3: 사진 단위 출력
PAGE_LINES = 40
PAGE_WIDTH = 100
PAGES = 5


class Pages:
    """Lines grouped into photo-sized pages; a page never exceeds PAGE_LINES (overflow is cut with a count)."""
    def __init__(self, stamp_text):
        self.stamp, self.pages = stamp_text, []

    def page(self, title):
        self.pages.append([f'━━━━ 사진 {len(self.pages) + 1}/{PAGES} · {title} · {self.stamp} ━━━━'])

    def line(self, text=''):
        self.pages[-1].append(cut(text, PAGE_WIDTH))

    def wrap(self, text, first='  · ', rest='    '):
        """A long sentence on up to three lines instead of cutting it."""
        line, prefix, count = '', first, 0
        for word in str(text).split(' '):
            if line and width(prefix + line + ' ' + word) > PAGE_WIDTH:
                self.pages[-1].append(prefix + line); prefix, line, count = rest, word, count + 1
                if count == 2:
                    break
            else:
                line = (line + ' ' + word) if line else word
        self.pages[-1].append(cut(prefix + line, PAGE_WIDTH))

    def lines(self):
        for number, page in enumerate(self.pages, 1):
            if len(page) > PAGE_LINES:
                extra = len(page) - PAGE_LINES + 1
                page[:] = page[:PAGE_LINES - 1] + [f'  … ({extra}줄 생략, 전체는 결과 파일 참고)']
            yield number, page


def merged_sites(summary):
    rows = summary.get('sql_sites') if summary and isinstance(summary.get('sql_sites'), list) else []
    out = []
    for row in rows:
        if isinstance(row, dict):
            out.append({'site': str(row.get('site') or ''), 'count': int(num(row.get('count')) or 0),
                        'ms': float(num(row.get('ms')) or 0.0), 'writes': int(num(row.get('writes')) or 0),
                        'bytes': num(row.get('bytes')), 'cpu_ms': num(row.get('cpu_ms')), 'rows': num(row.get('rows'))})
    out.sort(key=lambda r: -r['ms'])
    return out

           + (f' · 채널 {cut(str(release.get("query_channel")), 14)} · 캐시 {"켜짐" if release.get("serving_data_cache") else "꺼짐"}' if release else ''))
    P.line('')
    P.line(f'[A] 전체 {sec(total)}')

    if summary:
        P.line('    ' + pad('Oracle SQL', 12) + pad(sec(sql_ms), 10, True) + ' ' + pct(sql_ms, total) + '  ' + bar(sql_ms, total)

        P.line(f'    SQL: 가장 느린 1회 {sec(num(summary.get("sql_slowest_ms")) or 0)} · 실패 {int(num(summary.get("sql_failed_count")) or 0)} · '
               f'쓰기 {int(num(summary.get("sql_write_count")) or 0):,} · 캐시 적중 {int(num(summary.get("sql_cache_hits")) or 0):,}/'
               f'미적중 {int(num(summary.get("sql_cache_misses")) or 0):,}')
        P.line(f'    하나씩 기록된 SQL: 느린 {len(slow)}개 {sec(slow_ms)} (SQL 시간의 {pct(slow_ms, sql_ms).strip()}) · 실패 {len(failed)}개'
               + (f' · DEBUG {len(a["debug_sql"]):,}개' if a['debug_sql'] else ''))

            verdicts.append(f'짧은 SQL을 아주 많이 보냄({sql_n:,}회 × 평균 {sql_ms / sql_n:,.0f}ms)'
                            + (f'. 가장 많이 부른 곳 {busiest["count"]:,}회 → {site_label(busiest["site"])}' if busiest else ''))
        verdicts.append(f'Oracle SQL이 전체의 {share * 100:.0f}%, 모델 {model_ms / total * 100:.0f}%, 그 외 {other_ms / total * 100:.0f}%')

            verdicts.append(f'원장 저장(대화·조사 기록) SQL {agent_ms / sql_ms * 100:.0f}% · 원문·검색 조회 {100 - agent_ms / sql_ms * 100:.0f}% (상위 위치 기준)')
    biggest = max((s for s in a['segments'] if s['kind'] == 'G'), key=lambda s: s['ms'], default=None)
    if biggest and total:
        verdicts.append(f'가장 긴 모델 밖 구간 G{biggest["number"]} {sec(biggest["ms"])} ('
                        + (biggest['before']['stage'] if biggest['before'] else '요청 시작') + ' → '
                        + (biggest['after']['stage'] if biggest['after'] else '응답 반환') + ')  (사진 4)')
    for v in verdicts or ['판정할 SQL 기록 없음']:
        P.wrap(v)
    P.line('')
    P.line('[C] 시간 많이 쓴 SQL 위치 상위 5 (요청 전체 합계, 자세히는 사진 2)')
    for rank, row in enumerate(sites[:5], 1):
        P.line(f'  {rank}) ' + pad(sec(row['ms']), 9, True) + ' ' + pct(row['ms'], sql_ms) + pad(f'{row["count"]:,}회', 9, True)
               + '  ' + f'평균 {row["ms"] / row["count"] if row["count"] else 0:,.0f}ms')
        P.line('       ' + site_text(row['site'], 92))
    if not sites:
        tops = families(summary.get('sql_top')) if summary else []
        for family, count, ms in tops:
            P.line('  · ' + pad(family, 14) + pad(f'{count:,}회', 9, True) + pad(sec(ms), 10, True) + '  ' + cut(family_name(family), 40))
        if not tops:
            P.line('  위치 기록 없음(MAIN176 이전 코드)')
    if summary and not slow and not a['debug_sql']:
        P.line('')
        P.line('  ※ 하나씩 기록된 느린 SQL이 없습니다. 더 자세히: GENOS_ORACLE_SQL_SLOW_MS=200 후 재시작·같은 질문 (GUIDE)')

    P.page('SQL 호출 위치별 합계')
    if sites:
        P.line(f'위치 {int(num(summary.get("sql_site_count")) or len(sites))}곳 중 시간 상위 {min(12, len(sites))} '
               '(위치 = 이 조회를 보낸 코드 [보낸 함수<요청 쪽 함수], 합계는 요청 전체)')
        P.line('  순위   합계   SQL중   횟수     평균     최대 기록   쓰기')
        slow_by_site = defaultdict(lambda: [0, 0.0])
        for item in slow:
            row = slow_by_site[item['site']]; row[0] += 1; row[1] = max(row[1], item['ms'])
        for rank, row in enumerate(sites[:12], 1):
            worst = slow_by_site.get(row['site'])
            P.line(f'  {rank:>2}) ' + pad(sec(row['ms']), 8, True) + ' ' + pct(row['ms'], sql_ms) + pad(f'{row["count"]:,}회', 9, True)
                   + pad(f'{row["ms"] / row["count"] if row["count"] else 0:,.0f}ms', 9, True)
                   + pad(f'{sec(worst[1])}({worst[0]})' if worst else '-', 12, True) + pad(str(row['writes']), 6, True))
            P.line('       ' + site_text(row['site'], 92))
        covered = sum(r['ms'] for r in sites[:12])
        P.line(f'  상위 {min(12, len(sites))}곳 합계 {sec(covered)} = SQL 시간의 {pct(covered, sql_ms).strip()}  (최대 기록 = 느린 SQL 줄 중 가장 긴 1회(개수))')
    else:
        P.line('호출 위치 기록 없음 — 현장 코드가 MAIN176 이전입니다(사진 3·4의 해시로 확인).')
    tops = families(summary.get('sql_top')) if summary else []
    if tops:
        P.line('SQL 문장 종류(해시 앞 12자) 상위 3: ' + ' · '.join(f'{f} {n:,}회 {sec(ms)}' for f, n, ms in tops))
    calls = summary.get('agent_calls') if summary and isinstance(summary.get('agent_calls'), list) else []
    if calls:
        P.line('원장 계층: ' + cut(' · '.join(f"{AGENT_CALL_NAMES.get(c.get('name'), c.get('name'))} {int(num(c.get('count')) or 0):,}회 "
                                          f"{sec(num(c.get('ms')) or 0)}" for c in calls), 88))

    # 사진 3 — 실패·느린 SQL 하나하나
    P.page('실패·느린 SQL 하나하나')
    if failed:
        P.line(f'실패 SQL {len(failed)}개 (READ_FAILED=읽기 실패, ReadTimeout=ORDS 응답 기다리다 시간 초과)')
        for item in sorted(failed, key=lambda i: -i['ms'])[:5]:
            P.line(f'  ✕ +{offset(item["when"]) / 1000:,.1f}s ' + pad(sec(item['ms']), 8, True) + f'  {item["status"]}'
                   + (f'({cut(item["error"], 30)})' if item['error'] else '') + f'  해시 {item["family"][:12]}  {seg_name(item["segment"])}')
            P.line('       ' + site_text(item['site'], 92))
        P.line('')
    if slow:
        threshold = min(i['ms'] for i in slow)
        P.line(f'느린 SQL {len(slow)}개 합계 {sec(slow_ms)} (기록 기준 GENOS_ORACLE_SQL_SLOW_MS, 이번 기록 최소 {threshold:,.0f}ms) — 느린 순')
        shown = 15 if not failed else 10
        for rank, item in enumerate(sorted(slow, key=lambda i: -i['ms'])[:shown], 1):
            P.line(f'  {rank:>2}) +{offset(item["when"]) / 1000:,.1f}s ' + pad(sec(item['ms']), 8, True)
                   + ('  쓰기' if item['write'] else '  읽기') + f'  해시 {item["family"][:12]}  {seg_name(item["segment"])}'
                   + '  직전: ' + cut(item['context'], 34))
            P.line('       ' + site_text(item['site'], 92))
        if len(slow) > shown:
            P.line(f'  … 나머지 {len(slow) - shown}개 {sec(sum(i["ms"] for i in sorted(slow, key=lambda i: -i["ms"])[shown:]))} (사진 4 묶음에 포함)')
    elif not failed:
        P.line('하나씩 기록된 느린·실패 SQL이 없습니다.')
        P.line('  지금 기준(기본 1초)보다 짧은 SQL이 여러 번 반복된 것입니다 → 사진 2의 "횟수"가 큰 위치가 원인입니다.')
        P.line('  SQL마다 보려면: 코드서빙 환경변수 GENOS_ORACLE_SQL_SLOW_MS=200 (또는 100) → 재시작 → 같은 질문 → 이 명령 다시.')

    # 사진 4 — 구간별·묶음
    P.page('구간별 SQL과 느린 SQL 묶음')
    gaps = [s for s in a['segments'] if s['kind'] == 'G']
    def gap_sql(s):
        d = s.get('delta')
        return d['sql_ms_total'] if d else s['sql_slow'][1] + s['sql_debug'][1]
    P.line('모델 호출 사이 구간(G) — SQL 시간 많은 순 상위 8   (+시작후 / 구간 길이 / 구간 안 SQL)')
    for s in sorted(gaps, key=lambda s: -gap_sql(s))[:8]:
        d = s.get('delta')
        sql_part = (f'SQL {int(d["sql_count_total"]):,}회 {sec(d["sql_ms_total"])} · 원장 {sec(d["agent_ms_total"])}' if d
                    else f'느린 SQL {s["sql_slow"][0]}개 {sec(s["sql_slow"][1])}')
        P.line(f'  G{s["number"]:<3}' + pad(f'+{offset(s["start"]) / 1000:,.1f}s', 9, True) + pad(sec(s['ms']), 9, True) + '  ' + sql_part
               + (f' · 느린 {s["sql_slow"][0]}개' if d and s['sql_slow'][0] else ''))
        P.line('        ' + cut((s['before']['stage'] if s['before'] else '요청 시작') + ' → '
                                + (s['after']['stage'] if s['after'] else '응답 반환'), 88))
    P.line('')
    grouped = defaultdict(lambda: [0, 0.0, 0.0])
    for item in slow:
        row = grouped[(item['family'][:12], item['site'])]; row[0] += 1; row[1] += item['ms']; row[2] = max(row[2], item['ms'])
    if grouped:
        P.line('느린 SQL을 같은 문장(해시)·같은 위치로 묶음 — 합계 상위 6')
        for (family, site), (count, ms, worst) in sorted(grouped.items(), key=lambda kv: -kv[1][1])[:6]:
            P.line(f'  해시 {family} ' + pad(f'{count}회', 5, True) + pad(sec(ms), 9, True) + f'  최대 {sec(worst)}  '
                   + cut(family_name(family), 26))
            P.line('       ' + site_text(site, 92))
    debug = a['debug_sql']
    if debug:
        dgroup = defaultdict(lambda: [0, 0.0, 0.0])
        for item in debug:
            row = dgroup[item['family'][:12]]; row[0] += 1; row[1] += item['ms']; row[2] = max(row[2], item['ms'])
        P.line(f'DEBUG로 기록된 SQL {len(debug):,}개 — 문장 종류별 합계 상위 5')
        for family, (count, ms, worst) in sorted(dgroup.items(), key=lambda kv: -kv[1][1])[:5]:
            P.line(f'  해시 {family} ' + pad(f'{count:,}회', 8, True) + pad(sec(ms), 9, True) + f'  최대 {worst:,.0f}ms  ' + cut(family_name(family), 30))
    steps = sorted(a['steps'].items(), key=lambda item: -(item[1][1] - item[1][3]))[:4]
    if steps:
        P.line('기다린 시간(합계-CPU) 많은 단계 상위 4: ' + cut(' · '.join(f'{cut(n, 26)} {sec(max(ms - cpu, 0))}'
                                                               for n, (c, ms, w, cpu, f) in steps), 60))
    P.line('')
    P.line('해시·[함수 이름]으로 받는 쪽에서 SQL 문장을 찾습니다(질문·답변·원문·SQL 본문 미출력).')
    render_breakdown(P, summary, sites, sql_ms, sql_n)
    return P


def render_breakdown(P, summary, sites, sql_ms, sql_n):
    """사진 5 — DB 권한 없이 GENOS 쪽에서 SQL 시간을 나눈다(MAIN186 측정 필드)."""
    P.page('SQL 시간 쪼개기 (ORDS 왕복·보낸 크기·CPU)')
    probe = summary.get('sql_probe') if summary and isinstance(summary.get('sql_probe'), dict) else None
    buckets = summary.get('sql_size_buckets') if summary and isinstance(summary.get('sql_size_buckets'), list) else []
    if not probe and not buckets:
        P.line('이 요청 보고서에는 SQL 쪼개기 측정이 없습니다 — 현장 코드가 MAIN186 이전입니다.')
        P.line('MAIN186 적용 후 같은 질문을 다시 하고 이 명령을 실행하면 이 사진이 채워집니다.')
        return
    base = None
    if probe and int(num(probe.get('count')) or 0):
        n = int(num(probe.get('count')) or 0); base = (num(probe.get('ms')) or 0.0) / n
        P.line(f'[1] ORDS 빈 왕복 (SELECT 1 FROM DUAL, SQL {int(num(probe.get("every")) or 0):,}회마다 1번 · DB 일 거의 없음)')
        P.line(f'    {n}회 · 평균 {base:,.1f}ms · 최소 {num(probe.get("min_ms")) or 0:,.1f}ms · 최대 {num(probe.get("max_ms")) or 0:,.1f}ms'
               + (f' · 실패 {int(num(probe.get("failed")) or 0)}' if num(probe.get('failed')) else ''))
        P.line('    = 네트워크 + ORDS + 연결 처리의 기본 비용. SQL 한 번은 최소 이만큼 걸립니다.')
    else:
        P.line('[1] ORDS 빈 왕복 측정 없음 (GENOS_ORACLE_SQL_PROBE_EVERY=0 이거나 SQL이 너무 적음)')
    thread_cpu, process_cpu = num(summary.get('sql_thread_cpu_ms')), num(summary.get('sql_process_cpu_ms'))
    if thread_cpu is not None and sql_ms:
        P.line(f'[2] SQL 도중 CPU: 요청 스레드 {sec(thread_cpu)} = SQL {sec(sql_ms)}의 {pct(thread_cpu, sql_ms).strip()}'
               + (f' · 프로세스 전체 {sec(process_cpu)}' if process_cpu is not None else ''))
        P.line('    스레드 CPU = JSON 만들고 읽기, 나머지는 기다림(네트워크·ORDS·DB)')
        P.line('    프로세스 전체가 스레드보다 훨씬 크면 그동안 다른 스레드가 CPU를 쓴 것')
    P.line(f'[3] 보낸 크기별 (읽기·쓰기 따로, 평균 ms - 빈 왕복 = 그 크기에서 더 걸린 시간)')
    P.line('    종류  크기         횟수      합계     평균   빈 왕복 초과   평균 보낸 크기')
    for row in buckets:
        if not isinstance(row, dict):
            continue
        n = int(num(row.get('count')) or 0); ms = num(row.get('ms')) or 0.0; size = num(row.get('bytes')) or 0
        avg = ms / n if n else 0.0
        P.line('    ' + pad('쓰기' if row.get('write') else '읽기', 6) + pad(str(row.get('size')), 10)
               + pad(f'{n:,}회', 9, True) + pad(sec(ms), 9, True) + pad(f'{avg:,.0f}ms', 9, True)
               + pad(f'{avg - base:+,.0f}ms' if base is not None else '-', 14, True) + pad(f'{size / n / 1024 if n else 0:,.1f}KB', 15, True))
    P.line('[4] 위치별 (시간 상위 8) — 평균 보낸 크기·CPU와 빈 왕복 대비 초과')
    P.line('    합계      횟수     평균   빈 왕복 초과  보낸 크기/회  CPU/회   받은 행/회')
    for row in sites[:8]:
        n = row['count'] or 1; avg = row['ms'] / n
        size = row.get('bytes'); cpu = row.get('cpu_ms'); rows = row.get('rows')
        P.line('  ' + pad(sec(row['ms']), 8, True) + pad(f'{row["count"]:,}회', 9, True) + pad(f'{avg:,.0f}ms', 8, True)
               + pad(f'{avg - base:+,.0f}ms' if base is not None else '-', 13, True)
               + pad(f'{size / n / 1024:,.1f}KB' if size is not None else '-', 13, True)
               + pad(f'{cpu / n:,.1f}ms' if cpu is not None else '-', 9, True)
               + pad(f'{rows / n:,.1f}' if rows is not None else '-', 11, True))
        P.line('       ' + site_text(row['site'], 92))
    P.line('[5] 판정')
    verdicts = []
    if base is not None and sql_n and sql_ms:
        avg_all = sql_ms / sql_n
        share = min(1.0, base * sql_n / sql_ms)
        verdicts.append(f'SQL 평균 {avg_all:,.0f}ms 중 빈 왕복 {base:,.0f}ms → SQL 시간의 약 {share * 100:.0f}%({sec(base * sql_n)})는 '
                        '왕복 자체(네트워크·ORDS)' + (' — 횟수를 줄이는 것이 가장 큼' if share >= 0.5 else ''))
    writes = {str(r.get('size')): r for r in buckets if isinstance(r, dict) and r.get('write') and num(r.get('count'))}
    small = writes.get('<4KB')
    if small and base is not None:
        avg_small = (num(small.get('ms')) or 0) / num(small.get('count'))
        verdicts.append(f'작은 쓰기(<4KB)도 평균 {avg_small:,.0f}ms = 빈 왕복보다 {avg_small - base:,.0f}ms 더 → '
                        + ('DB 쓰기·커밋 처리 자체가 느림(크기 탓 아님)' if avg_small - base >= max(30, base) else '쓰기 처리 비용은 작음'))
    sized = [r for r in buckets if isinstance(r, dict) and r.get('write') and num(r.get('count'))]
    if len(sized) >= 2:
        first, last = sized[0], sized[-1]
        a1 = num(first['ms']) / num(first['count']); a2 = num(last['ms']) / num(last['count'])
        b1 = num(first['bytes']) / num(first['count']); b2 = num(last['bytes']) / num(last['count'])
        if b2 > b1:
            verdicts.append(f'큰 쓰기({last["size"]}, 평균 {b2 / 1024:,.1f}KB) {a2:,.0f}ms vs 작은 쓰기({b1 / 1024:,.1f}KB) {a1:,.0f}ms → '
                            + ('크기가 커질수록 크게 느려짐. 큰 쓰기는 대부분 원장 커밋이라, 보낸 크기 탓인지 커밋 처리 탓인지는 '
                               '[4]의 커밋 위치 평균 크기와 같이 봐야 함' if a2 >= a1 * 2 and a2 - a1 >= 30 else '크기 영향은 작음'))
    if thread_cpu is not None and sql_ms:
        verdicts.append(f'SQL 시간 중 이쪽 CPU {thread_cpu / sql_ms * 100:.0f}% → '
                        + ('JSON 만들기·읽기 같은 GENOS 쪽 계산이 큼' if thread_cpu >= sql_ms * 0.3 else '대부분 응답 기다림(GENOS 계산 아님)'))
    for v in verdicts or ['판정할 측정 없음']:
        P.wrap(v)


def render_sum(root, rows, out, limit=10):
    """Several recent requests at once: when it is not clear which question was slow."""
    loaded, skipped = [], 0
    for row in rows[:limit]:
        if not row['report'].is_file():
            skipped += 1; continue
        try:
            report, size = load(row['report'])
            loaded.append((row, report, analyse(report)))
        except Stop:
            skipped += 1
    P = Pages(datetime.now(KST).strftime('%m-%d %H:%M'))
    P.page('최근 요청 합산')
    P.pages[-1][0] = P.pages[-1][0].replace(f'/{PAGES}', '/1')
    P.line(f'GENOS Oracle 병목 보고 R4 ({VERSION}) · 위치 {cut(str(root), 60)}')
    P.line(f'최근 요청 {len(loaded)}건 합산' + (f' (읽지 못함 {skipped})' if skipped else '') + ' — 번호는 list와 같음')
    P.line('  번호 요청시각        전체     모델     SQL      SQL횟수   느린   실패  가장 느린 1회')
    merged = defaultdict(lambda: [0, 0.0, 0])
    for index, (row, report, a) in enumerate(loaded, 1):
        s = a['sql_summary'] or {}
        sql_ms = num(s.get('sql_ms')) or 0.0
        P.line(f'  [{rows.index(row) + 1:>2}] {row["when"].astimezone(KST):%m-%d %H:%M:%S} ' + pad(sec(a['total_ms']), 8, True)
               + pad(sec(a['model_ms']), 8, True) + pad(sec(sql_ms), 9, True) + pad(f'{int(num(s.get("sql_count")) or 0):,}', 9, True)
               + pad(str(len(a['slow'])), 6, True) + pad(str(len(a['failed_sql'])), 6, True) + pad(sec(num(s.get('sql_slowest_ms')) or 0), 10, True))
        for site in merged_sites(s):
            m = merged[site['site']]; m[0] += site['count']; m[1] += site['ms']; m[2] += 1
    whole = sum(v[1] for v in merged.values())
    P.line('')
    P.line('여러 요청에 걸쳐 시간 많이 쓴 SQL 위치 상위 10 (각 요청 상위 12곳 합)')
    for rank, (site, (count, ms, requests)) in enumerate(sorted(merged.items(), key=lambda kv: -kv[1][1])[:10], 1):
        P.line(f'  {rank:>2}) ' + pad(sec(ms), 9, True) + ' ' + pct(ms, whole) + pad(f'{count:,}회', 10, True) + pad(f'{requests}건', 6, True)
               + pad(f'평균 {ms / count if count else 0:,.0f}ms', 12, True))
        P.line('       ' + site_text(site, 92))
    if not merged:
        P.line('  위치 기록 없음(요약 줄 없음 또는 MAIN176 이전 코드)')
    P.line(f'자세히: {COMMAND} <번호>  (느린 요청 1건을 사진 4장으로)')
    for number, page in P.lines():
        for line in page:
            out(line)


def main(argv):
    root_arg, rest, only = None, [], None
    items = list(argv)
    while items:
        item = items.pop(0)
        if item == '--root':
            if not items:
                raise Stop(2, '--root 다음에 폴더를 주세요.')
            root_arg = items.pop(0)
        elif item == '--page':
            if not items or not items[0].isdigit() or not 1 <= int(items[0]) <= PAGES:
                raise Stop(2, f'--page 다음에 1~{PAGES} 숫자를 주세요.')
            only = int(items.pop(0))
        elif item in ('-h', '--help', 'help'):
            print(__doc__)
            return 0
        else:
            rest.append(item)

    def out(text=''):
        print(text, flush=True)

    if rest and rest[0] == 'list':
        root = report_root(root_arg)
        rows, unreadable = recent_requests(root)
        print_list(root, rows, unreadable, out)
        return 0
    if rest and rest[0] == 'sum':
        root = report_root(root_arg)
        rows, unreadable = recent_requests(root)
        if not rows:
            raise Stop(3, f'최근 {LOOK_BACK_DAYS + 1}일 안의 요청 보고서가 없습니다: {root}')
        render_sum(root, rows, out)
        return 0
    if rest and (rest[0].endswith('.json') or rest[0].endswith('.json.gz') or '/' in rest[0]):
        path = Path(rest[0]).expanduser()
        if not path.is_file():
            raise Stop(2, f'파일이 없습니다: {path}')
    else:
        if rest and not rest[0].isdigit():
            raise Stop(2, f'알 수 없는 인자: {rest[0]}  (list / sum / 번호 / 보고서.json 경로)')
        path = pick(report_root(root_arg), rest[0] if rest else None)
    report, size = load(path)
    pages = render_r3(path, size, report, analyse(report))
    for number, page in pages.lines():
        if only and number != only:
            continue
        for line in page:
            out(line)
        out('')
    return 0


if __name__ == '__main__':
    try:
        sys.exit(main(sys.argv[1:]))
    except Stop as stop:
        print('중단: ' + str(stop), file=sys.stderr)
        sys.exit(stop.code)
