#!/usr/bin/env bash
set -euo pipefail
LATEST_PACKAGE="$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" && pwd -P)"
LATEST_LOG="./latest-internal-$(date +%Y%m%dT%H%M%S)-$$.log"
exec > >(tee -a "$LATEST_LOG") 2>&1
trap 'rc=$?; echo "LATEST_INTERNAL EXIT=$rc LOG=$LATEST_LOG"; exit "$rc"' EXIT
printf 'TRANSFER=%s\nPACKAGE=%s\n' "$PWD" "$LATEST_PACKAGE"
python3 "$LATEST_PACKAGE/apply.py" --root "${GENOS_DEV_ROOT:-$HOME/DEV}" "$@"
