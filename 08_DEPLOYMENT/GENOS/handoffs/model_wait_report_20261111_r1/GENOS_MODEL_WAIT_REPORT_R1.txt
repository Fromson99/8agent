def width(text):
    return sum(2 if unicodedata.east_asian_width(ch) in 'WF' else 1 for ch in text)


def pad(text, size, right=False):
    text = str(text)
    gap = max(0, size - width(text))
    return (' ' * gap + text) if right else (text + ' ' * gap)


def cut(text, size):
    text = str(text)
    if width(text) <= size:
        return text
    out = ''
    for ch in text:
        if width(out + ch) > size - 1:
            break
        out += ch
    return out + '…'


def num(value):
    return value if type(value) in (int, float) and math.isfinite(value) else None


def whole(value):
    return f'{value:,}' if num(value) is not None else '-'


def stamp(value):
    if not isinstance(value, str):
        return None
    try:
        parsed = datetime.fromisoformat(value.replace('Z', '+00:00'))
    except ValueError:
        return None
    return parsed if parsed.tzinfo else parsed.replace(tzinfo=timezone.utc)


def clock(value):
    return value.astimezone(KST).strftime('%H:%M:%S') if value else '-'


def stage_label(stage):

    for prefix, label in STAGES:

    return text or '단계 기록 없음'

def report_root(argument):
    for value in (argument, os.environ.get('GENOS_SHARED_LOG_DIR', '').strip(), DEFAULT_ROOT):
        if value:
            root = Path(value).expanduser()
            if root.is_dir():
                return root
            if value == argument:
                raise Stop(2, f'요청 보고서 폴더가 없습니다: {root}')
    raise Stop(2, '요청 보고서 폴더를 찾지 못했습니다. 코드서빙의 GENOS_SHARED_LOG_DIR 값을 --root 로 주세요.\n'
                  f'  기본 위치 {DEFAULT_ROOT} 없음. 확인: ls -d /nfs-root/*/8-agent/rag-request-reports')


def read_row(row_path):
    with row_path.open('rb') as source:
        raw = source.read(MAX_ROW_BYTES + 1)
    if len(raw) > MAX_ROW_BYTES:
        raise ValueError
    row = json.loads(raw)
    when = stamp(row['requested_at'])
    if when is None:
        raise ValueError
    return row, when, row_path.parent.parent / str(row.get('report') or '')


def recent_requests(root):
    today = datetime.now(KST).date()
    rows, unreadable = [], 0
    for back in range(LOOK_BACK_DAYS + 1):
        day = root / (today - timedelta(days=back)).isoformat()
        if not day.is_dir():
            continue
        seen = set()
        for row_path in day.glob('*/*/_qa/*.row'):
            try:
                row, when, report = read_row(row_path)
                seen.add(report.name)
                rows.append({'when': when, 'seconds': num(row.get('seconds')), 'team': str(row.get('team') or ''),
                             'component': row_path.parent.parent.name, 'status': str(row.get('status') or ''),
                             'http': row.get('http_status'), 'report': report})
            except (OSError, ValueError, TypeError, KeyError):
                unreadable += 1
        for report in day.glob('*/*/*.json'):
            if report.name in seen:
                continue
            try:
                when = datetime.fromtimestamp(report.stat().st_mtime, timezone.utc)
            except OSError:
                continue
            rows.append({'when': when, 'seconds': None, 'team': report.parent.parent.name,
                         'component': report.parent.name, 'status': '요약 줄 없음(파일 시각)', 'http': None,
                         'report': report})
        if len(rows) >= LIST_LIMIT:
            break
    rows.sort(key=lambda r: r['when'], reverse=True)
    return rows, unreadable


def print_list(root, rows, unreadable, out):
    out(f'요청 보고서 위치: {root}')
    out(f'최근 요청 {min(len(rows), LIST_LIMIT)}건 (최근 {LOOK_BACK_DAYS + 1}일, 새것부터)'
        + (f' · 읽지 못한 요약 줄 {unreadable}건' if unreadable else ''))
    out(pad('번호', 5) + pad('요청 시각(한국)', 21) + pad('서버 처리', 11, True) + '  ' + pad('HTTP', 5)
        + pad('상태', 12) + '팀/구분')
    for index, row in enumerate(rows[:LIST_LIMIT], 1):
        seconds = f'{row["seconds"]:,.1f}초' if row['seconds'] is not None else '-'
        exists = '' if row['report'].is_file() else '  (JSON 아직 없음)'
        out(pad(f'[{index}]', 5) + pad(row['when'].astimezone(KST).strftime('%m-%d %H:%M:%S'), 21)
            + pad(seconds, 11, True) + '  ' + pad(str(row['http'] or '-'), 5) + pad(cut(row['status'], 11), 12)
            + f'{row["team"]}/{row["component"]}{exists}')
    if not rows:
        out('요청이 없습니다. --root 위치와 날짜 폴더를 확인하세요.')
    out(f'특정 요청 보고: {COMMAND} <번호>')


def pick(root, choice):
    rows, _ = recent_requests(root)
    if not rows:
        raise Stop(3, f'최근 {LOOK_BACK_DAYS + 1}일 안의 요청 보고서가 없습니다: {root}\n'
                      '  코드서빙에 LOG=on 또는 GENOS_SHARED_LOG_DIR 이 켜져 있는지 확인하세요.')
    index = int(choice) if choice else 1
    if not 1 <= index <= min(len(rows), LIST_LIMIT):
        raise Stop(2, f'번호는 1~{min(len(rows), LIST_LIMIT)} 입니다. 목록: {COMMAND} list')
    row = rows[index - 1]
    if not row['report'].is_file():
        raise Stop(3, f'요청 보고서 JSON이 아직 없습니다(응답 뒤 기록 중이거나 기록 실패): {row["report"]}\n'
                      '  10~20초 뒤 다시 실행하세요.')
    return row['report']


def load(path, limit=MAX_REPORT_BYTES):
    try:
        size = path.stat().st_size
    except OSError as exc:
        raise Stop(3, f'보고서를 읽지 못했습니다: {path} ({type(exc).__name__})')
    if size > limit:
        raise Stop(3, f'보고서가 너무 큽니다({size:,} bytes): {path}')
    opener = open
    if path.suffix == '.gz':
        import gzip
        opener = gzip.open
    try:
        with opener(path, 'rt', encoding='utf-8') as source:
            report = json.load(source)
    except (OSError, ValueError) as exc:
        raise Stop(3, f'보고서 JSON을 읽지 못했습니다: {path} ({type(exc).__name__})')
    if not isinstance(report, dict) or not isinstance(report.get('service_events'), list):
        raise Stop(3, f'요청 보고서 형식이 아닙니다(service_events 없음): {path}')
    return report

def collect_calls(report):
    """Physical sends in the on-screen order (model.call.started), joined by attempt_id."""
    info, calls, by_id = {}, [], {}
    for row in report['service_events']:
        if not isinstance(row, dict) or not row.get('attempt_id'):
            continue
        name, attempt = row.get('event'), row['attempt_id']
        slot = info.setdefault(attempt, {})
        if name == 'model.attempt.started':
            slot['started'] = row
        elif name == 'model.attempt.received':
            slot['received'] = row
        elif name == 'model.attempt.failed':
            slot['failed'] = row
        elif name in ('model_http.finished', 'model_http.failed'):
            slot['http_ms'] = num(row.get('duration_ms'))
        elif name == 'model.call.started':
            call = {'number': len(calls) + 1, 'attempt': attempt, 'start': stamp(row.get('ts')),
                    'purpose': row.get('purpose'), 'ms': None, 'end': None}
            calls.append(call)
            by_id[attempt] = call
        elif name == 'model.call.finished' and attempt in by_id:
            by_id[attempt]['ms'] = num(row.get('duration_ms'))
            by_id[attempt]['end'] = stamp(row.get('ts'))
    for call in calls:
        slot = info.get(call['attempt'], {})
        started, received, failed = slot.get('started') or {}, slot.get('received') or {}, slot.get('failed') or {}
        call.update(stage=started.get('source_stage') or started.get('phase') or received.get('source_stage'),
                    schema=started.get('schema'), attempt_no=started.get('attempt_no'),
                    request_bytes=num(started.get('request_bytes')), system_chars=num(started.get('input_system_chars')),
                    user_chars=num(started.get('input_user_chars')), schema_chars=num(started.get('response_format_chars')),
                    stream=started.get('stream_requested') is True, thinking=started.get('thinking_requested'),
                    prompt=num(received.get('prompt_tokens')), output=num(received.get('completion_tokens')),
                    reasoning=num(received.get('reasoning_tokens')), content_chars=num(received.get('response_content_chars')),
                    finish=received.get('finish_reason'), http_ms=slot.get('http_ms'),
                    failed=bool(failed), failure=failed.get('error_code') or failed.get('failure_kind') or failed.get('error_type'),
                    model=started.get('model'))
        if call['end'] is None and call['start'] is not None and call['ms'] is not None:
            call['end'] = call['start'] + timedelta(milliseconds=call['ms'])
    return calls


def journal_attempts(journal):
    found = {}
    def walk(node):
        if not isinstance(node, dict):
            return
        for entry in node.get('timeline') or []:
            for created in (entry.get('created_records') or []) if isinstance(entry, dict) else []:
                if isinstance(created, dict) and created.get('table') == 'model_attempt':
                    record = created.get('record') or {}
                    found[record.get('attempt_id')] = record
        for record in ((node.get('state') or {}).get('model_attempt') or []):
            if isinstance(record, dict):
                found[record.get('attempt_id')] = record
        for child in node.get('children') or []:
            walk(child.get('journal') if isinstance(child, dict) else None)
    walk(journal)
    return found


def response_shape(record):
    """What the model decided, as enum values and item counts only; never its text."""
    response = record.get('response') if isinstance(record, dict) else None
    body = response.get('body_json') if isinstance(response, dict) else None
    extras = []
    if isinstance(body, dict):
        for owner, values in (('', body), ('usage.', body.get('usage'))):
            for key, value in (values.items() if isinstance(values, dict) else ()):
                if TIME_KEY.search(str(key)) and num(value) is not None:
                    extras.append(f'{owner}{key}={value}')
    parts = []
    try:
        content = body['choices'][0]['message']['content']
        decoded = json.loads(content)
    except (KeyError, IndexError, TypeError, ValueError):
        decoded = None
    if isinstance(decoded, dict):
        for key, value in decoded.items():
            if isinstance(value, str) and ENUM_TEXT.fullmatch(value):
                parts.append(f'{key}={value}')
            elif isinstance(value, list):
                parts.append(f'{key} {len(value)}개')
    return parts[:6], extras[:6]


def windows_of_day(day):
    rows = []
    for row_path in day.glob('*/*/_qa/*.row'):
        try:
            row, when, report = read_row(row_path)
        except (OSError, ValueError, TypeError, KeyError):
            continue
        rows.append((when, num(row.get('seconds')), report))
    return rows


def other_calls(path, report, calls):
    """Model calls of other requests overlapping this one; None when no neighbours can be listed."""
    starts = [c['start'] for c in calls if c['start']]
    ends = [c['end'] for c in calls if c['end']]
    if not starts or not ends:
        return None, 0
    low, high = min(starts), max(ends)
    day = path.parent.parent.parent
    if not re.fullmatch(r'\d{4}-\d{2}-\d{2}', day.name):
        return None, 0
    rows, listed = [], False
    try:
        center = datetime.fromisoformat(day.name).date()
    except ValueError:
        return None, 0
    for back in (-1, 0, 1):
        folder = day.parent / (center + timedelta(days=back)).isoformat()
        if folder.is_dir():
            found = windows_of_day(folder)
            listed = listed or bool(found)
            rows += found
    if not listed:
        return None, 0
    picked, skipped = [], 0
    for when, seconds, candidate in rows:
        if candidate.name == path.name:
            continue
        end = when + timedelta(seconds=seconds) if seconds is not None else high
        if when <= high and end >= low:
            picked.append(candidate)
    others = []
    for candidate in picked[:MAX_OVERLAP_REPORTS]:
        try:
            others += collect_calls(load(candidate, MAX_OVERLAP_BYTES))
        except Stop:
            skipped += 1
    return others, skipped + max(0, len(picked) - MAX_OVERLAP_REPORTS)


def overlaps(call, pool):
    """Calls sharing at least a second with this one, and those already running when it was sent."""
    if not call['start'] or not call['end']:
        return None, None
    shared, ahead = 0, 0
    for other in pool:
        if other is call or not other['start'] or not other['end']:
            continue
        span = (min(call['end'], other['end']) - max(call['start'], other['start'])).total_seconds()
        if span >= 1.0:
            shared += 1
            if other['start'] <= call['start']:
                ahead += 1
    return shared, ahead


def baseline(calls, others):
    """Fastest observed output rate: the time an unqueued call of this size would need is at most tokens/rate."""
    best = None
    for origin, group in (('이 질문', calls), ('겹친 다른 질문', others or [])):
        for call in group:
            if call['ms'] and call['output'] and call['output'] >= MIN_BASE_TOKENS and call['ms'] >= MIN_BASE_MS:
                rate = call['output'] / (call['ms'] / 1000)
                if best is None or rate > best['rate']:
                    best = {'rate': rate, 'origin': origin, 'call': call}
    return best


def judge(call, base):
    if call['failed']:
        return '실패', None
    if call['ms'] is None:
        return '끝 기록 없음', None
    if call['ms'] < MIN_JUDGE_MS:
        return '짧음', None
    if call['output'] is None:
        return '토큰 기록 없음', None
    if base is None:
        return '기준 없음', None
    expected = call['output'] / base['rate'] * 1000
    excess = call['ms'] - expected
    if excess < max(5000, 0.4 * call['ms']):
        return '출력량 비례', excess
    if call['shared']:
        return '대기 의심', excess
    if call['prompt'] and base['call']['prompt'] and call['prompt'] >= 1.5 * base['call']['prompt']:
        return '입력 큼?', excess
    return '대기 의심*', excess

class Pages:
    def __init__(self, stamp_text):
        self.stamp, self.pages = stamp_text, []

    def page(self, title):
        self.pages.append([f'━━━━ 사진 {len(self.pages) + 1}/{PAGES} · {title} · {self.stamp} ━━━━'])

    def line(self, text=''):
        self.pages[-1].append(cut(text, PAGE_WIDTH))

    def wrap(self, text, first='  · ', rest='    '):
        line, prefix, count = '', first, 0
        for word in str(text).split(' '):
            if line and width(prefix + line + ' ' + word) > PAGE_WIDTH:
                self.pages[-1].append(prefix + line)
                prefix, line, count = rest, word, count + 1
                if count == 2:
                    break
            else:
                line = (line + ' ' + word) if line else word
        self.pages[-1].append(cut(prefix + line, PAGE_WIDTH))

    def lines(self):
        for number, page in enumerate(self.pages, 1):
            if len(page) > PAGE_LINES:
                extra = len(page) - PAGE_LINES + 1
                page[:] = page[:PAGE_LINES - 1] + [f'  … ({extra}줄 생략, 전체는 결과 파일 참고)']
            yield number, page


def secs(ms):
    return f'{ms / 1000:.2f}' if ms is not None else '-'


def render(path, report, slow_ms):
    calls = collect_calls(report)
    events = report['service_events']
    requested = stamp(report.get('requested_at'))
    pages = Pages((requested.astimezone(KST).strftime('%m-%d %H:%M:%S') if requested else '시각 없음') + ' · ' + path.stem[:8])
    metrics = next((row.get('execution_stats') for row in reversed(events)
                    if isinstance(row, dict) and row.get('event') == 'execution.metrics'), None)
    metrics = metrics if isinstance(metrics, dict) else {}
    server_ms = num(metrics.get('request_duration_ms'))
    model_ms = sum(c['ms'] for c in calls if c['ms'] is not None)
    others, skipped = other_calls(path, report, calls)
    known_neighbours = others is not None
    pool = calls + (others or [])
    for call in calls:
        call['shared'], call['ahead'] = overlaps(call, pool) if known_neighbours else (None, None)
    base = baseline(calls, others)
    attempts = journal_attempts(report.get('journal'))
    first_ts = next((stamp(row.get('ts')) for row in events if isinstance(row, dict) and stamp(row.get('ts'))), None)

    pages.page('모델 호출별 시간')
    outside = f' · 모델 밖 {(server_ms - model_ms) / 1000:,.2f}초' if server_ms is not None else ''
    pages.line(f'서버 처리 {secs(server_ms)}초 · 모델 호출 합계 {secs(model_ms)}초{outside} · 호출 {len(calls)}회')
    model_name = next((c['model'] for c in calls if c['model']), '-')
    stream_text = '일부 스트리밍' if any(c['stream'] for c in calls) else '스트리밍 아님(첫 글자 시각 기록 없음)'
    pages.line(f'모델 {cut(model_name, 44)} · {stream_text}')
    if base:
        where = f'{base["origin"]} #{base["call"]["number"]}'
        pages.line(f'기준 속도: 출력 {base["rate"]:,.1f}토큰/초 ({where}, 출력 {whole(base["call"]["output"])}토큰) - 이 질문·겹친 질문에서 가장 빠른 값')
    else:
        pages.line('기준 속도: 계산 불가(출력 200토큰 이상 호출의 토큰 기록 없음)')
    if not calls:
        pages.line('이 요청에는 모델 호출 기록이 없습니다.')
    pages.line()
    pages.line(pad('번호', 4, True) + ' ' + pad('화면', 10) + pad('실제 단계', 24) + pad('시간(초)', 9, True) + pad('앞간격', 7, True)
               + pad('입력토큰', 9, True) + pad('출력', 7, True) + pad('토큰/초', 8, True) + pad('동시', 5, True) + '  판정')
    previous_end = first_ts
    for call in calls:
        gap = '-'
        if previous_end and call['start']:
            gap = f'{(call["start"] - previous_end).total_seconds():.1f}'
        previous_end = call['end'] or previous_end
        rate = f'{call["output"] / (call["ms"] / 1000):.1f}' if call['output'] and call['ms'] else '-'
        call['verdict'], call['excess'] = judge(call, base)
        shared = '-' if call['shared'] is None else str(call['shared'])
        pages.line(pad(call['number'], 4, True) + ' ' + pad(SCREEN.get(call['purpose'], str(call['purpose'])), 10)
                   + pad(cut(stage_label(call['stage']), 23), 24) + pad(secs(call['ms']), 9, True) + pad(gap, 7, True)
                   + pad(whole(call['prompt']), 9, True) + pad(whole(call['output']), 7, True) + pad(rate, 8, True)
                   + pad(shared, 5, True) + '  ' + call['verdict'])
    pages.line()
    screen = metrics.get('calls')
    if isinstance(screen, list) and [r.get('duration_ms') for r in screen if isinstance(r, dict)] != [c['ms'] for c in calls]:
        pages.line('주의: 화면 표의 호출 시간과 순서가 이 표와 다릅니다. 번호 대조에 주의하세요.')
    pages.line('요약')
    long_calls = [c for c in calls if c['ms'] is not None and c['ms'] >= slow_ms]
    fits = [c for c in long_calls if c['verdict'] == '출력량 비례']
    waits = [c for c in long_calls if c['verdict'] in ('대기 의심', '대기 의심*')]
    inputs = [c for c in long_calls if c['verdict'] == '입력 큼?']
    label = lambda group: ', '.join(f'#{c["number"]}' for c in group)
    pages.wrap(f'{slow_ms / 1000:.0f}초 이상 호출 {len(long_calls)}개: '
               + ('; '.join(f'#{c["number"]} {stage_label(c["stage"])} {secs(c["ms"])}초' for c in long_calls[:4]) or '없음')
               + (' 외' if len(long_calls) > 4 else ''))
    if fits:
        pages.wrap(f'출력량에 비례(실제 생성 시간에 가까움): {label(fits)}')
    if waits:
        pages.wrap(f'출력량보다 오래 걸림(대기 의심): {label(waits)}')
    if inputs:
        pages.wrap(f'출력량보다 오래 걸렸고 입력이 큼(입력 처리 가능): {label(inputs)}')
    if any(c['verdict'] == '대기 의심*' for c in calls):
        pages.wrap('대기 의심* = 같은 시각 우리 서비스의 다른 호출은 없음. 같은 모델을 쓰는 다른 앱의 대기, 모델 서버 부하일 수 있습니다.')
    pages.wrap('화면의 "모델처리"는 화면 분류표에 없는 단계 전체입니다. 위 "실제 단계"가 그 호출의 내용입니다.')
    if base:
        pages.wrap(f'전 호출이 같이 밀렸으면 모두 "출력량 비례"로 나옵니다. 기준 {base["rate"]:,.0f}토큰/초가 평소 모델 속도보다 낮으면 전체 대기를 의심하세요.')

    pages.page(f'느린 호출 상세 ({slow_ms / 1000:.0f}초 이상)')
    if not long_calls:
        pages.line('해당 호출이 없습니다. --slow 초 값을 낮춰 다시 실행하세요.')
    offset_base = requested or first_ts
    shown = sorted(sorted(long_calls, key=lambda c: -c['ms'])[:MAX_DETAIL], key=lambda c: c['number'])
    for call in shown:
        repeat = f' · {call["attempt_no"]}번째 전송' if isinstance(call['attempt_no'], int) and call['attempt_no'] > 1 else ''
        pages.line(f'#{call["number"]} {stage_label(call["stage"])} ({cut(call["stage"] or "-", 40)}) {secs(call["ms"])}초{repeat}')
        http = f' · HTTP {secs(call["http_ms"])}초' if call['http_ms'] is not None else ''
        offset = f'{(call["start"] - offset_base).total_seconds():.1f}' if call['start'] and offset_base else '-'
        pages.line(f'   시작 {clock(call["start"])} (요청 후 {offset}초){http} · 요청 크기 {whole(call["request_bytes"])}바이트')
        pages.line(f'   입력 글자: 시스템 {whole(call["system_chars"])} · 사용자 {whole(call["user_chars"])} · 응답형식 {whole(call["schema_chars"])}'
                   f' · 입력토큰 {whole(call["prompt"])}')
        thinking = {True: '요청 켬', False: '요청 끔'}.get(call['thinking'], '기록 없음')
        pages.line(f'   출력: {whole(call["output"])}토큰 · 응답 {whole(call["content_chars"])}자 · 종료 {call["finish"] or "-"}'
                   f' · 생각(thinking) {thinking}, 토큰 {whole(call["reasoning"])}')
        parts, extras = response_shape(attempts.get(call['attempt']))
        if call['failed']:
            pages.line(f'   실패: {cut(call["failure"] or "-", 70)}')
        pages.line('   모델이 고른 것: ' + (', '.join(parts) if parts else '기록 없음'))
        if extras:
            pages.line('   응답의 시간 값: ' + ', '.join(extras))
        if call['shared'] is None:
            together = '같은 시각 다른 호출: 확인 불가(요약 줄 없음)'
        else:
            together = f'같은 시각 다른 호출 {call["shared"]}건(이 호출보다 먼저 보낸 것 {call["ahead"]}건)'
        if call['excess'] is not None and base:
            expected = call['output'] / base['rate']
            together += f' · 기준 속도면 {expected:.1f}초, 남는 시간 {call["excess"] / 1000:.1f}초'
        pages.line('   ' + together)
    if len(long_calls) > MAX_DETAIL:
        pages.line(f'가장 긴 {MAX_DETAIL}개만 표시했습니다. 나머지 {len(long_calls) - MAX_DETAIL}개는 사진 1 표를 보세요.')
    pages.line()
    pages.line('이 보고의 한계')
    pages.wrap('모델 서버 안의 대기열 시간은 로그에 없습니다. 판정은 출력 토큰·시간으로 추정한 값입니다.')
    note = '같은 모델을 쓰는 다른 앱과 아직 끝나지 않은 질문의 호출은 "동시"에 나오지 않습니다.'
    if skipped:
        note += f' 겹친 질문 보고서 {skipped}건은 크기·개수 한도로 읽지 않았습니다.'
    pages.wrap(note)
    return pages


def main(argv):
    root_arg, rest, only, slow = None, [], None, 20.0
    items = list(argv)
    while items:
        item = items.pop(0)
        if item == '--root':
            if not items:
                raise Stop(2, '--root 다음에 폴더를 주세요.')
            root_arg = items.pop(0)
        elif item == '--page':
            if not items or not items[0].isdigit() or not 1 <= int(items[0]) <= PAGES:
                raise Stop(2, f'--page 다음에 1~{PAGES} 숫자를 주세요.')
            only = int(items.pop(0))
        elif item == '--slow':
            try:
                slow = float(items.pop(0))
            except (IndexError, ValueError):
                raise Stop(2, '--slow 다음에 초 숫자를 주세요.')
        elif item in ('-h', '--help', 'help'):
            print(__doc__)
            return 0
        else:
            rest.append(item)

    def out(text=''):
        print(text, flush=True)

    if rest and rest[0] == 'list':
        root = report_root(root_arg)
        rows, unreadable = recent_requests(root)
        print_list(root, rows, unreadable, out)
        return 0
    if rest and (rest[0].endswith('.json') or rest[0].endswith('.json.gz') or '/' in rest[0]):
        path = Path(rest[0]).expanduser()
        if not path.is_file():
            raise Stop(2, f'파일이 없습니다: {path}')
    else:
        if rest and not rest[0].isdigit():
            raise Stop(2, f'알 수 없는 인자: {rest[0]}  (list / 번호 / 보고서.json 경로)')
        path = pick(report_root(root_arg), rest[0] if rest else None)
    report = load(path)
    for number, page in render(path, report, slow * 1000).lines():
        if only and number != only:
            continue
        for line in page:
            out(line)
        out('')
    return 0


if __name__ == '__main__':
    try:
        sys.exit(main(sys.argv[1:]))
    except Stop as stop:
        print('중단: ' + str(stop), file=sys.stderr)
        sys.exit(stop.code)
