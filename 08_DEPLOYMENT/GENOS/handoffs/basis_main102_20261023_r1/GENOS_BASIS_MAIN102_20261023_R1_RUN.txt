#!/usr/bin/env bash
set -Eeuo pipefail
umask 077
TASK_HERE="$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" && pwd -P)"
TASK_KIT="$TASK_HERE"
if [[ ! -f "$TASK_KIT/PACKAGE.json" ]]; then TASK_KIT="$TASK_HERE/GENOS_BASIS_MAIN102_20261023_R1"; fi
TASK_LOGDIR="$PWD/basis-main102-logs"
[[ ! -L "$TASK_LOGDIR" ]] || { echo "로그 경로 심볼릭 링크 거부" >&2; exit 1; }
mkdir -p "$TASK_LOGDIR"
TASK_LOG="$TASK_LOGDIR/$(date -u +%Y%m%dT%H%M%SZ)-$$.log"
exec > >(tee -a "$TASK_LOG") 2>&1
trap 'TASK_RC=$?; printf "%s\n" "$TASK_RC" > "$TASK_LOG.exit"; printf "BASIS_MAIN102_EXIT=%s LOG=%s\n" "$TASK_RC" "$TASK_LOG"' EXIT
TASK_ACTION="${1:-check}"
TASK_JOB="${2:-main102-links-20261023-r1}"
TASK_ROOT="${GENOS_BASIS_ROOT:-$HOME/DEV}"
[[ $# -le 2 ]] || { echo "사용법: bash 실행파일.txt check|apply|plan|start|resume|status|report|export|jobs|paths|rollback [작업이름]"; exit 1; }
python3 - "$TASK_KIT" <<'PY'
import hashlib,json,sys
from pathlib import Path
p=Path(sys.argv[1])
for n,h in {'PACKAGE.json': '2c5dbd1d3fff1c4917ce0c16bd7fd0b38a2f402c638f39410292727f34c02a88', 'installer.py': 'e24d51509b4dcb63ac1a9cd0c562046bec9d72499e936030f3606760717387cb'}.items():
    f=p/n
    if f.is_symlink() or not f.is_file() or hashlib.sha256(f.read_bytes()).hexdigest()!=h:
        raise SystemExit('패키지가 없거나 해시가 다릅니다. 먼저 ZIP을 지정 폴더에 풀어 주세요: '+str(f))
PY
case "$TASK_ACTION" in
  check|apply|rollback|install-status)
    TASK_INSTALL_ACTION="$TASK_ACTION"
    [[ "$TASK_ACTION" != install-status ]] || TASK_INSTALL_ACTION=status
    python3 -B "$TASK_KIT/installer.py" "$TASK_INSTALL_ACTION" --root "$TASK_ROOT"
    if [[ "$TASK_ACTION" == apply ]]; then python3 -B "$TASK_ROOT/06_TOOLS/pipeline.py" basis-work check; fi
    ;;
  plan|start|resume|status|report|export|jobs|paths)
    python3 -B "$TASK_KIT/installer.py" check --root "$TASK_ROOT"
    TASK_COMMAND="$TASK_ACTION"
    [[ "$TASK_ACTION" != plan ]] || TASK_COMMAND=plan-all
    [[ "$TASK_ACTION" != start ]] || TASK_COMMAND=all
    if [[ "$TASK_ACTION" == jobs || "$TASK_ACTION" == paths ]]; then
      python3 -B "$TASK_ROOT/06_TOOLS/pipeline.py" basis-work "$TASK_COMMAND"
    else
      python3 -B "$TASK_ROOT/06_TOOLS/pipeline.py" basis-work "$TASK_COMMAND" "$TASK_JOB"
    fi
    ;;
  *) echo "알 수 없는 명령: $TASK_ACTION" >&2; exit 1 ;;
esac
