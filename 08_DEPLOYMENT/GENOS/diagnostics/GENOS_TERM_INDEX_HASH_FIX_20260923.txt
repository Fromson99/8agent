#!/usr/bin/env bash
# GENOS term index generation/hash mismatch 수정 (2026-09-23)
# 실행: bash TRANSFER/GENOS_TERM_INDEX_HASH_FIX_20260923.txt
# NFS generation, 원본 데이터, 인증키, 실행 중인 코드서빙은 수정하지 않습니다.
# 확인된 원인: ~/DEV term_index.py는 source manifest 해시에서 navigation만
# 제외하지만 E2E에 사용한 r4 코드는 navigation, fulltext,
# fulltext_derivation을 제외합니다. 새 fulltext 세대는 구 코드에서 실패합니다.
set -euo pipefail

cd -P "$HOME/DEV"
R4='TRANSFER/fulltext-aligned-nojs-20260923-r4'
GEN='/nfs-root/diamo/8-agent/term-releases/fulltext-repair-20260923T081336Z-31987d27/generation'
SOURCE="$R4/02_BACKEND/semantic_runtime/term_index.py"
TARGET='02_BACKEND/semantic_runtime/term_index.py'
BACKUP='TRANSFER/term_index.py.before-fulltext-fix'

test -d .git
test -f "$SOURCE"
test -f "$R4/06_TOOLS/COMMANDS/genos_fulltext_repin.py"
test -d "$GEN"
if [ -e "$BACKUP" ]; then
  printf '백업 파일이 이미 있습니다: %s\n재실행 전에 현재 코드와 백업을 확인하세요.\n' "$BACKUP" >&2
  exit 1
fi

printf '1/4 기존 term_index.py 백업 및 r4 코드 반영\n'
cp "$TARGET" "$BACKUP"
cp "$SOURCE" "$TARGET"
python3 -m py_compile "$TARGET"

printf '2/4 엔진 manifest 및 PIPELINE.json 코드 pin 갱신\n'
python3 "$R4/06_TOOLS/COMMANDS/genos_fulltext_repin.py" --code-root .

printf '3/4 push할 ~/DEV 코드로 선택된 generation 검증\n'
GEN="$GEN" PYTHONPATH="$PWD/02_BACKEND" python3 - <<'PY'
import os
from lawrag.pipeline.term_serving_bundle import verify_assets

result = verify_assets(os.environ['GEN'])
assert result['verified'] is True
print(result)
PY

printf '4/4 Git 변경 범위 확인\n'
git diff --check
git status --short
git diff --stat -- "$TARGET" 00_SYSTEM/PIPELINE.json 02_BACKEND/ENGINES
printf '\n검증 성공. 변경 파일을 검토한 뒤 commit/push하고 새 코드서빙 리비전을 배포하세요.\n'
printf 'GENOS_GENERATION=%s\n' "$GEN"
printf '배포 후 같은 질문을 다시 실행해 오류가 사라졌는지 확인하세요.\n'
# 실패하면 BACKUP으로 코드 파일을 되돌리고, 배포했다면 이전 리비전으로 복귀합니다.
# 데이터 manifest 해시를 수동으로 바꾸거나 전체 색인을 다시 만들지 않습니다.
